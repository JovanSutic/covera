## Step 7.1 — The list query

**Touch.** `src/lib/api/queries/specialAccess.ts` (create)

**Do.** *(logic mode — the key and the enabled rule are the decisions)*

```ts
import { useQuery } from "@tanstack/react-query";

import { listSpecialAccess } from "@/lib/api/specialAccess";
import { QUERY_ACTIONS } from "@/lib/api/queryKeys";
import { PREFETCH_STALE_TIME_MS } from "./shared";

/**
 * D6/D7: the host's grants for one reservation. Bearer-authorized — a guest has no business
 * enumerating links, and the backend requires a session for this read.
 *
 * The key carries the reservationId, so two reservations' lists cannot collide. There is no
 * `accessId` in the key: this read is never made with a link.
 */
export function specialAccessListQueryKey(reservationId: string) {
  return [...QUERY_ACTIONS.SPECIAL_ACCESS_LIST, reservationId] as const;
}

export function useSpecialAccessList(reservationId: string | undefined) {
  return useQuery({
    queryKey: specialAccessListQueryKey(reservationId ?? ""),
    queryFn: ({ signal }) => {
      if (!reservationId) throw new Error("reservationIdRequired");
      return listSpecialAccess(reservationId, signal);
    },
    enabled: Boolean(reservationId),
    staleTime: PREFETCH_STALE_TIME_MS,
  });
}
```

## Step 7.2 — A grant row's display state

**Touch.** `src/lib/helpers/specialAccessState.ts` (create)

**Do.** *(logic mode — pure, and the reason strings are the decision)*

```ts
import type { SpecialAccess } from "@/api/generated/requests/types.gen";

/**
 * D6: what a listed grant's state is, and the i18n key that says it. One function, so the
 * list and any future surface cannot disagree — the same reason `claimStatus.ts` exists.
 *
 * The order of the checks IS the behaviour: a revoked link that has also expired is
 * "revoked" — that is the fact the host acted on, and the one they can act on again.
 */
export type GrantState = "active" | "used" | "revoked" | "expired";

const LABEL_KEYS: Record<GrantState, string> = {
  active: "reservationPage.links.state.active",
  used: "reservationPage.links.state.used",
  revoked: "reservationPage.links.state.revoked",
  expired: "reservationPage.links.state.expired",
};

export function grantState(grant: SpecialAccess, now: number): GrantState {
  if (grant.revokedAt !== null) return "revoked";
  if (grant.usedAt !== null) return "used";

  const expires = new Date(grant.expiresAt).getTime();

  // An unparseable expiry reads as expired, not as "valid forever" — the safe direction for a
  // credential. This is the one place in the feature where a corrupt date fails CLOSED.
  if (!Number.isFinite(expires) || expires <= now) return "expired";

  return "active";
}

export function grantStateLabelKey(state: GrantState): string {
  return LABEL_KEYS[state];
}

/** D6: only an active grant is worth revoking — the rest are already dead. */
export function isRevocable(state: GrantState): boolean {
  return state === "active";
}
```

## Step 7.3 — Test the grant state

Required: this phase contains testable logic.

**Touch.** `src/lib/helpers/specialAccessState.test.ts` (create)

**Do.**

```ts
import { describe, expect, it } from "vitest";

import { grantState, grantStateLabelKey, isRevocable } from "./specialAccessState";
import type { SpecialAccess } from "@/api/generated/requests/types.gen";

const NOW = Date.parse("2026-06-15T12:00:00.000Z");

function grant(overrides: Partial<SpecialAccess> = {}): SpecialAccess {
  return {
    id: "g1",
    accessId: "abc123",
    scope: "inspection_proof",
    apartmentId: "a1",
    reservationId: "r1",
    incidentItemId: null,
    expiresAt: new Date(NOW + 60 * 60 * 1000).toISOString(),
    revokedAt: null,
    usedAt: null,
    createdAt: new Date(NOW - 60 * 60 * 1000).toISOString(),
    ...overrides,
  };
}

describe("grantState", () => {
  it("is active while unrevoked, unused and unexpired", () => {
    expect(grantState(grant(), NOW)).toBe("active");
  });

  it("is expired once the expiry has passed", () => {
    expect(
      grantState(grant({ expiresAt: new Date(NOW - 1000).toISOString() }), NOW),
    ).toBe("expired");
  });

  it("is used once it has been redeemed", () => {
    expect(grantState(grant({ usedAt: new Date(NOW - 1000).toISOString() }), NOW)).toBe(
      "used",
    );
  });

  it("reports revoked ahead of used and expired", () => {
    const revoked = grant({
      revokedAt: new Date(NOW - 1000).toISOString(),
      usedAt: new Date(NOW - 2000).toISOString(),
      expiresAt: new Date(NOW - 3000).toISOString(),
    });

    expect(grantState(revoked, NOW)).toBe("revoked");
  });

  it("fails closed on an unparseable expiry", () => {
    expect(grantState(grant({ expiresAt: "garbage" }), NOW)).toBe("expired");
  });

  it("treats the exact expiry instant as expired", () => {
    expect(grantState(grant({ expiresAt: new Date(NOW).toISOString() }), NOW)).toBe(
      "expired",
    );
  });
});

describe("the label keys", () => {
  it("maps every state to its own key", () => {
    const keys = (["active", "used", "revoked", "expired"] as const).map(
      grantStateLabelKey,
    );

    expect(new Set(keys).size).toBe(4);
    expect(keys.every((key) => key.startsWith("reservationPage.links.state."))).toBe(true);
  });
});

describe("isRevocable", () => {
  it("is true only for an active grant", () => {
    expect(isRevocable("active")).toBe(true);
    expect(isRevocable("used")).toBe(false);
    expect(isRevocable("revoked")).toBe(false);
    expect(isRevocable("expired")).toBe(false);
  });
});
```

## Step 7.4 — The links section lists and revokes

**Touch.** `src/components/host/ReservationLinksSection.tsx` (edit)

**Do.** *(surface mode — specify, do not transcribe)*

- **Exports.** `export function ReservationLinksSection(props: ReservationLinksSectionProps): JSX.Element` — props unchanged (`reservationId`, `apartmentId`).
- **Consumes.** `useSpecialAccessList` from `@/lib/api/queries/specialAccess`; `specialAccessListQueryKey`; `createSpecialAccess`, `revokeSpecialAccess` from `@/lib/api/specialAccess`; `grantState`, `grantStateLabelKey`, `isRevocable` from `@/lib/helpers/specialAccessState`; `QUERY_ACTIONS`; `absoluteUrl`; `appPaths`, `buildPath`.
- **The `useState` link is deleted.** D6: the list is the source of truth. A link minted this
  session appears in the list because the create mutation invalidates
  `specialAccessListQueryKey(reservationId)` — not because it is held in component state.
- **States.**
  - loading → the existing intro copy plus a spinner row.
  - **empty** → the existing intro copy and the create button (the current `link === null` branch).
  - error → a rose row reading `reservationPage.links.listError`, with the create button still
    offered (a failed list must not block minting).
  - success → one row per grant.
- **Each row** shows: the URL (a readonly input, as today), the state chip from
  `grantStateLabelKey(grantState(grant, now))`, the copy button, and a revoke button **rendered only
  when `isRevocable(state)`**.
- **The URL is rebuilt per row** from `grant.accessId` — the same `absoluteUrl(buildPath(...)) +
  "?accessId=" + grant.accessId` shape the create success handler already uses. Do not store URLs.
- **`now`** is captured once per render in this component (`const now = Date.now()`), which is
  correct here: this component is not the reservation page and has no clock to inherit (invariant 10
  is about one clock per PAGE, and this section is self-contained).
- **Callbacks.** The create mutation's `onSuccess` no longer sets state; it invalidates the list key
  and toasts. The revoke mutation's `onSuccess` invalidates the list key and the reservations list.
- **Pattern to follow.** The component's own current structure — the create/copy/revoke affordances
  and their `ActionButton` usage are already right; only the data source changes.
- **Invariant.** Mutations invalidate what they changed (invariant 7).
