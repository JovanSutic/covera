```ts
import type { Incident, IncidentItem } from "@/api/generated/requests/types.gen";

/**
 * D16: the ONE answer to "what does this status mean and what may be done next".
 * The reservation section, the claim page and the field portal all ask these
 * functions — the same reason `proofWindow.ts` and `reservationActions.ts` exist.
 * The labels are i18n KEYS, resolved by the caller's `t()`.
 */
export type ClaimTone = "neutral" | "warning" | "info" | "success" | "danger";

export interface ClaimCopy {
  tone: ClaimTone;
  labelKey: string;
}

/**
 * D15/blueprint View A. Three of these tones are the blueprint's own colours:
 * Draft = gray, Ready to File = yellow, Submitted to Guest = blue. The three
 * statuses the backend cannot currently produce are mapped anyway (out of scope
 * to OFFER, in scope to RENDER correctly if they ever arrive).
 */
const INCIDENT_STATUS: Record<Incident["status"], ClaimCopy> = {
  draft: { tone: "neutral", labelKey: "incidentPage.status.draft" },
  ready_to_file: { tone: "warning", labelKey: "incidentPage.status.readyToFile" },
  submitted_to_guest: {
    tone: "info",
    labelKey: "incidentPage.status.submittedToGuest",
  },
  escalated_to_aircover: {
    tone: "info",
    labelKey: "incidentPage.status.escalatedToAircover",
  },
  resolved: { tone: "success", labelKey: "incidentPage.status.resolved" },
  unfeasible: { tone: "danger", labelKey: "incidentPage.status.unfeasible" },
};

export function claimStatusCopy(status: Incident["status"]): ClaimCopy {
  return INCIDENT_STATUS[status];
}

/**
 * `documents_pending` has no writer in `covera-hono` today (D15) — it is mapped so
 * an item that arrives in it renders as itself rather than as "unknown".
 */
const ITEM_STATUS: Record<IncidentItem["status"], ClaimCopy> = {
  link_generated: { tone: "warning", labelKey: "incidentPage.itemStatus.linkGenerated" },
  photos_collected: { tone: "info", labelKey: "incidentPage.itemStatus.photosCollected" },
  documents_pending: {
    tone: "warning",
    labelKey: "incidentPage.itemStatus.documentsPending",
  },
  complete: { tone: "success", labelKey: "incidentPage.itemStatus.complete" },
};

export function itemStatusCopy(status: IncidentItem["status"]): ClaimCopy {
  return ITEM_STATUS[status];
}

/**
 * The backend's own `isClaimEditable`: `["draft", "ready_to_file"]`. Named here so
 * the UI's editable surface is the same set the API will accept a write for.
 */
export function isIncidentEditable(incident: Incident): boolean {
  return incident.status === "draft" || incident.status === "ready_to_file";
}

/**
 * The filing surface opens only at `ready_to_file`.
 *
 * The backend's guard on POST /incidents/{id}/confirm-airbnb-filing is wider — it
 * accepts `draft` too — but a draft is by definition a claim with an uncollected
 * item, and `syncIncidentReadiness` is what flips it. Offering "file this now" on a
 * draft would be offering an action whose own precondition the page has not met.
 */
export function canFileClaim(incident: Incident): boolean {
  return incident.status === "ready_to_file";
}

/**
 * D17: the item's field link is live only while the item is still in
 * `link_generated` AND the token has not passed. `recomputeItemStatus` burns the
 * token (`tokenExpiresAt = new Date()`) the moment the status leaves
 * `link_generated`, so the status check is not redundant with the date check — it
 * is the half that closes the link after a successful submit.
 *
 * An unparseable date reads as NOT live, matching `proofWindow.ts`'s rule that a
 * corrupt date is "no such instant" rather than epoch 0.
 */
export function isItemLinkLive(item: IncidentItem, now: number): boolean {
  if (item.status !== "link_generated") return false;

  const expires = new Date(item.tokenExpiresAt).getTime();

  return Number.isFinite(expires) && expires > now;
}

/**
 * D17: `reissueToken` matches only `status = 'link_generated'`, so a reissue
 * control anywhere else would call an endpoint that cannot succeed. This is
 * therefore "the link lapsed while still collectable" — and nothing else.
 */
export function canReissueItemLink(item: IncidentItem, now: number): boolean {
  return item.status === "link_generated" && !isItemLinkLive(item, now);
}

/**
 * D11: there is NO incident-by-reservation endpoint — `GET /incidents` requires an
 * `apartmentId` and returns a paginated list. Both host surfaces resolve "the claim
 * for this reservation" here, so they cannot match differently.
 *
 * Returns the first match, which under the list's `createdAt desc` sort is the
 * newest. One reservation can only have one incident (the backend refuses a second
 * with `INCIDENT_ALREADY_EXISTS`), so a second element would be a backend bug, not
 * a choice to make here.
 */
export function findIncidentForReservation<T extends { reservationId: string }>(
  incidents: readonly T[],
  reservationId: string,
): T | null {
  return incidents.find((incident) => incident.reservationId === reservationId) ?? null;
}
```

### Step 1.3 — Create the deadline arithmetic

**Touch.** `src/lib/helpers/claimDeadline.ts` (create)

**Do.**

```ts
import type { Incident } from "@/api/generated/requests/types.gen";

/**
 * D32: the deadline arithmetic, defined once. `now` is a PARAMETER, never
 * `Date.now()` inside — the same rule `getProofWindowState` follows, so the
 * behaviour is testable without faking timers and the caller owns the clock.
 *
 * The backend owns both deadlines: `confirmAirbnbFiling` sets
 * `guestResponseDeadlineAt = NOW() + 24h`, and item creation sets
 * `aircoverEscalationDeadlineAt = effectiveCheckOut + 30 days`
 * (`ESCALATION_DAYS`, `covera-hono/src/modules/incidents/incidents.service.ts`).
 * Nothing here recomputes them; it only reads them.
 */
export const GUEST_RESPONSE_WINDOW_MS = 24 * 60 * 60 * 1000;

export interface ClaimDeadlineState {
  /** Epoch ms of `guestResponseDeadlineAt`, or `null` when the claim is unfiled. */
  guestDeadlineTime: number | null;
  /** Milliseconds remaining; `0` once passed. Never negative. */
  msUntilGuestDeadline: number;
  /** Whole minutes remaining, rounded up — the countdown's display unit. */
  minutesUntilGuestDeadline: number;
  /** True only when there IS a deadline and it has passed. */
  isGuestResponseOverdue: boolean;
  /** Epoch ms of `aircoverEscalationDeadlineAt`, or `null` when unusable. */
  aircoverDeadlineTime: number | null;
  /** Whole days remaining, rounded up; `0` once passed. */
  daysUntilAircover: number;
  isAircoverOverdue: boolean;
}

/**
 * `Number.isFinite`, not a truthiness check: `new Date("garbage").getTime()` is
 * NaN, and a corrupt date must read as "no such instant" — never as epoch 0, which
 * would report every claim as decades overdue.
 */
function toEpoch(value: string | null): number | null {
  if (!value) return null;

  const parsed = new Date(value).getTime();

  return Number.isFinite(parsed) ? parsed : null;
}

export function getClaimDeadlineState(incident: Incident, now: number): ClaimDeadlineState {
  const guestDeadlineTime = toEpoch(incident.guestResponseDeadlineAt);
  const aircoverDeadlineTime = toEpoch(incident.aircoverEscalationDeadlineAt);

  const msUntilGuestDeadline =
    guestDeadlineTime === null ? 0 : Math.max(0, guestDeadlineTime - now);
  const msUntilAircover =
    aircoverDeadlineTime === null ? 0 : Math.max(0, aircoverDeadlineTime - now);

  return {
    guestDeadlineTime,
    msUntilGuestDeadline,
    // Rounded UP, not down: with 30 seconds left "0 minutes" reads as already over.
    minutesUntilGuestDeadline: Math.ceil(msUntilGuestDeadline / 60_000),
    isGuestResponseOverdue: guestDeadlineTime !== null && guestDeadlineTime <= now,
    aircoverDeadlineTime,
    daysUntilAircover: Math.ceil(msUntilAircover / (24 * 60 * 60 * 1000)),
    isAircoverOverdue: aircoverDeadlineTime !== null && aircoverDeadlineTime <= now,
  };
}

/**
 * The countdown's own numbers, so the component never divides by 60 000. A partial
 * hour is shown as an hour — "23h 0m" with 22h 40m left would understate the time
 * the host has, which is the opposite of what this feature is for.
 */
export function countdownParts(ms: number): { hours: number; minutes: number } {
  const clamped = Math.max(0, ms);

  return {
    hours: Math.floor(clamped / (60 * 60 * 1000)),
    minutes: Math.ceil((clamped % (60 * 60 * 1000)) / 60_000) % 60,
  };
}

/**
 * D18-adjacent: how the filing banner chooses its urgency. Three bands, because
 * the copy differs at each — an emergency check-in, a deadline inside a day, and
 * anything longer.
 */
export type ClaimUrgency = "emergency" | "soon" | "normal";

export function claimUrgency(incident: Incident, now: number): ClaimUrgency {
  if (incident.isEmergencyMode) return "emergency";

  const { msUntilGuestDeadline } = getClaimDeadlineState(incident, now);

  return msUntilGuestDeadline > 0 && msUntilGuestDeadline < GUEST_RESPONSE_WINDOW_MS
    ? "soon"
    : "normal";
}
```

```ts
function incident(overrides: Partial<Incident> = {}): Incident {

    reservationId: "reservation-1",
    status: "submitted_to_guest",
    isEmergencyMode: false,

    resolutionCenterSubmittedAt: "2026-11-08T11:00:00.000Z",
    guestResponseDeadlineAt: "2026-11-09T11:00:00.000Z",
    aircoverEscalationDeadlineAt: "2026-12-08T11:00:00.000Z",

    updatedAt: "2026-11-08T11:00:00.000Z",

describe("getClaimDeadlineState", () => {
  it("counts down while the guest window is open", () => {
    const state = getClaimDeadlineState(incident(), NOW);

    expect(state.guestDeadlineTime).toBe(Date.parse("2026-11-09T11:00:00.000Z"));
    expect(state.msUntilGuestDeadline).toBe(23 * 60 * 60 * 1000);
    expect(state.minutesUntilGuestDeadline).toBe(23 * 60);
    expect(state.isGuestResponseOverdue).toBe(false);
  });

    const state = getClaimDeadlineState(
      incident({ guestResponseDeadlineAt: "2026-11-08T11:00:00.000Z" }),
      NOW,
    );

    expect(state.msUntilGuestDeadline).toBe(0);
    expect(state.minutesUntilGuestDeadline).toBe(0);
    expect(state.isGuestResponseOverdue).toBe(true);
  });

  it("treats the deadline instant itself as overdue", () => {
    const state = getClaimDeadlineState(
      incident({ guestResponseDeadlineAt: "2026-11-08T12:00:00.000Z" }),
      NOW,
    );

    expect(state.isGuestResponseOverdue).toBe(true);
  });

  it("reports no guest deadline on an unfiled claim", () => {
    const state = getClaimDeadlineState(
      incident({ status: "draft", guestResponseDeadlineAt: null }),
      NOW,

    expect(state.isGuestResponseOverdue).toBe(false);
    expect(state.msUntilGuestDeadline).toBe(0);
  });

  it("treats a corrupt guest deadline as no deadline, not as epoch 0", () => {
    const state = getClaimDeadlineState(
      incident({ guestResponseDeadlineAt: "garbage" }),
      NOW,

    expect(state.guestDeadlineTime).toBeNull();
    expect(state.isGuestResponseOverdue).toBe(false);
  });

  it("counts AirCover days up, not down", () => {
    // 30 days minus one hour: 30 to the day, because a partial day still exists.
    const state = getClaimDeadlineState(incident(), NOW);

    expect(state.daysUntilAircover).toBe(30);
    expect(state.isAircoverOverdue).toBe(false);

  it("flags the AirCover deadline overdue and clamps its days to zero", () => {
    const state = getClaimDeadlineState(
      incident({ aircoverEscalationDeadlineAt: "2026-11-01T00:00:00.000Z" }),
      NOW,
    );

    expect(state.daysUntilAircover).toBe(0);
    expect(state.isAircoverOverdue).toBe(true);
  });

  it("splits a 23-hour window into hours and minutes", () => {
    expect(countdownParts(23 * 60 * 60 * 1000)).toEqual({ hours: 23, minutes: 0 });
  });

  it("rounds a partial hour up rather than reporting a whole hour lost", () => {
    // 22h 40m left must not read as "22h 0m" — understating the host's time is the

    expect(countdownParts(22 * 60 * 60 * 1000 + 40 * 60 * 1000)).toEqual({
      hours: 22,
      minutes: 40,
    });
    expect(countdownParts(30_000)).toEqual({ hours: 0, minutes: 1 });
  });

  it("returns zeroes for zero and for a negative input", () => {
    expect(countdownParts(0)).toEqual({ hours: 0, minutes: 0 });
    expect(countdownParts(-5_000)).toEqual({ hours: 0, minutes: 0 });
  });
});

describe("claimUrgency", () => {
  it("is emergency whenever the backend says so, whatever the clock says", () => {
    expect(claimUrgency(incident({ isEmergencyMode: true }), NOW)).toBe("emergency");
  });

  it("is soon while the guest window is under a day and still open", () => {
    const soon = incident({
      guestResponseDeadlineAt: new Date(NOW + GUEST_RESPONSE_WINDOW_MS - 60_000).toISOString(),
    });

    expect(claimUrgency(soon, NOW)).toBe("soon");
  });

  it("is normal with no deadline and with an overdue one", () => {
    expect(
      claimUrgency(incident({ status: "draft", guestResponseDeadlineAt: null }), NOW),
    ).toBe("normal");
    expect(
      claimUrgency(
        incident({ guestResponseDeadlineAt: "2026-11-08T11:00:00.000Z" }),
        NOW,
      ),
    ).toBe("normal");
  });
});
```

```ts
import type { IncidentItem } from "@/api/generated/requests/types.gen";

/**
 * D21: the evidence vocabulary. `PHOTO_EVIDENCE_FOR_TYPE` MIRRORS the backend's
 * constant of the same name in `covera-hono/src/modules/incidents/`:
 *
 *   damage: ["damage_photo_wide", "damage_photo_closeup"]
 *   theft:  ["theft_empty_space_photo"]
 *
 * It has to be duplicated: `getIncidentsById` returns `IncidentItem`, which carries
 * no `requiredEvidenceTypes` field — only the FIELD payload does. If that backend
 * constant changes, this one must change with it. `incidentEvidence.test.ts` names
 * it in the test that pins these values, so the dependency is visible.
 */
export type EvidenceType = IncidentItem["evidence"][number]["evidenceType"];

const PHOTO_EVIDENCE_FOR_TYPE: Record<IncidentItem["type"], readonly EvidenceType[]> = {
  damage: ["damage_photo_wide", "damage_photo_closeup"],
  theft: ["theft_empty_space_photo"],
};

/** The four document types. The field portal never submits these (out of scope). */
export const DOCUMENT_EVIDENCE_TYPES: readonly EvidenceType[] = [
  "original_purchase_receipt",
  "replacement_retail_quote",
  "handyman_invoice",
  "police_report",
];

/** What the host must be guided to shoot for an item of this type, in order. */
export function requiredPhotoTypes(type: IncidentItem["type"]): readonly EvidenceType[] {
  return PHOTO_EVIDENCE_FOR_TYPE[type];
}

export function isPhotoEvidence(type: EvidenceType): boolean {
  return !DOCUMENT_EVIDENCE_TYPES.includes(type);
}

/**
 * The label for every one of the seven types, as an i18n key. Total on purpose: a
 * `Record` over the union means a type the backend adds is a compile error here
 * rather than a raw key on screen.
 */
const EVIDENCE_LABEL_KEYS: Record<EvidenceType, string> = {
  damage_photo_wide: "incidentPage.evidence.damagePhotoWide",
  damage_photo_closeup: "incidentPage.evidence.damagePhotoCloseup",
  theft_empty_space_photo: "incidentPage.evidence.theftEmptySpacePhoto",
  original_purchase_receipt: "incidentPage.evidence.originalPurchaseReceipt",
  replacement_retail_quote: "incidentPage.evidence.replacementRetailQuote",
  handyman_invoice: "incidentPage.evidence.handymanInvoice",
  police_report: "incidentPage.evidence.policeReport",
};

export function evidenceLabelKey(type: EvidenceType): string {
  return EVIDENCE_LABEL_KEYS[type];
}

/**
 * Evidence that still counts. A `soft_deleted` row is the app's delete-with-history
 * convention (`RoomPhotoItem` reads `status === "active" && !deletedAt` the same
 * way, for the same reason) — it must not be counted as collected, or a host who
 * deleted a bad photo would see the item reported as complete.
 */
export function activeEvidence(item: IncidentItem): IncidentItem["evidence"] {
  return item.evidence.filter(
    (evidence) => evidence.status === "active" && evidence.deletedAt === null,
  );
}

/** How many ACTIVE photos of this exact type the item already carries. */
export function activePhotoCount(item: IncidentItem, type: EvidenceType): number {
  return activeEvidence(item).filter(
    (evidence) => evidence.kind === "photo" && evidence.evidenceType === type,
  ).length;
}

/**
 * What is still missing, as data — the list of required types with none collected.
 * The host's checklist and the "is this item done" answer both read this rather
 * than counting themselves.
 */
export function missingPhotoTypes(item: IncidentItem): EvidenceType[] {
  return requiredPhotoTypes(item.type).filter(
    (type) => activePhotoCount(item, type) === 0,
  );
}

/** True when every required photo type has at least one active photo. */
export function hasAllRequiredPhotos(item: IncidentItem): boolean {
  return missingPhotoTypes(item).length === 0;
}
```
