# Phase 2 — Money and the filing text

**Prerequisite:** phase 1 (this phase's message builder consumes phase 1's vocabularies)
**Gate:** `npm run gate`

Two pure helpers. Nothing renders, nothing requests. After this phase the figures and the text block
that goes into Airbnb's request box are both computed in one place and testable without a DOM.

```ts
import type { Incident, IncidentItem } from "@/api/generated/requests/types.gen";

/**
 * D31: money formatting, defined once. Every value this feature shows arrives as an
 * INTEGER number of cents — `placeholderTotalAmountCents`,
 * `finalPayoutRequestedCents`, `estimatedPlaceholderValueCents`,
 * `approximateValueCents`. Nothing in the repo formats cents today (`ApartmentHeader`
 * prints the bare currency code), so this is the first and only place it happens, and
 * the figures module, the item rows and the copy block all read it.
 *
 * `EUR` is the default because every existing apartment in this app is priced in EUR
 * (`ApartmentHeader` renders `apartment.currency` beside a euro-formatted total).
 */
export const DEFAULT_CURRENCY = "EUR";

export function formatCents(cents: number, currency: string = DEFAULT_CURRENCY): string {
  if (!Number.isFinite(cents)) return "";

  return new Intl.NumberFormat(undefined, {
    style: "currency",
    currency,
  }).format(cents / 100);
}

export interface ClaimTotals {
  /** What the items' provisional valuations add up to, in cents. */
  itemsSubtotalCents: number;
  /** What the host is asking for: the filed figure, or the subtotal while unfiled. */
  requestedCents: number;
  /** True once the host has set an explicit figure, so the copy stops saying "provisional". */
  hasFinalFigure: boolean;
}

/**
 * The aggregate the blueprint's "Aggregate Figures Module" asks for. `Incident`
 * already carries `placeholderTotalAmountCents` from the backend — but it is the
 * value at the last write, so it is NOT recomputed here and NOT trusted as the
 * subtotal: an item added a second ago is in `items` before it is in that column.
 * The sum is taken from the items the caller is looking at, and the header's figure
 * is only used once it is authoritative (`finalPayoutRequestedCents`).
 */
export function claimTotals(
  incident: Pick<Incident, "finalPayoutRequestedCents">,
  items: readonly Pick<IncidentItem, "estimatedPlaceholderValueCents">[],
): ClaimTotals {
  const itemsSubtotalCents = items.reduce(
    (total, item) => total + (item.estimatedPlaceholderValueCents || 0),
    0,
  );

  const hasFinalFigure = incident.finalPayoutRequestedCents > 0;

  return {
    itemsSubtotalCents,
    requestedCents: hasFinalFigure
      ? incident.finalPayoutRequestedCents
      : itemsSubtotalCents,
    hasFinalFigure,
  };
}

/**
 * "12.34" or "12,34" → 1234 cents. `null` when the input is not a usable amount.
 *
 * Accepts a comma as the decimal separator because `de`, `tr` and `sr-Latn` users
 * will type one, and the alternative — a number input that silently discards it —
 * is a bug the host cannot see. Empty input is `null`, not `0`: "I have not said
 * yet" and "it is worth nothing" are different answers, and only one of them should
 * reach the API.
 */
export function parseCentsFromInput(value: string): number | null {
  const normalised = value.trim().replace(",", ".");

  if (normalised === "") return null;

  const parsed = Number(normalised);

  if (!Number.isFinite(parsed) || parsed < 0) return null;

  return Math.round(parsed * 100);
}
```

```ts
import {
  DEFAULT_CURRENCY,

describe("formatCents", () => {
  it("renders zero rather than an empty string", () => {

    expect(formatCents(12345)).toContain("123");

  it("groups thousands", () => {
    expect(formatCents(1234567)).toMatch(/12[.,  ]345/);
  });

  it("honours an explicit currency", () => {
    expect(formatCents(1000, "USD")).toContain("$");
  });

  it("defaults to EUR", () => {
    expect(formatCents(1000)).toContain("€");
    expect(DEFAULT_CURRENCY).toBe("EUR");
  });

  it("returns an empty string for a non-finite value rather than throwing", () => {
    expect(formatCents(Number.NaN)).toBe("");
  });

describe("claimTotals", () => {
  it("sums the items", () => {
    expect(
      claimTotals({ finalPayoutRequestedCents: 0 }, [
        { estimatedPlaceholderValueCents: 5000 },
        { estimatedPlaceholderValueCents: 2500 },

    ).toEqual({
      itemsSubtotalCents: 7500,
      requestedCents: 7500,
      hasFinalFigure: false,
    });
  });

  it("is zero on a claim with no items", () => {
    expect(claimTotals({ finalPayoutRequestedCents: 0 }, [])).toEqual({
      itemsSubtotalCents: 0,
      requestedCents: 0,
      hasFinalFigure: false,
    });
  });

  it("prefers the filed figure once the host has set one", () => {
    expect(
      claimTotals({ finalPayoutRequestedCents: 9000 }, [
        { estimatedPlaceholderValueCents: 5000 },

    ).toEqual({
      itemsSubtotalCents: 5000,
      requestedCents: 9000,
      hasFinalFigure: true,
    });
  });

  it("treats a missing item value as zero rather than as NaN", () => {
    const totals = claimTotals({ finalPayoutRequestedCents: 0 }, [
      { estimatedPlaceholderValueCents: undefined as unknown as number },
      { estimatedPlaceholderValueCents: 1000 },
    ]);

    expect(totals.itemsSubtotalCents).toBe(1000);
  });
});

describe("parseCentsFromInput", () => {
  it("parses a dot decimal", () => {
    expect(parseCentsFromInput("12.34")).toBe(1234);
  });

  it("parses a comma decimal, which is what de, tr and sr-Latn users type", () => {
    expect(parseCentsFromInput("12,34")).toBe(1234);
  });

  it("parses a whole number", () => {
    expect(parseCentsFromInput("250")).toBe(25000);
  });

  it("rounds to whole cents rather than carrying a fraction", () => {
    expect(parseCentsFromInput("1.005")).toBe(101);
  });

  it("returns null for empty input — 'not said yet' is not 'zero'", () => {
    expect(parseCentsFromInput("")).toBeNull();
    expect(parseCentsFromInput("   ")).toBeNull();
  });

  it("returns null for anything that is not a positive amount", () => {
    expect(parseCentsFromInput("abc")).toBeNull();
    expect(parseCentsFromInput("-5")).toBeNull();
  });
});
```

### Step 2.3 — Create the resolution-centre message builder

**Touch.** `src/lib/helpers/claimMessage.ts` (create)

**Do.**

```ts
import type { IncidentItem } from "@/api/generated/requests/types.gen";

import { evidenceLabelKey } from "./incidentEvidence";
import { formatCents } from "./claimMoney";
import { requiredPhotoTypes } from "./incidentEvidence";

/**
 * D25: the text block the host pastes into Airbnb's resolution-centre description
 * box. Pure, so the part that can actually be wrong — which items, in what order,
 * with which figures — is testable without a DOM or a clipboard.
 *
 * It is deliberately NOT translated. The host pastes this into an English-language
 * Airbnb form that Airbnb's own support reads; translating the item names into
 * Turkish would produce a claim nobody at Airbnb can process. The surrounding
 * labels ARE translated, because those are the host's own screen. That split is the
 * reason this function takes a `t` and uses it only for the fixed sentences.
 *
 * The per-item lines are assembled from the ASSET's own name and room, which are
 * free text the host wrote when they created the asset — not enum keys — so they go
 * in verbatim.
 */
export interface ClaimMessageLabels {
  /** `incidentPage.message.header`, with `{{ apartmentName }}`. */
  header: string;
  /** `incidentPage.message.itemLine`, with `{{ index }}`, `{{ asset }}`, `{{ room }}`, `{{ type }}`, `{{ value }}`. */
  itemLine: string;
  /** `incidentPage.message.totalLine`, with `{{ amount }}`. */
  totalLine: string;
  /** `incidentPage.message.footer`. */
  footer: string;
  /** `incidentPage.itemType.damage` / `.theft`. */
  typeLabels: Record<IncidentItem["type"], string>;
  /** `incidentPage.message.missingEvidence`, with `{{ evidence }}`. */
  missingEvidence: string;
}

export interface BuildClaimMessageParams {
  guestName: string;
  apartmentName: string;
  items: readonly IncidentItem[];
  requestedCents: number;
  /** The caller's `t`, bound to the `host` namespace. */
  t: (key: string, options?: Record<string, unknown>) => string;
  labels: ClaimMessageLabels;
}

/**
 * The count of each required view an item already has, as the "what is attached"
 * clause. Reads `activePhotoCount`'s rule through `item.evidence` directly rather
 * than importing the counter, so an item with a soft-deleted photo is described the
 * same way the item row describes it.
 */
function attachedClause(
  item: IncidentItem,
  t: BuildClaimMessageParams["t"],
  labels: ClaimMessageLabels,
): string {
  const attached = requiredPhotoTypes(item.type)
    .map((type) =>
      item.evidence.some(
        (evidence) =>
          evidence.status === "active" &&
          evidence.deletedAt === null &&
          evidence.evidenceType === type,
      )
        ? t(evidenceLabelKey(type))
        : null,
    )
    .filter((label): label is string => label !== null);

  return attached.length === 0
    ? t(labels.missingEvidence, { evidence: "" })
    : attached.join(", ");
}

export function buildClaimMessage({
  guestName,
  apartmentName,
  items,
  requestedCents,
  t,
  labels,
}: BuildClaimMessageParams): string {
  const lines: string[] = [
    t(labels.header, { guestName, apartmentName }),
    "",
  ];

  items.forEach((item, index) => {
    lines.push(
      t(labels.itemLine, {
        index: index + 1,
        asset: item.asset.name,
        // `roomLocations` lives in the `assets` namespace, not `host` — and this
        // file's `t` is bound to `host`, so an unqualified key would render the raw
        // key string. `ReservationPhotoGallery` carries exactly this prefix for
        // exactly this reason.
        room: t(`assets:roomLocations.${item.asset.roomLocation}`, {
          defaultValue: item.asset.roomLocation,
        }),
        type: t(labels.typeLabels[item.type]),
        value: formatCents(item.estimatedPlaceholderValueCents),
        attached: attachedClause(item, t, labels),
      }),
    );
  });

  lines.push("");
  lines.push(t(labels.totalLine, { amount: formatCents(requestedCents) }));
  lines.push("");
  lines.push(t(labels.footer));

  return lines.join("\n");
}
```

```ts
import { describe, expect, it, vi } from "vitest";

import { buildClaimMessage } from "./claimMessage";
import type { ClaimMessageLabels } from "./claimMessage";
import type { IncidentItem } from "@/api/generated/requests/types.gen";

/**
 * A `t` that returns the key plus its interpolations, so the test can assert WHICH
 * keys were asked for as well as what the result contains. `test-setup.ts` mocks
 * react-i18next so `t(key) => key`; this stub does the same but keeps the values,
 * which is what makes the item lines checkable.
 */
function stubT() {
  return vi.fn((key: string, options?: Record<string, unknown>) => {
    if (!options) return key;

    return Object.entries(options).reduce(
      (text, [name, value]) => text.replace(`{{ ${name} }}`, String(value)),
      key,
    );
  });
}

const LABELS: ClaimMessageLabels = {
  header: "Claim {{ guestName }} {{ apartmentName }}",
  itemLine: "{{ index }}. {{ asset }} ({{ room }}) — {{ type }} — {{ value }} — {{ attached }}",
  totalLine: "Total {{ amount }}",
  footer: "footer",
  typeLabels: { damage: "incidentPage.itemType.damage", theft: "incidentPage.itemType.theft" },
  missingEvidence: "missing {{ evidence }}",
};

let sequence = 0;

function item(overrides: Partial<IncidentItem> = {}): IncidentItem {
  sequence += 1;

  return {
    id: `item-${sequence}`,
    incidentId: "incident-1",
    assetId: `asset-${sequence}`,
    type: "damage",
    status: "link_generated",
    tokenExpiresAt: "2026-11-08T20:00:00.000Z",
    estimatedPlaceholderValueCents: 5000,
    descriptionByCleaner: null,
    createdAt: "2026-11-08T12:00:00.000Z",
    updatedAt: "2026-11-08T12:00:00.000Z",
    asset: {
      id: `asset-${sequence}`,
      name: "Bedroom Curtains",
      roomLocation: "BEDROOM_PRIMARY",
    } as IncidentItem["asset"],
    evidence: [],
    ...overrides,
  };
}

describe("buildClaimMessage", () => {
  it("names the guest and the apartment", () => {
    const message = buildClaimMessage({
      guestName: "Ada Lovelace",
      apartmentName: "Villa Aurora",
      items: [],
      requestedCents: 0,
      t: stubT(),
      labels: LABELS,
    });

    expect(message).toContain("Ada Lovelace");
    expect(message).toContain("Villa Aurora");
  });

  it("lists every item exactly once, numbered in order", () => {
    const message = buildClaimMessage({
      guestName: "Ada",
      apartmentName: "Villa",
      items: [
        item({ asset: { name: "TV" } as IncidentItem["asset"] }),
        item({ asset: { name: "Sofa" } as IncidentItem["asset"] }),
      ],
      requestedCents: 10000,
      t: stubT(),
      labels: LABELS,
    });

    expect(message.match(/1\. TV/g)).toHaveLength(1);
    expect(message.match(/2\. Sofa/g)).toHaveLength(1);
  });

  it("carries the item's value and the requested total", () => {
    const message = buildClaimMessage({
      guestName: "Ada",
      apartmentName: "Villa",
      items: [item({ estimatedPlaceholderValueCents: 12345 })],
      requestedCents: 99999,
      t: stubT(),
      labels: LABELS,
    });

    expect(message).toContain("123");
    expect(message).toContain("999");
  });

  it("says which views are attached for a damage item", () => {
    const message = buildClaimMessage({
      guestName: "Ada",
      apartmentName: "Villa",
      items: [
        item({
          type: "damage",
          evidence: [
            {
              id: "e1",
              incidentItemId: "item-1",
              evidenceType: "damage_photo_wide",
              storageKey: "k1",
              exifCapturedAt: null,
              exifLatitude: null,
              exifLongitude: null,
              status: "active",
              deletedAt: null,
              createdAt: "2026-11-08T12:00:00.000Z",
              kind: "photo",
            },
          ],
        }),
      ],
      requestedCents: 5000,
      t: stubT(),
      labels: LABELS,
    });

    expect(message).toContain("incidentPage.evidence.damagePhotoWide");
    expect(message).not.toContain("incidentPage.evidence.damagePhotoCloseup");
  });

  it("does not count a soft-deleted photo as attached", () => {
    const message = buildClaimMessage({
      guestName: "Ada",
      apartmentName: "Villa",
      items: [
        item({
          evidence: [
            {
              id: "e1",
              incidentItemId: "item-1",
              evidenceType: "damage_photo_wide",
              storageKey: "k1",
              exifCapturedAt: null,
              exifLatitude: null,
              exifLongitude: null,
              status: "soft_deleted",
              deletedAt: "2026-11-08T13:00:00.000Z",
              createdAt: "2026-11-08T12:00:00.000Z",
              kind: "photo",
            },
          ],
        }),
      ],
      requestedCents: 0,
      t: stubT(),
      labels: LABELS,
    });

    expect(message).not.toContain("incidentPage.evidence.damagePhotoWide");
    expect(message).toContain("missing");
  });

  it("marks a theft item's missing view as the empty-space photo, not a damage photo", () => {
    const t = stubT();

    buildClaimMessage({
      guestName: "Ada",
      apartmentName: "Villa",
      items: [item({ type: "theft" })],
      requestedCents: 0,
      t,
      labels: LABELS,
    });

    const asked = t.mock.calls.map(([key]) => key);

    expect(asked).toContain("incidentPage.evidence.theftEmptySpacePhoto");
    expect(asked).not.toContain("incidentPage.evidence.damagePhotoWide");
  });

  it("produces a five-item message with one line per item", () => {
    const items = Array.from({ length: 5 }, (_, index) =>
      item({ asset: { name: `Asset ${index + 1}` } as IncidentItem["asset"] }),
    );

    const message = buildClaimMessage({
      guestName: "Ada",
      apartmentName: "Villa",
      items,
      requestedCents: 25000,
      t: stubT(),
      labels: LABELS,
    });

    for (let index = 1; index <= 5; index += 1) {
      expect(message).toContain(`${index}. Asset ${index}`);
    }
  });

  it("returns a usable string on a claim with no items", () => {
    const message = buildClaimMessage({
      guestName: "Ada",
      apartmentName: "Villa",
      items: [],
      requestedCents: 0,
      t: stubT(),
      labels: LABELS,
    });

    expect(typeof message).toBe("string");
    expect(message.trim().length).toBeGreaterThan(0);
    expect(message).not.toContain("undefined");
  });
});
```
