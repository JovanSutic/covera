# Phase 3 — The three reads

**Prerequisite:** none (independent of phases 1–2)
**Gate:** `npm run gate`

Registers the three prefetchable reads. This is the phase with the most mechanical traps in the whole
plan, and all of them fail loudly except one:

- `registry.ts` names each resource, and `resolveResource` throws at import for a name that is not
  there — so phase 4's routes cannot be written before this lands.
- `manifest.ts` resolves a hook name to a resource **by object identity at import** and throws when
  the pair does not line up.
- `registry.test.ts` calls every resource's `key({})`.
- **The silent one:** `QUERY_HOOKS` is built from `manifest.ts`'s `MODULES` map. A query module that
  is registered in `QUERY_RESOURCES` but missing from `MODULES` has its hooks **invisible** to
  `prefetchCoverage.test.ts`, which then passes vacuously — a page that reads the new hook and
  declares nothing would go green. Step 3.6 is that step, and the test in step 3.7 pins `QUERY_HOOKS`
  so it cannot silently shrink.

### Step 3.2 — Create the incident list module

**Touch.** `src/lib/api/queries/incidents.ts` (create)

**Do.**

```ts
import { getIncidents } from "@/api/generated/requests/sdk.gen";
import { withAuth } from "@/lib/api/api";
import { QUERY_ACTIONS } from "@/lib/api/queryKeys";
import { useQuery } from "@tanstack/react-query";
import { PREFETCH_STALE_TIME_MS } from "./shared";
import type { ResourceDefinition } from "./shared";

/**
 * D14: one page, the maximum the API allows. Paging the claim list is out of scope,
 * and the sort is `createdAt desc` so the newest claims — the ones a host is acting
 * on — are on page 1. An apartment with more than 100 incidents could therefore hide
 * an older one from this surface; that is recorded as a deviation in the PR.
 */
export const INCIDENTS_DEFAULT_PAGE = 1;
export const INCIDENTS_DEFAULT_LIMIT = 100;

export interface IncidentsParams {
  apartmentId: string;
  page?: number;
  limit?: number;
}

/**
 * `apartmentId` is REQUIRED by the API (`GetIncidentsData.query.apartmentId`) and is
 * the only filter this feature uses. The composite key keeps page and limit in the
 * identity so a future caller with different paging cannot collide with this one.
 *
 * The `?? ""` fallback is not cosmetic: `registry.test.ts` calls every resource's
 * `key({})`, and a key builder that threw on missing params would fail that check. A
 * key is a cache address, not a validation — the fetch below is what refuses.
 */
export function incidentsQueryKey({
  apartmentId,
  page = INCIDENTS_DEFAULT_PAGE,
  limit = INCIDENTS_DEFAULT_LIMIT,
}: IncidentsParams) {
  return [...QUERY_ACTIONS.INCIDENTS_GET_BY_APARTMENT, apartmentId ?? "", page, limit] as const;
}

export async function fetchIncidents(
  {
    apartmentId,
    page = INCIDENTS_DEFAULT_PAGE,
    limit = INCIDENTS_DEFAULT_LIMIT,
  }: IncidentsParams,
  signal?: AbortSignal,
) {
  const config = await withAuth(signal ? { signal } : {});
  const response = await getIncidents({
    ...config,
    query: { apartmentId, page, limit, sortBy: "createdAt", order: "desc" },
  });

  return response.data;
}

export function useIncidents(params: IncidentsParams) {
  return useQuery({
    queryKey: incidentsQueryKey(params),
    queryFn: ({ signal }) => {
      if (!params.apartmentId) throw new Error("apartmentIdRequired");
      return fetchIncidents(params, signal);
    },
    enabled: Boolean(params.apartmentId),
    staleTime: PREFETCH_STALE_TIME_MS,
  });
}

export const incidentsResource: ResourceDefinition = {
  action: QUERY_ACTIONS.INCIDENTS_GET_BY_APARTMENT,
  key: (params: IncidentsParams) => incidentsQueryKey(params),
  fetch: (params: IncidentsParams, { signal }) => fetchIncidents(params, signal),
};
```

### Step 3.3 — Create the incident detail module

**Touch.** `src/lib/api/queries/incident.ts` (create)

**Do.**

```ts
import { getIncidentsById } from "@/api/generated/requests/sdk.gen";
import { withAuth } from "@/lib/api/api";
import { QUERY_ACTIONS } from "@/lib/api/queryKeys";
import { useQuery } from "@tanstack/react-query";
import { PREFETCH_STALE_TIME_MS } from "./shared";
import type { ResourceDefinition } from "./shared";

export interface IncidentParams {
  id: string;
}

export function incidentQueryKey(id: string) {
  return [...QUERY_ACTIONS.INCIDENTS_GET_ID, id ?? ""] as const;
}

/**
 * The detail read, and the one that carries the items: `IncidentWithItems`. Every
 * host surface that renders an item reads it from here rather than from the list,
 * because the two are the same shape but only this key is invalidated by an item
 * write — and after phase 7 the field portal writes items from another surface
 * entirely.
 */
export async function fetchIncident(id: string, signal?: AbortSignal) {
  const config = await withAuth(signal ? { signal } : {});
  const response = await getIncidentsById({ ...config, path: { id } });

  return response.data;
}

/**
 * `enabled` mirrors `useInspection`'s signature: without it a caller passing
 * `undefined` gets `isLoading: true` forever, because `enabled: Boolean(undefined)`
 * is false and the query never runs and never resolves.
 */
export function useIncident(id: string | undefined, enabled = true) {
  return useQuery({
    queryKey: incidentQueryKey(id ?? ""),
    queryFn: ({ signal }) => {
      if (!id) throw new Error("incidentIdRequired");
      return fetchIncident(id, signal);
    },
    enabled: Boolean(id) && enabled,
    staleTime: PREFETCH_STALE_TIME_MS,
  });
}

export const incidentResource: ResourceDefinition = {
  action: QUERY_ACTIONS.INCIDENTS_GET_ID,
  key: ({ id }: IncidentParams) => incidentQueryKey(id),
  fetch: ({ id }: IncidentParams, { signal }) => fetchIncident(id, signal),
};
```

### Step 3.4 — Create the anonymous field module

**Touch.** `src/lib/api/queries/fieldIngest.ts` (create)

**Do.**

```ts
import { getFieldIngestByToken } from "@/api/generated/requests/sdk.gen";
import { QUERY_ACTIONS } from "@/lib/api/queryKeys";
import { useQuery } from "@tanstack/react-query";
import { PREFETCH_STALE_TIME_MS } from "./shared";
import type { ResourceDefinition } from "./shared";

export interface FieldIngestParams {
  token: string;
}

export function fieldIngestQueryKey(token: string) {
  return [...QUERY_ACTIONS.FIELD_INGEST_GET_BY_TOKEN, token ?? ""] as const;
}

/**
 * D12: NO withAuth() here, deliberately — and this module's own reason for existing
 * is just that. This URL is the credential: it is pasted to a cleaner who has no
 * Covera account, so the request must be anonymous. `withAuth()` would attach
 * `Bearer undefined` and the request would fail for a reason nothing in the UI could
 * explain. `queries/inspection.ts` carries the same exception for the same reason.
 *
 * The backend answers `404 FIELD_TOKEN_INVALID` for an unknown token, an expired one
 * and an already-used one alike (D23). This function therefore does not classify the
 * failure — it lets the error through, and the page renders one unusable-link screen
 * for all three.
 */
export async function fetchFieldIngest(token: string, signal?: AbortSignal) {
  const response = await getFieldIngestByToken({
    path: { token },
    ...(signal ? { signal } : {}),
  });

  return response.data;
}

export function useFieldIngest(token: string | undefined) {
  return useQuery({
    queryKey: fieldIngestQueryKey(token ?? ""),
    queryFn: ({ signal }) => {
      if (!token) throw new Error("fieldTokenRequired");
      return fetchFieldIngest(token, signal);
    },
    enabled: Boolean(token),
    staleTime: PREFETCH_STALE_TIME_MS,
  });
}

export const fieldIngestResource: ResourceDefinition = {
  action: QUERY_ACTIONS.FIELD_INGEST_GET_BY_TOKEN,
  key: ({ token }: FieldIngestParams) => fieldIngestQueryKey(token),
  fetch: ({ token }: FieldIngestParams, { signal }) => fetchFieldIngest(token, signal),
};
```

**Accept.**
- `grep -c "withAuth" src/lib/api/queries/fieldIngest.ts` → `0` (invariant 10)
- `grep -c "publicApi" src/lib/api/queries/fieldIngest.ts` → `0` — the generated client's default client already carries no token for an anonymous caller, and `inspection.ts` sets the precedent by passing none at all

### Step 3.5 — Register the three resources

**Touch.** `src/lib/api/queries/registry.ts` (edit)

**Do.** Add the three imports and the three entries. The file becomes:

```ts
import type { ParsedParams } from "@/lib/router/params";
import { apartmentResource, apartmentsHostResource } from "./apartment";
import { assetsResource } from "./assets";
import { shotsResource } from "./shots";
import { reservationsResource } from "./reservations";
import { reservationResource } from "./reservation";
import { reservationPhotosResource } from "./reservationPhotos";
import { inspectionResource } from "./inspection";
import { incidentResource } from "./incident";
import { incidentsResource } from "./incidents";
import { fieldIngestResource } from "./fieldIngest";
import {
  apartmentsAllResource,
  locationsResource,
  usersResource,
} from "./admin";
import type { PrefetchSpec, ResourceDefinition } from "./shared";

/**
 * Every prefetchable resource, by name. A route declares `data: [{ resource: "assets", … }]`
 * and the builder resolves each name here — which is what lets the coverage check
 * see a route's data dependencies as DATA rather than as an opaque callback (D2).
 */
export const QUERY_RESOURCES = {
  apartment: apartmentResource,
  apartmentsHost: apartmentsHostResource,
  apartmentsAll: apartmentsAllResource,
  assets: assetsResource,
  shots: shotsResource,
  reservations: reservationsResource,
  reservation: reservationResource,
  reservationPhotos: reservationPhotosResource,
  inspection: inspectionResource,
  incident: incidentResource,
  incidents: incidentsResource,
  fieldIngest: fieldIngestResource,
  users: usersResource,
  locations: locationsResource,
} as const satisfies Record<string, ResourceDefinition>;

export type ResourceName = keyof typeof QUERY_RESOURCES;

/**
 * `params` returns the values the resource needs, or `null` when THIS navigation
 * does not need it — the tab-gated case (`?tab=apartments`), where prefetching
 * would waste a request for a tab the user may never open (D9, D13).
 */
export interface ResourceRequest {
  resource: ResourceName;
  params: (
    params: ParsedParams,
    searchParams: URLSearchParams,
  ) => Record<string, unknown> | null;
}

/**
 * Resolves a declaration to a concrete prefetch, or `null` when the declaration
 * declines this navigation. Returning `null` is normal, not an error.
 */
export function resolveResource(
  request: ResourceRequest,
  params: ParsedParams,
  searchParams: URLSearchParams,
): PrefetchSpec | null {
  const values = request.params(params, searchParams);
  if (values === null) return null;

  const definition = QUERY_RESOURCES[request.resource];

  return {
    queryKey: definition.key(values),
    queryFn: ({ signal }) => definition.fetch(values, { signal }),
  };
}
```

const APARTMENT_ID = "22222222-2222-2222-2222-222222222222";

const { getIncidents, getFieldIngestByToken, withAuth } = vi.hoisted(() => ({

  getFieldIngestByToken.mockClear();

    expect(incidentsQueryKey({ apartmentId: APARTMENT_ID })).toEqual([
      ...QUERY_ACTIONS.INCIDENTS_GET_BY_APARTMENT,
      APARTMENT_ID,
      INCIDENTS_DEFAULT_PAGE,
      INCIDENTS_DEFAULT_LIMIT,
