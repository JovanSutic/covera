# Phase 7 — Evidence and the cleaner's link

**Prerequisite:** phase 6 (an item can only carry evidence once the page can create one)
**Gate:** `npm run gate`

Two components and three more writes. The host can now attach photos to an item, hand a cleaner a
link that needs no account, and reissue that link when it lapses.

Four things to know before starting:

1. **`cleanerToken` exists on `IncidentItemWithToken` only.** `GET /incidents/{id}` returns
   `IncidentItem`, which has no token field — verified in `types.gen.ts`. So the link cannot be
   read back from the API and is kept as **session state** on the page, seeded from the create and
   reissue responses (D50). With no token in hand the card shows the link's *state* and no copy
   control. This is deliberate and it is the phase's sharpest constraint.
2. **`IncidentItemList` gains an optional `renderItemExtra` slot (D51)** so the page does not loop
   the items a second time. Phase 6's tests must pass **unmodified** — the prop is optional.
3. **The token is minted by `postIncidentsByIdItems` and `postIncidentsByIdItemsByItemIdReissueToken`,
   and by nothing else.** There is no "get my link" endpoint. A plan step that looks for one has
   misread the contract.
4. **Upload is a three-call sequence**, the same shape `submitPhotoProofs.ts` already proves:
   mint presigned URLs → `PUT` the bytes to the presigned URL → attach the returned `key`. The
   presigned `PUT` carries **no** Authorization header (it is signed by the URL itself).

### Step 7.1 — Append the evidence and link writes

```ts
export async function attachIncidentEvidence(
  incidentId: string,
  itemId: string,
  evidence: AttachEvidenceItem[],
  signal?: AbortSignal,
) {
  const config = await withAuth(signal ? { signal } : {});
  const response = await postIncidentsByIdItemsByItemIdEvidence({
    ...config,
    path: { id: incidentId, itemId },
    body: { evidence },
  });

  return response.data;
}

export async function removeIncidentEvidence(
  incidentId: string,
  itemId: string,
  evidenceId: string,
  signal?: AbortSignal,
) {
  const config = await withAuth(signal ? { signal } : {});
  const response = await deleteIncidentsByIdItemsByItemIdEvidenceByEvidenceId({
    ...config,
    path: { id: incidentId, itemId, evidenceId },
  });

  return response.data;
}

/**
 * D50: the ONLY two responses that carry a `cleanerToken` are this one and the item
 * create. `GET /incidents/{id}` answers with `IncidentItem`, which has no token field,
 * so a reissued token has to be handed up to the page the moment it arrives — there is
 * no way to read it back later.
 */
export async function reissueItemToken(
  incidentId: string,
  itemId: string,
  signal?: AbortSignal,
) {
  const config = await withAuth(signal ? { signal } : {});
  const response = await postIncidentsByIdItemsByItemIdReissueToken({
    ...config,
    path: { id: incidentId, itemId },
  });

  return response.data;
}

/**
 * Mint presigned URLs, PUT the bytes, attach the keys. The three steps are
 * `submitPhotoProofs.ts`'s, in the same order and for the same reason.
 *
 * The `PUT` deliberately does NOT carry `withAuth()`'s header: the URL is presigned, so
 * its query string is the credential, and adding an Authorization header to a
 * third-party origin is both useless and a small leak of the session token.
 */
export async function uploadIncidentEvidence(
  incidentId: string,
  itemId: string,
  uploads: readonly { evidenceType: EvidenceType; file: File }[],
  signal?: AbortSignal,
) {
  if (uploads.length === 0) return null;

  const config = await withAuth(signal ? { signal } : {});

  const tokensResponse = await postIncidentsByIdItemsByItemIdUploadTokens({
    ...config,
    path: { id: incidentId, itemId },
    body: {
      files: uploads.map((upload) => ({
        fileType: upload.file.type || "image/jpeg",
      })),
    },
  });

  const targets = tokensResponse.data?.tokens ?? [];

  const evidence = await Promise.all(
    uploads.map(async (upload, index): Promise<AttachEvidenceItem> => {
      const target = targets[index];

      if (!target?.uploadUrl) {
        throw new Error("Missing presigned URL for evidence upload.");
      }

      const putResponse = await fetch(target.uploadUrl, {
        method: "PUT",
        headers: { "Content-Type": upload.file.type || "image/jpeg" },
        body: upload.file,
      });

      if (!putResponse.ok) {
        throw new Error("Failed to upload the evidence photo.");
      }

      return {
        evidenceType: upload.evidenceType,
        storageKey: target.key,
      };
    }),
  );

  return attachIncidentEvidence(incidentId, itemId, evidence, signal);
}
```

```tsx
/* eslint-disable @typescript-eslint/no-explicit-any */
import { useRef, useState } from "react";
import type { ChangeEvent, ReactNode } from "react";
import { useTranslation } from "react-i18next";
import { useMutation, useQueryClient } from "@tanstack/react-query";
import { Camera, Loader2, Trash2 } from "lucide-react";
import { toast } from "sonner";

import {
  removeIncidentEvidence,
  uploadIncidentEvidence,
} from "@/lib/api/incidentClaims";
import { incidentQueryKey } from "@/lib/api/queries/incident";
import {
  activeEvidence,
  activePhotoCount,
  evidenceLabelKey,
  isPhotoEvidence,
  requiredPhotoTypes,
} from "@/lib/helpers/incidentEvidence";
import type { EvidenceType } from "@/lib/helpers/incidentEvidence";
import type { IncidentItem } from "@/api/generated/requests/types.gen";

const R2_IMAGE_URL = import.meta.env.VITE_R2_IMAGE_URL;

interface IncidentItemEvidenceProps {
  incidentId: string;
  item: IncidentItem;
  /**
   * D19's seam, copied from `ShotStepPanel`: jsdom implements no `URL.createObjectURL`,
   * so the tests inject a fake control and drive the real handler instead of stubbing
   * the whole browser. `ShotStepPanel` imports its types the same explicit way.
   */
  renderCapture?: (
    evidenceType: EvidenceType,
    onCapture: (file: File) => void,
  ) => ReactNode;
}

/**
 * The evidence for ONE item: what it still needs, what it has, and the control to add
 * more.
 *
 * It reads phase 1's vocabulary rather than deriving its own rule (invariant 9):
 * `requiredPhotoTypes` says which views this item's type demands, `activePhotoCount`
 * says how many are already there, and `activeEvidence` filters out the soft-deleted
 * rows so a host who deleted a bad photo does not see it counted.
 *
 * Documents are NOT offered here. The four document types are real (`DOCUMENT_EVIDENCE_TYPES`)
 * and the backend accepts them, but the blueprint's host-side flow asks the host to
 * photograph the asset; attaching receipts and invoices is out of scope for this
 * feature and recorded as such.
 */
export function IncidentItemEvidence({
  incidentId,
  item,
  renderCapture,
}: IncidentItemEvidenceProps) {
  const { t } = useTranslation("host");
  const queryClient = useQueryClient();
  const fileInputRef = useRef<HTMLInputElement>(null);

  // Which required view the native picker is currently collecting. A ref would lose
  // the render the label change needs, so it is state.
  const [pendingType, setPendingType] = useState<EvidenceType | null>(null);

  const invalidate = () => {
    queryClient.invalidateQueries({
      queryKey: incidentQueryKey(incidentId),
    });
  };

  const uploadMutation = useMutation({
    mutationFn: (input: { evidenceType: EvidenceType; file: File }) =>
      uploadIncidentEvidence(incidentId, item.id, [input]),
    onSuccess: () => {
      invalidate();
      toast.success(t("incidentPage.evidence.toasts.uploaded"));
    },
    onError: (error: any) => {
      toast.error(error?.error?.message || t("incidentPage.evidence.toasts.uploadError"));
    },
  });

  const deleteMutation = useMutation({
    mutationFn: (evidenceId: string) =>
      removeIncidentEvidence(incidentId, item.id, evidenceId),
    onSuccess: () => {
      invalidate();
      toast.success(t("incidentPage.evidence.toasts.removed"));
    },
    onError: (error: any) => {
      toast.error(error?.error?.message || t("incidentPage.evidence.toasts.removeError"));
    },
  });

  const isBusy = uploadMutation.isPending || deleteMutation.isPending;
  const required = requiredPhotoTypes(item.type);
  // `isPhotoEvidence` rather than `evidence.kind === "photo"`: the kind/type
  // correspondence is phase 1's rule, and this is the second surface that would
  // otherwise restate it (invariant 9's principle, applied to evidence kind).
  const photos = activeEvidence(item).filter((evidence) =>
    isPhotoEvidence(evidence.evidenceType),
  );

  const startCapture = (evidenceType: EvidenceType) => {
    setPendingType(evidenceType);
    fileInputRef.current?.click();
  };

  const handleFileChange = (event: ChangeEvent<HTMLInputElement>) => {
    const file = event.target.files?.[0];
    const evidenceType = pendingType;

    // Reset the input so choosing the SAME file twice fires `change` twice.
    event.target.value = "";
    setPendingType(null);

    if (!file || !evidenceType) return;

    uploadMutation.mutate({ evidenceType, file });
  };

  return (
    <div className="mt-3 border-t border-gray-100 pt-3 dark:border-gray-800">
      {renderCapture ? (
        <fieldset disabled={isBusy} className="contents">
          {required.map((type) =>
            renderCapture(type, (file) =>
              uploadMutation.mutate({ evidenceType: type, file }),
            ),
          )}
        </fieldset>
      ) : (
        <div className="flex flex-wrap gap-2">
          {required.map((type) => (
            <button
              key={type}
              type="button"
              disabled={isBusy}
              onClick={() => startCapture(type)}
              className="inline-flex items-center gap-1.5 rounded-lg border border-gray-300 px-2.5 py-1.5 text-xs font-medium text-gray-700 transition-colors hover:bg-gray-50 disabled:opacity-40 dark:border-gray-700 dark:text-gray-300 dark:hover:bg-gray-800"
            >
              {uploadMutation.isPending && pendingType === type ? (
                <Loader2 className="h-3.5 w-3.5 animate-spin" />
              ) : (
                <Camera className="h-3.5 w-3.5" />
              )}
              {activePhotoCount(item, type) > 0
                ? t("incidentPage.evidence.retakeFor", {
                    view: t(evidenceLabelKey(type)),
                  })
                : t("incidentPage.evidence.captureFor", {
                    view: t(evidenceLabelKey(type)),
                  })}
            </button>
          ))}

          <input
            ref={fileInputRef}
            type="file"
            accept="image/*"
            capture="environment"
            onChange={handleFileChange}
            className="hidden"
          />
        </div>
      )}

      {photos.length > 0 && (
        <ul className="mt-3 flex flex-wrap gap-2">
          {photos.map((photo) => (
            <li key={photo.id} className="relative">
              <img
                src={`${R2_IMAGE_URL}/${photo.storageKey}`}
                alt={t(evidenceLabelKey(photo.evidenceType))}
                className="h-16 w-16 rounded-lg border border-gray-200 object-cover dark:border-gray-700"
              />

              <button
                type="button"
                disabled={isBusy}
                onClick={() => deleteMutation.mutate(photo.id)}
                aria-label={t("incidentPage.evidence.remove")}
                className="absolute -right-1.5 -top-1.5 inline-flex h-5 w-5 items-center justify-center rounded-full border border-gray-200 bg-white text-gray-500 shadow-sm transition-colors hover:text-rose-600 disabled:opacity-40 dark:border-gray-700 dark:bg-gray-900 dark:text-gray-400"
              >
                <Trash2 className="h-3 w-3" />
              </button>
            </li>
          ))}
        </ul>
      )}
    </div>
  );
}
```

### Step 7.4 — Create the field link card

**Touch.** `src/components/host/incident/IncidentFieldLink.tsx` (create)

**Do.** D17/D50. The link renders only when a token is in hand; the reissue control renders only when
`canReissueItemLink` says the token lapsed while the item was still collectable.

```tsx
/* eslint-disable @typescript-eslint/no-explicit-any */
import { useEffect, useState } from "react";
import { useTranslation } from "react-i18next";
import { useMutation, useQueryClient } from "@tanstack/react-query";
import { Check, Copy, Link2 } from "lucide-react";
import { toast } from "sonner";

import { ActionButton } from "@/components/shared/ActionButton";
import { reissueItemToken } from "@/lib/api/incidentClaims";
import { incidentQueryKey } from "@/lib/api/queries/incident";
import { canReissueItemLink, isItemLinkLive } from "@/lib/helpers/claimStatus";
import { appPaths, buildPath } from "@/lib/router";
import type { IncidentItem } from "@/api/generated/requests/types.gen";

interface IncidentFieldLinkProps {
  incidentId: string;
  /**
   * `IncidentItem`, NOT `IncidentItemWithToken`. The token is a separate prop because
   * item objects read back from the API do not have one (D50) — typing this as the
   * `…WithToken` variant would invite the page to cast its way past exactly the fact
   * this component exists to handle.
   */
  item: IncidentItem;
  /** D50: `null` for every item read back from the API — the detail response has no token. */
  token: string | null;
  /** The page's clock, read once per page (D46). */
  now: number;
  onTokenIssued: (token: string) => void;
}

/**
 * The cleaner's link for ONE item.
 *
 * D50 is the whole shape of this component: `cleanerToken` is returned by the item
 * create and by reissue, and by nothing else. `GET /incidents/{id}` answers with
 * `IncidentItem`, which has no token field at all. So `token === null` is the NORMAL
 * state on a page load, not an error — and this component's job in that state is to
 * say what the link's status is and offer the reissue path, not to render a copy
 * button for a value it does not have.
 *
 * D17: reissue is offered only for an item whose link lapsed while still collectable
 * (`canReissueItemLink`). Every other case — already collected, never generated — gets
 * copy that describes the state instead of a control the API would refuse.
 */
export function IncidentFieldLink({
  incidentId,
  item,
  token,
  now,
  onTokenIssued,
}: IncidentFieldLinkProps) {
  const { t } = useTranslation("host");
  const queryClient = useQueryClient();
  const [copied, setCopied] = useState(false);

  const link = token ? buildPath(appPaths.fieldIncident, { token }) : null;
  const isLive = token !== null && isItemLinkLive(item, now);
  const canReissue = canReissueItemLink(item, now);

  const reissueMutation = useMutation({
    mutationFn: () => reissueItemToken(incidentId, item.id),
    onSuccess: (reissued) => {
      if (reissued?.cleanerToken) onTokenIssued(reissued.cleanerToken);

      queryClient.invalidateQueries({ queryKey: incidentQueryKey(incidentId) });
      toast.success(t("incidentPage.fieldLink.toasts.reissued"));
    },
    onError: (error: any) => {
      toast.error(
        error?.error?.message || t("incidentPage.fieldLink.toasts.reissueError"),
      );
    },
  });

  // The tick is a courtesy: it clears the "Copied" label without the host having to
  // look at it. Not a timer in the data path — nothing here refetches.
  useEffect(() => {
    if (!copied) return;

    const timer = window.setTimeout(() => setCopied(false), 2_000);

    return () => window.clearTimeout(timer);
  }, [copied]);

  const copyLink = async () => {
    if (!link) return;

    try {
      await navigator.clipboard.writeText(link);
      setCopied(true);
      toast.success(t("incidentPage.fieldLink.toasts.copied"));
    } catch {
      toast.error(t("incidentPage.fieldLink.toasts.copyError"));
    }
  };

  return (
    <div className="mt-3 flex flex-wrap items-center gap-2 border-t border-gray-100 pt-3 dark:border-gray-800">
      <span className="inline-flex items-center gap-1.5 text-xs text-gray-500 dark:text-gray-400">
        <Link2 className="h-3.5 w-3.5 shrink-0" />
        {isLive
          ? t("incidentPage.fieldLink.live")
          : token
            ? t("incidentPage.fieldLink.usedOrExpired")
            : t("incidentPage.fieldLink.notInThisSession")}
      </span>

      {link && (
        <>
          <input
            readOnly
            value={link}
            aria-label={t("incidentPage.fieldLink.label")}
            onFocus={(event) => event.currentTarget.select()}
            className="min-w-0 flex-1 rounded-lg border border-gray-200 bg-gray-50 px-2 py-1 text-xs text-gray-600 dark:border-gray-800 dark:bg-gray-950 dark:text-gray-400"
          />

          <ActionButton
            variant={copied ? "neutral" : "info"}
            size="sm"
            icon={
              copied ? (
                <Check className="h-3.5 w-3.5 shrink-0" />
              ) : (
                <Copy className="h-3.5 w-3.5 shrink-0" />
              )
            }
            label={
              copied
                ? t("incidentPage.fieldLink.copied")
                : t("incidentPage.fieldLink.copy")
            }
            onClick={copyLink}
          />
        </>
      )}

      {canReissue && (
        <ActionButton
          variant="neutral"
          size="sm"
          label={t("incidentPage.fieldLink.reissue")}
          loading={reissueMutation.isPending}
          onClick={() => reissueMutation.mutate()}
        />
      )}
    </div>
  );
}
```
