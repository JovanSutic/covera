## Step 4.1 — `formatStayRange`

**Touch:** `src/lib/helpers/claimMessage.ts`, `src/lib/helpers/claimMessage.test.ts`

**Do:** add to `claimMessage.ts`, above `buildClaimMessage`:

```ts
/**
 * D15: the stay as one string. `locale` is a parameter rather than a module read so the
 * helper is testable and the message follows the UI language the host is reading — the
 * dates in the pasted text are the host's dates, not Airbnb's.
 *
 * An unparseable instant returns `""` rather than "Invalid Date", the same rule
 * `toEpoch` applies in `claimDeadline.ts`.
 */
export function formatStayRange(
  checkIn: string,
  checkOut: string,
  locale: string,
): string {
  const formatter = new Intl.DateTimeFormat(locale, {
    day: "numeric",
    month: "short",
    year: "numeric",
  });

  const start = new Date(checkIn);
  const end = new Date(checkOut);

  if (!Number.isFinite(start.getTime()) || !Number.isFinite(end.getTime())) {
    return "";
  }

  return `${formatter.format(start)} – ${formatter.format(end)}`;
}
```

Add to `claimMessage.test.ts`:

```ts
describe("formatStayRange", () => {
  it("renders both dates with the locale's own formatting", () => {
    const range = formatStayRange(
      "2026-11-01T15:00:00.000Z",
      "2026-11-08T10:00:00.000Z",
      "en",
    );

    expect(range).toContain("Nov 1, 2026");
    expect(range).toContain("Nov 8, 2026");
    expect(range).toContain("–");
  });

  it("returns an empty string when either instant is unparseable", () => {
    expect(formatStayRange("garbage", "2026-11-08T10:00:00.000Z", "en")).toBe("");
  });
});
```
