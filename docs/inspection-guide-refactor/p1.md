### Step 1.1 — Create the flow model

**Touch.** `src/lib/helpers/guideFlow.ts` (create)

**Do.**

```ts
import type { CapturedApartmentShot } from "@/lib/api/submitPhotoProofs";

/**
 * D1: the ONLY computation of "where is the host in the photo-proof flow". The strip,
 * the step panel, the review screen and the exit guard all ask these functions instead
 * of comparing `capturedImageUrl` themselves — the same reason `proofWindow.ts` exists:
 * the control that offers an action and the surface that refuses it cannot drift if
 * they share one answer.
 *
 * Everything here is pure. `shots` is always a `CapturedApartmentShot[]`; these
 * functions never mutate it and never read a clock.
 */

/** One item of the transparency list (prep §1). */
export interface StripItem {
  index: number;
  shotId: string;
  title: string;
  roomLocation: string;
  /** The local object URL of the capture, or null while the step is pending. */
  imageUrl: string | null;
  isCaptured: boolean;
  /** The step sequential progress stopped at (prep §1). Exactly one item is current. */
  isCurrent: boolean;
}

/**
 * A step is captured when it carries a preview URL. The upload's own test is `rawFile`,
 * and `InlineCamera` sets both in the same call — this is the same pair, read for the
 * screen rather than for the request.
 */
export function isCaptured(shot: CapturedApartmentShot | undefined): boolean {
  return Boolean(shot?.capturedImageUrl);
}

/** How many steps carry a photo. The progress counter's numerator. */
export function capturedCount(shots: CapturedApartmentShot[]): number {
  return shots.filter(isCaptured).length;
}

/**
 * The first pending step strictly after `fromIndex`, or null when none remains.
 * Strictly after, not at: `fromIndex` is the step just captured, and a captured step
 * is never "next".
 */
export function nextPendingIndex(
  shots: CapturedApartmentShot[],
  fromIndex: number,
): number | null {
  for (let index = fromIndex + 1; index < shots.length; index += 1) {
    if (!isCaptured(shots[index])) return index;
  }

  return null;
}

/**
 * D20: where a guide opens. A resumed guide (the page passes `initialShots`, which can
 * carry captures) lands on the first pending step; a fresh one lands on 0, which is also
 * its first pending step. An all-captured guide lands on 0 rather than nowhere — the
 * review screen ignores the cursor, and a null cursor would be a second empty state.
 */
export function initialCursorIndex(shots: CapturedApartmentShot[]): number {
  return nextPendingIndex(shots, -1) ?? 0;
}

/**
 * D2: the cursor moves on an explicit Next and on nothing else. It never moves backwards
 * and never wraps: when no pending step follows, it stays where it is. `isFlowComplete`
 * is the signal that the flow is done, not a moved cursor.
 */
export function advanceCursor(
  shots: CapturedApartmentShot[],
  cursorIndex: number,
): number {
  return nextPendingIndex(shots, cursorIndex) ?? cursorIndex;
}

/**
 * D21: tapping a strip item shows that step and moves nothing else. A pending step is not
 * selectable (prep §1: "Only the photo components with the photo ... are clickable"), so
 * the guard lives here rather than in the item's `onClick` — the item renders no button
 * for a pending step, and this is the second half of that same rule, at the state layer
 * where a stray call cannot bypass it.
 */
export function selectStep(
  shots: CapturedApartmentShot[],
  index: number,
  cursorIndex: number,
): number {
  return isCaptured(shots[index]) ? index : cursorIndex;
}

/**
 * True while the host is looking at a captured step that is not the cursor (prep §2).
 * This, not a flag, is what puts "Go to current step" on screen and takes "Next" off it.
 */
export function isViewingEarlierStep(
  cursorIndex: number,
  viewedIndex: number,
): boolean {
  return viewedIndex !== cursorIndex;
}

/**
 * D16: the flow is complete when every step carries a photo. An empty guide is NOT
 * complete — completeness renders the review screen, and `shots.length === 0` renders the
 * existing `noShots` panel instead. Conflating the two would put a Submit button on a
 * guide that has nothing to submit, which `submitInspectionPhotos` would then reject.
 */
export function isFlowComplete(shots: CapturedApartmentShot[]): boolean {
  return shots.length > 0 && shots.every(isCaptured);
}

/**
 * The strip's data. Invariant 10: the strip computes nothing — every blank/pending,
 * captured, and current decision is made here. Every item is returned, in the API's
 * order, blank ones included: seeing how many photos are needed is the point (prep §1).
 */
export function stripItems(
  shots: CapturedApartmentShot[],
  cursorIndex: number,
): StripItem[] {
  return shots.map((shot, index) => ({
    index,
    shotId: shot.id,
    title: shot.title,
    roomLocation: shot.roomLocation,
    imageUrl: shot.capturedImageUrl ?? null,
    isCaptured: isCaptured(shot),
    isCurrent: index === cursorIndex,
  }));
}
```

### Step 1.2 — Test the flow model

**Touch.** `src/lib/helpers/guideFlow.test.ts` (create)

**Do.**

```ts
import { describe, expect, it } from "vitest";

import type { CapturedApartmentShot } from "@/lib/api/submitPhotoProofs";
import {
  advanceCursor,
  capturedCount,
  initialCursorIndex,
  isCaptured,
  isFlowComplete,
  isViewingEarlierStep,
  nextPendingIndex,
  selectStep,
  stripItems,
} from "./guideFlow";

/** Three steps, none captured — the state the guide opens in. */
function threePending(): CapturedApartmentShot[] {
  return [0, 1, 2].map((index) => ({
    id: `shot-${index}`,
    apartmentId: "apartment-1",
    roomLocation: "KITCHEN",
    shotType: "SWEEP_ONLY",
    title: `Shot ${index}`,
    instructions: `Instructions ${index}`,
    assetIds: [],
    createdAt: "2026-09-01T14:00:00.000Z",
    updatedAt: "2026-09-01T14:00:00.000Z",
  }));
}

/** The same three, with the given indices captured. */
function withCaptured(indices: number[]): CapturedApartmentShot[] {
  return threePending().map((shot, index) =>
    indices.includes(index)
      ? { ...shot, capturedImageUrl: `blob:preview-${index}` }
      : shot,
  );
}

describe("isCaptured", () => {
  it("is false for a pending step, an undefined shot, and an empty URL", () => {
    const [pending] = threePending();

    expect(isCaptured(pending)).toBe(false);
    expect(isCaptured(undefined)).toBe(false);
    expect(isCaptured({ ...pending, capturedImageUrl: "" })).toBe(false);
  });

  it("is true once a preview URL is set", () => {
    expect(isCaptured(withCaptured([0])[0])).toBe(true);
  });
});

describe("capturedCount", () => {
  it("counts the captured steps and nothing else", () => {
    expect(capturedCount(threePending())).toBe(0);
    expect(capturedCount(withCaptured([0, 2]))).toBe(2);
    expect(capturedCount([])).toBe(0);
  });
});

describe("nextPendingIndex", () => {
  it("finds the first pending step strictly after the cursor", () => {
    expect(nextPendingIndex(threePending(), -1)).toBe(0);
    expect(nextPendingIndex(threePending(), 0)).toBe(1);
  });

  it("skips the steps already captured", () => {
    // 1 is captured, so from 0 the next pending step is 2, not 1.
    expect(nextPendingIndex(withCaptured([1]), 0)).toBe(2);
  });

  it("returns null when nothing pending remains", () => {
    expect(nextPendingIndex(withCaptured([0, 1, 2]), 0)).toBeNull();
    expect(nextPendingIndex([], -1)).toBeNull();
  });

  it("never looks backwards", () => {
    // 0 is pending but sits behind the cursor: from 1 there is no next pending step.
    expect(nextPendingIndex(withCaptured([2]), 1)).toBeNull();
  });
});

describe("initialCursorIndex", () => {
  it("opens a fresh guide on the first step", () => {
    expect(initialCursorIndex(threePending())).toBe(0);
  });

  it("opens a resumed guide on the first PENDING step", () => {
    // D20: steps 0 and 1 are done, so the host lands where the work is.
    expect(initialCursorIndex(withCaptured([0, 1]))).toBe(2);
  });

  it("opens an all-captured guide on 0, so the cursor is never null", () => {
    expect(initialCursorIndex(withCaptured([0, 1, 2]))).toBe(0);
  });

  it("opens an empty guide on 0", () => {
    expect(initialCursorIndex([])).toBe(0);
  });
});

describe("advanceCursor", () => {
  it("moves to the next pending step after a capture", () => {
    // Step 0 was just captured; the cursor was on it.
    expect(advanceCursor(withCaptured([0]), 0)).toBe(1);
  });

  it("skips over steps captured out of order", () => {
    expect(advanceCursor(withCaptured([0, 1]), 0)).toBe(2);
  });

  it("stays put when nothing pending follows — it does not wrap to a captured step", () => {
    expect(advanceCursor(withCaptured([0, 1, 2]), 2)).toBe(2);
    expect(advanceCursor(withCaptured([0]), 0)).toBe(0);
  });

  it("is a no-op on an empty guide", () => {
    expect(advanceCursor([], 0)).toBe(0);
  });
});

describe("selectStep", () => {
  it("returns the tapped index for a captured step", () => {
    expect(selectStep(withCaptured([0, 1]), 0, 1)).toBe(0);
  });

  it("returns the cursor for a pending step — a pending item is not selectable", () => {
    expect(selectStep(withCaptured([0]), 1, 1)).toBe(1);
  });

  it("returns the cursor for an index with no step at all", () => {
    expect(selectStep(threePending(), 99, 0)).toBe(0);
  });
});

describe("isViewingEarlierStep", () => {
  it("is false at the cursor, true anywhere else", () => {
    expect(isViewingEarlierStep(2, 2)).toBe(false);
    expect(isViewingEarlierStep(2, 0)).toBe(true);
  });
});

describe("isFlowComplete", () => {
  it("is true only when every step carries a photo", () => {
    expect(isFlowComplete(withCaptured([0, 1, 2]))).toBe(true);
    expect(isFlowComplete(withCaptured([0, 1]))).toBe(false);
    expect(isFlowComplete(threePending())).toBe(false);
  });

  it("is false for an empty guide — there is nothing to submit", () => {
    expect(isFlowComplete([])).toBe(false);
  });
});

describe("stripItems", () => {
  it("returns every step, blank ones included, in order", () => {
    const items = stripItems(withCaptured([1]), 1);

    expect(items.map((item) => item.index)).toEqual([0, 1, 2]);
    expect(items.map((item) => item.isCaptured)).toEqual([false, true, false]);
  });

  it("marks exactly one item current, and it is the cursor", () => {
    const items = stripItems(withCaptured([0]), 1);

    expect(items.filter((item) => item.isCurrent).map((item) => item.index)).toEqual([1]);
  });

  it("carries the preview URL only for a captured step", () => {
    const items = stripItems(withCaptured([2]), 0);

    expect(items[2].imageUrl).toBe("blob:preview-2");
    expect(items[0].imageUrl).toBeNull();
  });

  it("carries the shot's title and room so the item needs no lookup", () => {
    const items = stripItems(threePending(), 0);

    expect(items[1].title).toBe("Shot 1");
    expect(items[1].roomLocation).toBe("KITCHEN");
    expect(items[1].shotId).toBe("shot-1");
  });

  it("returns nothing for an empty guide", () => {
    expect(stripItems([], 0)).toEqual([]);
  });
});
```

## Phase 1 invariants

- [ ] No hand-edits under `src/api/generated/`
- [ ] Every new i18n key exists in all four locales (this phase is what makes that true)
- [ ] Query keys come from `QUERY_ACTIONS` — this phase adds no key and moves no call site
- [ ] No mutation added, so nothing new to invalidate
- [ ] No secret added to a `VITE_` env var
- [ ] **Invariant 9:** every flow decision is a function in `guideFlow.ts` — no component yet reads
      `capturedImageUrl` directly (nothing consumes the module at this phase, so this holds trivially;
      it is re-checked at 2, 3 and 4)
- [ ] `git diff --stat` names only `guideFlow.ts`, `guideFlow.test.ts`, and the four locale files
