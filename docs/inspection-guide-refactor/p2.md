### Step 2.1 — Create the strip item

**Touch.** `src/components/host/guide/PhotoStripItem.tsx` (create)

**Do.**

```tsx
import { useTranslation } from "react-i18next";

import type { StripItem } from "@/lib/helpers/guideFlow";

interface PhotoStripItemProps {
  item: StripItem;
  total: number;
  onSelect: (index: number) => void;
}

/**
 * One photo in the transparency list (prep §1). It renders the three states the helper
 * already decided, and makes no decision of its own (D25/invariant 10).
 *
 * The captured state is a <button>; the pending state is a <div>. That is the whole of
 * "only the photo components with the photo are clickable" — a pending item cannot be
 * tabbed to, activated, or reached by a screen reader as a control, because it is not
 * one. Invariant 12 pins this by grep.
 */
export function PhotoStripItem({ item, total, onSelect }: PhotoStripItemProps) {
  const { t } = useTranslation("assets");

  // The current step gets the ring; a captured item the host is looking at gets the
  // border. They can be the same item, which is why they are two classes and not three
  // states (prep §1 asks for the highlight on the current step, not on the viewed one).
  const currentRing = item.isCurrent
    ? "ring-2 ring-emerald-500 ring-offset-1 dark:ring-offset-gray-900"
    : "";

  const position = { index: item.index + 1, total };

  if (!item.isCaptured) {
    return (
      <div
        data-testid="guide-strip-item"
        aria-label={t("shotGuide.strip.pendingItem", position)}
        aria-disabled="true"
        className={`flex h-16 w-16 shrink-0 items-center justify-center rounded-lg border border-dashed border-gray-300 bg-gray-50 text-gray-400 dark:border-gray-700 dark:bg-gray-900 ${currentRing}`}
      >
        <span aria-hidden="true" className="text-lg">
          📷
        </span>
      </div>
    );
  }

  return (
    <button
      type="button"
      data-testid="guide-strip-item"
      onClick={() => onSelect(item.index)}
      aria-current={item.isCurrent ? "step" : undefined}
      aria-label={t("shotGuide.strip.capturedItem", position)}
      className={`relative h-16 w-16 shrink-0 cursor-pointer overflow-hidden rounded-lg border border-gray-300 bg-gray-950 transition-transform hover:scale-105 dark:border-gray-700 ${currentRing}`}
    >
      <img
        src={item.imageUrl ?? ""}
        alt={item.title}
        className="h-full w-full object-cover"
      />

      {item.isCurrent && (
        <span className="absolute inset-x-0 bottom-0 truncate bg-emerald-600/90 px-1 text-[9px] font-semibold leading-4 text-white">
          {t("shotGuide.strip.currentItem")}
        </span>
      )}
    </button>
  );
}
```

button, inside the captured branch). `npx tsc -b` → exit 0.

### Step 2.2 — Create the strip

**Touch.** `src/components/host/guide/PhotoStrip.tsx` (create)

**Do.**

```tsx
import { useTranslation } from "react-i18next";

import type { StripItem } from "@/lib/helpers/guideFlow";
import { PhotoStripItem } from "./PhotoStripItem";

interface PhotoStripProps {
  items: StripItem[];
  onSelect: (index: number) => void;
}

/**
 * The Photos transparency list (prep §1): every step the apartment requires, as a
 * horizontally scrollable row of small photo components, blank until taken, with the
 * current step highlighted. Visible in every flow state except while the camera is open
 * (D14) — the flow decides that, not this component.
 *
 * It holds no state and computes no reachability (invariant 10): `items` arrives from
 * `stripItems()` and `onSelect` is the flow's `selectStep`.
 */
export function PhotoStrip({ items, onSelect }: PhotoStripProps) {
  const { t } = useTranslation("assets");

  if (items.length === 0) return null;

  return (
    <div
      data-testid="guide-photo-strip"
      // D13: pinned under the `h-12` header (Header.tsx:68) while the page scrolls, at a
      // z-index below the header's z-40. The -mx-4/px-4 pair lets items scroll under the
      // page's own px-4 edge while the bar's background stays opaque across the full width.
      className="sticky top-12 z-30 -mx-4 mb-3 border-b border-gray-100 bg-white/95 px-4 py-2 backdrop-blur-md dark:border-gray-800 dark:bg-gray-900/95"
      role="list"
      aria-label={t("shotGuide.strip.ariaLabel")}
    >
      <div className="flex items-center gap-2 overflow-x-auto pb-1">
        {items.map((item) => (
          <div key={item.shotId} role="listitem">
            <PhotoStripItem item={item} total={items.length} onSelect={onSelect} />
          </div>
        ))}
      </div>
    </div>
  );
}
```

**Accept.** `npx tsc -b` → exit 0. `grep -c "useState" src/components/host/guide/PhotoStrip.tsx` → `0`
(invariant 10: the strip holds no state).

**Verify.** `npm run gate`
