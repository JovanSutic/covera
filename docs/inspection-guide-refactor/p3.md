### Step 3.1 — Create the captured-shots hook

**Touch.** `src/components/host/guide/useCapturedShots.ts` (create)

**Do.**

```ts
import { useCallback, useState } from "react";

import type { CapturedApartmentShot } from "@/lib/api/submitPhotoProofs";

/**
 * The captured photos, and the three things the flow does to them.
 *
 * D8/phase 3: this is the state the old `ApartmentShotGuide` kept inline as `shots` plus
 * `setShots`, lifted so the panel, the actions and the orchestrator do not each hold a
 * copy of the same accumulator. It is a hook rather than four `useState` calls in the
 * flow because the update rules (replace one step, never append; seed from the server
 * only once) are the part worth naming.
 *
 * `capture` replaces the step at `index`, it never appends or reorders: the API's shots
 * are the steps, and their order is the sequence (prep §1). A retake is therefore the
 * same call as a first capture, which is why there is one function and not two.
 */
export function useCapturedShots(initialShots: CapturedApartmentShot[]) {
  const [shots, setShots] = useState<CapturedApartmentShot[]>(initialShots);

  const capture = useCallback(
    (index: number, previewUrl: string, file?: File): void => {
      setShots((previous) =>
        previous.map((shot, shotIndex) =>
          shotIndex === index
            ? { ...shot, capturedImageUrl: previewUrl, rawFile: file }
            : shot,
        ),
      );
    },
    [],
  );

  return { shots, capture };
}
```

### Step 3.2 — Create the step actions

**Touch.** `src/components/host/guide/ShotStepActions.tsx` (create)

**Do.**

```tsx
import { useTranslation } from "react-i18next";

interface ShotStepActionsProps {
  /** True when this step carries a photo — the step's own state, not the flow's. */
  isCaptured: boolean;
  /** D4: true while viewing a captured step that is not the cursor. */
  isViewingEarlierStep: boolean;
  /** D5: false when no pending step follows — the review screen is the next surface. */
  hasNextStep: boolean;
  disabled: boolean;
  onNext: () => void;
  onGoToCurrent: () => void;
}

/**
 * The step's exits (prep §2). At most two are ever rendered, and which two is entirely
 * decided by the caller — this component compares nothing.
 *
 *   normal          → Next            (disabled until the step carries a photo, D5)
 *   viewing earlier → Go to current   (never Next, D4)
 *   last pending    → neither; the hint says why, and the review screen follows a capture
 */
export function ShotStepActions({
  isCaptured,
  isViewingEarlierStep,
  hasNextStep,
  disabled,
  onNext,
  onGoToCurrent,
}: ShotStepActionsProps) {
  const { t } = useTranslation("assets");

  if (isViewingEarlierStep) {
    return (
      <button
        type="button"
        onClick={onGoToCurrent}
        disabled={disabled}
        className="w-full rounded-xl bg-gray-900 px-4 py-2.5 text-sm font-medium text-white transition-colors hover:bg-gray-800 disabled:cursor-not-allowed disabled:opacity-40 dark:bg-gray-100 dark:text-gray-900 dark:hover:bg-gray-200"
      >
        {t("shotGuide.actions.goToCurrent")}
      </button>
    );
  }

  if (!hasNextStep) {
    return (
      <p className="py-2 text-center text-xs text-gray-500 dark:text-gray-400">
        {isCaptured
          ? t("shotGuide.review.checklist")
          : t("shotGuide.actions.takePhotoFirst")}
      </p>
    );
  }

  return (
    <button
      type="button"
      onClick={onNext}
      disabled={!isCaptured || disabled}
      className="w-full rounded-xl bg-gray-900 px-4 py-2.5 text-sm font-medium text-white transition-colors hover:bg-gray-800 disabled:cursor-not-allowed disabled:opacity-40 dark:bg-gray-100 dark:text-gray-900 dark:hover:bg-gray-200"
    >
      {t("shotGuide.actions.next")}
    </button>
  );
}
```

### Step 3.3 — Create the step panel

**Touch.** `src/components/host/guide/ShotStepPanel.tsx` (create)

**Do.** This is the main panel: the step's heading, instruction and type chip — moved from
`ShotsGuide.tsx` unchanged — and either the camera or the captured/preview area with the single
camera control.

```tsx
import type { ReactNode } from "react";
import { useTranslation } from "react-i18next";

import Typography from "@/components/Typography";
import type { CapturedApartmentShot } from "@/lib/api/submitPhotoProofs";
import { ShotStepActions } from "./ShotStepActions";

interface ShotStepPanelProps {
  shot: CapturedApartmentShot;
  /** Index within the room's own shots — the "3 of 5 in room" counter, as today. */
  shotIndexInRoom: number;
  roomShotCount: number;
  completedCount: number;
  totalCount: number;
  isCameraOpen: boolean;
  isCaptured: boolean;
  isViewingEarlierStep: boolean;
  hasNextStep: boolean;
  disabled: boolean;
  onCapture: (previewUrl: string, file: File) => void;
  onCameraCancel: () => void;
  onOpenCamera: () => void;
  onNext: () => void;
  onGoToCurrent: () => void;
  /**
   * D24b: the camera is injected. Prep §2's sequence puts the native camera outside the
   * app, so the seam between "the app's step" and "the device's camera" is real — and it
   * is what lets the phase-3 test drive the flow without a file input.
   */
  renderCamera: (onCapture: (url: string, file: File) => void, onCancel: () => void) => ReactNode;
}

/**
 * One step of the flow (prep §2). The structure is the same for every step; only the
 * instruction changes — which is exactly why the step is a component and not a branch
 * inside a page.
 */
export function ShotStepPanel({
  shot,
  shotIndexInRoom,
  roomShotCount,
  completedCount,
  totalCount,
  isCameraOpen,
  isCaptured,
  isViewingEarlierStep,
  hasNextStep,
  disabled,
  onCapture,
  onCameraCancel,
  onOpenCamera,
  onNext,
  onGoToCurrent,
  renderCamera,
}: ShotStepPanelProps) {
  const { t } = useTranslation("assets");

  const progressPercent = Math.round((completedCount / totalCount) * 100);

  return (
    <div className="space-y-2">
      {/* Header & progress bar — markup unchanged from ShotsGuide.tsx. */}
      <div className="space-y-2 rounded-xl bg-white p-3 pt-0 sm:p-2 dark:bg-gray-900">
        <div className="flex items-center justify-between gap-3">
          <div className="min-w-0">
            <span className="block truncate text-[10px] font-semibold uppercase tracking-wider text-gray-400 sm:text-xs">
              {t("shotGuide.header.currentLocation")}
            </span>
            <h3 className="truncate text-sm font-semibold text-gray-900 sm:text-base dark:text-gray-100">
              {t(`roomLocations.${shot.roomLocation}`, { defaultValue: shot.roomLocation })}
            </h3>
          </div>
          <div className="shrink-0 text-right">
            <span className="block text-[10px] text-gray-500 sm:text-xs dark:text-gray-400">
              {t("shotGuide.header.shotProgress", {
                current: shotIndexInRoom,
                total: roomShotCount,
              })}
            </span>
            <div className="text-xs font-semibold text-gray-900 sm:text-sm dark:text-gray-100">
              {t("shotGuide.header.completedCount", {
                completed: completedCount,
                total: totalCount,
              })}
            </div>
          </div>
        </div>

        <div className="h-1.5 w-full overflow-hidden rounded-full bg-gray-100 dark:bg-gray-800">
          <div
            className="h-full bg-emerald-500 transition-all duration-300 ease-out"
            style={{ width: `${progressPercent}%` }}
          />
        </div>
      </div>

      {/* The step's own content. */}
      <div className="space-y-4 rounded-2xl bg-white lg:p-5 dark:bg-gray-900">
        <div>
          <div className="flex items-center justify-between gap-2">
            <Typography
              type="h4"
              className="text-base font-semibold text-gray-900 sm:text-lg dark:text-gray-100"
            >
              {shot.title}
            </Typography>
            <span className="shrink-0 rounded-full border border-gray-200 bg-gray-100 px-2.5 py-0.5 text-xs font-medium text-gray-700 dark:border-gray-700 dark:bg-gray-800 dark:text-gray-300">
              {t(`shotGuide.shotTypes.${shot.shotType}`, { defaultValue: shot.shotType })}
            </span>
          </div>
          <p className="mt-1 whitespace-pre-line text-xs text-gray-600 dark:text-gray-300">
            {shot.instructions}
          </p>
        </div>

        {isCameraOpen ? (
          <fieldset disabled={disabled} className="contents">
            {renderCamera(onCapture, onCameraCancel)}
          </fieldset>
        ) : (
          <div className="space-y-3">
            <div className="relative flex aspect-16/12 w-full items-center justify-center overflow-hidden rounded-xl border border-gray-800 bg-gray-950 lg:aspect-16/9">
              {isCaptured ? (
                <img
                  src={shot.capturedImageUrl}
                  alt={shot.title}
                  className="h-full w-full object-cover"
                />
              ) : (
                <div className="space-y-2 p-6 text-center">
                  <div className="inline-flex h-10 w-10 items-center justify-center rounded-full bg-gray-800 text-gray-400">
                    📷
                  </div>
                  <p className="text-xs text-gray-400">{t("shotGuide.camera.noPhoto")}</p>
                </div>
              )}
            </div>

            <button
              type="button"
              onClick={onOpenCamera}
              disabled={disabled}
              className="flex w-full cursor-pointer items-center justify-center gap-2 rounded-xl bg-gray-900 px-4 py-2.5 text-sm font-medium text-white shadow-xs transition-colors hover:bg-gray-800 disabled:cursor-not-allowed disabled:opacity-40 dark:bg-gray-100 dark:text-gray-900 dark:hover:bg-gray-200"
            >
              <span>📷</span>
              <span>
                {isCaptured
                  ? t("shotGuide.camera.retakePhoto")
                  : t("shotGuide.camera.openCamera")}
              </span>
            </button>
          </div>
        )}

        <ShotStepActions
          isCaptured={isCaptured}
          isViewingEarlierStep={isViewingEarlierStep}
          hasNextStep={hasNextStep}
          disabled={disabled}
          onNext={onNext}
          onGoToCurrent={onGoToCurrent}
        />
      </div>
    </div>
  );
}
```
