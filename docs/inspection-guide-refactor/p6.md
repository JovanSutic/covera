# Phase 6 — leaving the guide asks first

**Prerequisite:** phase 5 (`InspectionGuideFlow.tsx` is what this phase adds the guard to).
**Gate:** `npm run gate`

Prep §3's "light security measure": in-app navigation away from a guide with unsubmitted photos asks
for confirmation, and a tab close keeps the existing warning. No history trapping, no disabled chrome
(operator decision E1, and the spec's out-of-scope list).

The pure rule gets a test (D25). The hook and the dialog are wiring and markup and do not.

---

### Step 6.1 — Create the exit rule

**Touch.** `src/lib/helpers/guideExit.ts` (create)

**Do.**

```ts
import type { CapturedApartmentShot } from "@/lib/api/submitPhotoProofs";
import { capturedCount } from "./guideFlow";

/**
 * D11: the ONE answer to "should leaving this guide be confirmed". Both exits ask it —
 * the data-router blocker for an in-app navigation, the `beforeunload` listener for a tab
 * close — so the two can never disagree about whether the host has unsaved work.
 *
 * `isSubmitting` is deliberately NOT a reason to block. While a submission is in flight
 * the page renders a full-screen overlay, so the host cannot reach a link anyway, and a
 * confirmation stacked on top of that overlay would be a trap rather than a protection.
 * The `beforeunload` warning still fires while submitting (D10) — that is the one exit
 * where a closed tab genuinely costs an upload.
 */
export function canLeaveGuide(
  shots: CapturedApartmentShot[],
  isSubmitting: boolean,
): boolean {
  if (isSubmitting) return true;

  return capturedCount(shots) === 0;
}
```

**Accept.** `npx tsc -b` → exit 0.

**Verify.** `npm run gate`

### Step 6.2 — Test the exit rule

**Touch.** `src/lib/helpers/guideExit.test.ts` (create)

```ts
import { describe, expect, it } from "vitest";

import type { CapturedApartmentShot } from "@/lib/api/submitPhotoProofs";
import { canLeaveGuide } from "./guideExit";

function shots(captured: number[]): CapturedApartmentShot[] {
  return [0, 1, 2].map((index) => ({
    id: `shot-${index}`,
    apartmentId: "apartment-1",
    roomLocation: "KITCHEN",
    shotType: "SWEEP_ONLY",
    title: `Shot ${index}`,
    instructions: `Instructions ${index}`,
    assetIds: [],
    createdAt: "2026-09-01T14:00:00.000Z",
    updatedAt: "2026-09-01T14:00:00.000Z",
    ...(captured.includes(index) ? { capturedImageUrl: `blob:preview-${index}` } : {}),
  }));
}

describe("canLeaveGuide", () => {
  it("is true when nothing has been captured — there is nothing to lose", () => {
    expect(canLeaveGuide(shots([]), false)).toBe(true);
  });

  it("is false as soon as one photo is captured", () => {
    expect(canLeaveGuide(shots([0]), false)).toBe(false);
  });

  it("is false when every step is captured and nothing is submitted yet", () => {
    // The review screen is exactly this state, and it is the one where leaving costs the
    // most — a full set of photos.
    expect(canLeaveGuide(shots([0, 1, 2]), false)).toBe(false);
  });

  it("is true while submitting, so the overlay is not a trap (D11)", () => {
    expect(canLeaveGuide(shots([0, 1, 2]), true)).toBe(true);
  });

  it("is true on a guide with no steps at all", () => {
    expect(canLeaveGuide([], false)).toBe(true);
  });
});
```

### Step 6.3 — Create the guard hook

**Touch.** `src/hooks/useGuideExitGuard.ts` (create)

**Do.**

```ts
import { useEffect } from "react";
import { useBlocker } from "react-router";

/**
 * D10: the two exits from a guide with unsaved photos, in one place.
 *
 *   in-app navigation → `useBlocker` (D9). It answers only for navigations the data
 *                       router performs, which is why the listener below is not redundant.
 *   tab close/refresh → `beforeunload`, the behaviour the old `ApartmentShotGuide` had
 *                       while uploading, kept and broadened to "has unsaved photos".
 *
 * `canLeave` must be passed in already-answered (`canLeaveGuide`, phase 6.1) rather than
 * the shots: the rule is the helper's, and a hook that decided it would be a second place
 * the two exits could drift apart.
 *
 * `useBlocker` is called unconditionally on every render — a hook behind a condition would
 * break the render order — so the boolean is the switch, not the call.
 */
export function useGuideExitGuard(canLeave: boolean) {
  const blocker = useBlocker(!canLeave);

  useEffect(() => {
    if (canLeave) return;

    const handleBeforeUnload = (event: BeforeUnloadEvent) => {
      event.preventDefault();
      event.returnValue = "";
    };

    window.addEventListener("beforeunload", handleBeforeUnload);
    return () => window.removeEventListener("beforeunload", handleBeforeUnload);
  }, [canLeave]);

  return blocker;
}
```

**Notes for the executor, not part of the Do block:** `Blocker`'s shape is
`{ state: "unblocked" | "blocked" | "proceeding", proceed?: () => void, reset?: () => void, location? }`
— the dialog rendered in 6.5 reads `blocker.state` and calls `blocker.proceed()` /
`blocker.reset()`. Do not add a second `useBlocker` call anywhere.

**Accept.** `npx tsc -b` → exit 0. `grep -c "useBlocker" src/hooks/useGuideExitGuard.ts` → `2`
(the import and its single call).

### Step 6.4 — Create the leave dialog

**Touch.** `src/components/host/guide/LeaveGuideDialog.tsx` (create)

**Do.**

```tsx
import { useTranslation } from "react-i18next";

interface LeaveGuideDialogProps {
  isOpen: boolean;
  onConfirm: () => void;
  onCancel: () => void;
}

/**
 * D12: not `Modal`. `Modal.tsx` closes on a backdrop click and on Escape, both of which
 * would silently discard the confirmation — the host would think they had been warned and
 * find their photos gone. This asks a question and requires an answer, so neither gesture
 * closes it.
 *
 * The overlay's classes are the ones `ShotsGuide.tsx` used for its uploading modal, so the
 * two overlays this screen can show look like the same thing.
 */
export function LeaveGuideDialog({
  isOpen,
  onConfirm,
  onCancel,
}: LeaveGuideDialogProps) {
  const { t } = useTranslation("assets");

  if (!isOpen) return null;

  return (
    <div className="fixed inset-0 z-50 flex items-center justify-center bg-gray-950/80 p-4 backdrop-blur-md">
      <div
        role="alertdialog"
        aria-modal="true"
        aria-labelledby="leave-guide-title"
        className="w-full max-w-md animate-in space-y-4 rounded-2xl border border-gray-200 bg-white p-6 text-center shadow-2xl fade-in zoom-in-95 duration-200 sm:p-8 dark:border-gray-800 dark:bg-gray-900"
      >
        <div className="text-3xl" aria-hidden="true">
          ⚠️
        </div>

        <div className="space-y-2">
          <h3
            id="leave-guide-title"
            className="text-lg font-bold text-gray-900 sm:text-xl dark:text-gray-100"
          >
            {t("shotGuide.leave.title")}
          </h3>
          <p className="mx-auto max-w-xs text-xs leading-relaxed text-gray-500 sm:text-sm dark:text-gray-400">
            {t("shotGuide.leave.description")}
          </p>
        </div>

        <div className="flex items-center justify-center gap-3 pt-1">
          <button
            type="button"
            onClick={onCancel}
            autoFocus
            className="cursor-pointer rounded-xl bg-gray-900 px-5 py-2.5 text-sm font-medium text-white transition-colors hover:bg-gray-800 dark:bg-gray-100 dark:text-gray-900 dark:hover:bg-gray-200"
          >
            {t("shotGuide.leave.cancel")}
          </button>
          <button
            type="button"
            onClick={onConfirm}
            className="cursor-pointer rounded-xl border border-gray-300 px-5 py-2.5 text-sm font-medium transition-colors hover:bg-gray-50 dark:border-gray-700 dark:hover:bg-gray-800"
          >
            {t("shotGuide.leave.confirm")}
          </button>
        </div>
      </div>
    </div>
  );
}
```
