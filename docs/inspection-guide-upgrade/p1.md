### Step 1.1 — the placement rule, as a pure function

**Touch.** `src/lib/helpers/tooltipPlacement.ts` (create)

**Do.**

```ts
/**
 * Prep §4: where a tooltip panel goes, given the trigger and the box it must stay inside.
 *
 * This module is deliberately NOT the measuring half. `getBoundingClientRect` is DOM and jsdom
 * reports it as all zeros, so a test of the measurement asserts nothing; the *rule* is four
 * comparisons and every one of them is a test case in the file beside this one. The same split
 * `proofWindow.ts` and `reservationActions.ts` already use.
 *
 * Coordinates are viewport coordinates — the shape `getBoundingClientRect()` returns — because the
 * panel is rendered `position: fixed`. That is what makes scroll and resize a re-measure rather
 * than an offset calculation.
 */

/** A rectangle in viewport coordinates. Satisfied by a `DOMRect`. */
export interface Rect {
  top: number;
  left: number;
  right: number;
  bottom: number;
  width: number;
  height: number;
}

export interface TooltipPlacementInput {
  /** The control the panel hangs off. */
  trigger: Rect;
  /** The box the panel may not cross — `PageLayout`'s element, or the viewport. */
  host: Rect;
  /** The width the panel would like. Clamped to the host, never the other way round. */
  preferredWidth: number;
  /** The panel's own measured height. The caller measures it; this function cannot. */
  panelHeight: number;
  /** Space between the trigger and the panel. Defaults to `TOOLTIP_GAP`. */
  gap?: number;
  /** Which side to try first. Defaults to below. */
  preferred?: "below" | "above";
}

export interface TooltipPlacement {
  /** Viewport x of the panel's top-left corner. */
  left: number;
  /** Viewport y of the panel's top-left corner. */
  top: number;
  /** The panel must render at this width or it can exceed the host. */
  width: number;
  /** Which side of the trigger the panel ended up on — also the arrow's side. */
  placeBelow: boolean;
  /** Which horizontal edge is flush with the trigger — also the arrow's side. */
  align: "left" | "right";
}

/** The house tooltip width: the `w-64` this replaced (decisions D32). */
export const DEFAULT_TOOLTIP_WIDTH = 288;

/** The house gap between a trigger and its panel. */
export const TOOLTIP_GAP = 8;

export function computeTooltipPlacement(
  input: TooltipPlacementInput,
): TooltipPlacement {
  const gap = input.gap ?? TOOLTIP_GAP;

  // Prep §4, sentence 3: the panel is never wider than the box it lives in. This is the one
  // clamp that has to happen before anything else, because every horizontal decision below
  // assumes a width that fits.
  const width = Math.min(input.preferredWidth, input.host.width);

  // Sentence 2, vertical half: below by default, above when below overflows and above does not.
  const belowTop = input.trigger.bottom + gap;
  const aboveTop = input.trigger.top - gap - input.panelHeight;
  const fitsBelow = belowTop + input.panelHeight <= input.host.bottom;
  const fitsAbove = aboveTop >= input.host.top;

  const preferBelow = input.preferred !== "above";
  // When neither side fits, the requested side wins and the clamp below pulls it back in —
  // flipping to a side that also overflows would only move which edge is cut off.
  const placeBelow = preferBelow
    ? fitsBelow || !fitsAbove
    : !fitsAbove && fitsBelow;

  const rawTop = placeBelow ? belowTop : aboveTop;
  const maxTop = Math.max(input.host.top, input.host.bottom - input.panelHeight);
  const top = Math.min(Math.max(rawTop, input.host.top), maxTop);

  // Sentence 2, horizontal half: point the panel away from the crowded edge. A trigger in the
  // host's right half gets a panel whose right edge is flush with it, and vice versa — which is
  // "if they are opened on the far right they should be pointed to left where the space is".
  const triggerCenter = input.trigger.left + input.trigger.width / 2;
  const hostCenter = input.host.left + input.host.width / 2;
  const align: "left" | "right" = triggerCenter <= hostCenter ? "left" : "right";

  const rawLeft =
    align === "left" ? input.trigger.left : input.trigger.right - width;
  // Sentence 3, horizontal half. `maxLeft` cannot go below `minLeft`, because `width` was already
  // clamped to the host — so a panel wider than its host pins to the host's left edge instead of
  // producing an inverted range.
  const minLeft = input.host.left;
  const maxLeft = Math.max(minLeft, input.host.right - width);
  const left = Math.min(Math.max(rawLeft, minLeft), maxLeft);

  return { left, top, width, placeBelow, align };
}
```

import { describe, expect, it } from "vitest";

import {
  DEFAULT_TOOLTIP_WIDTH,
  TOOLTIP_GAP,
  computeTooltipPlacement,
  type Rect,
} from "./tooltipPlacement";

/** A 1000×800 page wrapper, which is roughly `PageLayout size="lg"` on a laptop. */
function host(): Rect {
  return { top: 0, left: 0, right: 1000, bottom: 800, width: 1000, height: 800 };
}

/** A 20×20 icon button sitting at (left, top). */
function trigger(left: number, top: number, size = 20): Rect {
  return {
    left,
    top,
    right: left + size,
    bottom: top + size,
    width: size,
    height: size,
  };
}

const PANEL_HEIGHT = 100;

describe("computeTooltipPlacement", () => {
  it("sits below the trigger, flush left, when there is room below", () => {
    const placement = computeTooltipPlacement({
      trigger: trigger(300, 100),
      host: host(),
      preferredWidth: DEFAULT_TOOLTIP_WIDTH,
      panelHeight: PANEL_HEIGHT,
    });

    expect(placement.placeBelow).toBe(true);
    expect(placement.align).toBe("left");
    expect(placement.left).toBe(300);
    expect(placement.top).toBe(100 + 20 + TOOLTIP_GAP);
    expect(placement.width).toBe(DEFAULT_TOOLTIP_WIDTH);
  });

  it("flips above when below would overflow the host and above would not", () => {
    const placement = computeTooltipPlacement({
      trigger: trigger(300, 750),
      host: host(),
      preferredWidth: DEFAULT_TOOLTIP_WIDTH,
      panelHeight: PANEL_HEIGHT,
    });

    expect(placement.placeBelow).toBe(false);
    expect(placement.top).toBe(750 - TOOLTIP_GAP - PANEL_HEIGHT);
  });

  it("honours a preferred side when both sides fit", () => {
    const placement = computeTooltipPlacement({
      trigger: trigger(300, 400),
      host: host(),
      preferredWidth: DEFAULT_TOOLTIP_WIDTH,
      panelHeight: PANEL_HEIGHT,
      preferred: "above",
    });

    expect(placement.placeBelow).toBe(false);
    expect(placement.top).toBe(400 - TOOLTIP_GAP - PANEL_HEIGHT);

      top: 0,
      left: 0,
      right: 1000,
      bottom: 200,
      width: 1000,
      height: 200,

      trigger: trigger(300, 90),
      host: shortHost,
      preferredWidth: DEFAULT_TOOLTIP_WIDTH,
      panelHeight: PANEL_HEIGHT,

    expect(placement.placeBelow).toBe(true);
    expect(placement.top).toBe(200 - PANEL_HEIGHT);

      host: host(),

      panelHeight: PANEL_HEIGHT,

    expect(placement.align).toBe("right");
    // The panel's right edge is flush with the trigger's right edge (720).
    expect(placement.left).toBe(720 - DEFAULT_TOOLTIP_WIDTH);

      top: 0,
      left: 0,
      right: 200,
      bottom: 800,
      width: 200,
      height: 800,

      trigger: trigger(40, 100),
      host: narrowHost,
      preferredWidth: DEFAULT_TOOLTIP_WIDTH,
      panelHeight: PANEL_HEIGHT,

    expect(placement.left).toBe(0);

      trigger: trigger(-50, 100, 100),
      host: host(),
      preferredWidth: DEFAULT_TOOLTIP_WIDTH,
      panelHeight: PANEL_HEIGHT,

    expect(placement.left).toBe(0);
  });

  it("clamps the right edge back into the host when the trigger overhangs it", () => {

      top: 0,
      left: 0,
      right: 700,
      bottom: 800,
      width: 700,
      height: 800,
    };

    const placement = computeTooltipPlacement({
      trigger: trigger(900, 100, 100),
      host: narrowHost,
      preferredWidth: DEFAULT_TOOLTIP_WIDTH,
      panelHeight: PANEL_HEIGHT,
    });

    expect(placement.left).toBe(700 - DEFAULT_TOOLTIP_WIDTH);
  });

  it("takes a gap override", () => {
    const placement = computeTooltipPlacement({
      trigger: trigger(300, 100),
      host: host(),
      preferredWidth: DEFAULT_TOOLTIP_WIDTH,
      panelHeight: PANEL_HEIGHT,
      gap: 20,
    });

    expect(placement.top).toBe(100 + 20 + 20);
  });

  it("pins to the host's top when the panel is taller than the host", () => {
    const shortHost: Rect = {
      top: 50,
      left: 0,
      right: 1000,
      bottom: 120,
      width: 1000,
      height: 70,
    };

    const placement = computeTooltipPlacement({
      trigger: trigger(300, 60),
      host: shortHost,
      preferredWidth: DEFAULT_TOOLTIP_WIDTH,
      panelHeight: 200,
    });

    expect(placement.top).toBe(50);
  });
});

### Step 1.3 — the measuring half

**Do.**

```ts
import { useCallback, useEffect, useLayoutEffect, useRef, useState } from "react";
import type { RefObject } from "react";

import {
  DEFAULT_TOOLTIP_WIDTH,
  computeTooltipPlacement,
  type Rect,
  type TooltipPlacement,
} from "@/lib/helpers/tooltipPlacement";

/**
 * Prep §4's boundary, as a selector. `PageLayout` carries it; a panel whose trigger has no such
 * ancestor is clamped to the viewport instead (decisions D28).
 */
export const TOOLTIP_HOST_ATTR = "data-tooltip-host";

const EMPTY_RECT: Rect = {
  top: 0,
  left: 0,
  right: 0,
  bottom: 0,
  width: 0,
  height: 0,
};

function viewportRect(): Rect {
  const width = window.innerWidth;
  const height = window.innerHeight;

  return { top: 0, left: 0, right: width, bottom: height, width, height };
}

/**
 * Measures a tooltip's trigger, its host and the panel itself, and asks
 * `computeTooltipPlacement` where the panel goes. The DOM half of phase 1 — the rule is in
 * `lib/helpers/`, this is only the reading.
 *
 * The effect has NO dependency array, on purpose. The panel's width is set BY the placement and
 * its height is an input TO the placement, so the first pass measures a panel that is still at
 * width 0 — wrapped to one character per line and therefore far too tall — and the second pass
 * measures the real thing. The `setPlacement` below returns the previous object whenever nothing
 * changed, which makes React bail out of the re-render and is what stops the third pass being a
 * fourth. Never drop that comparison guard.
 */
export function useTooltipPlacement(
  triggerRef: RefObject<HTMLElement | null>,
  preferred: "below" | "above" = "below",
) {
  const panelRef = useRef<HTMLDivElement>(null);
  const [placement, setPlacement] = useState<TooltipPlacement>(() =>
    computeTooltipPlacement({
      trigger: EMPTY_RECT,
      host: EMPTY_RECT,
      preferredWidth: DEFAULT_TOOLTIP_WIDTH,
      panelHeight: 0,
      preferred,
    }),
  );

  const measure = useCallback(() => {
    const trigger = triggerRef.current;
    const panel = panelRef.current;
    if (!trigger || !panel) return;

    const host = trigger.closest(`[${TOOLTIP_HOST_ATTR}]`);
    const next = computeTooltipPlacement({
      trigger: trigger.getBoundingClientRect(),
      host: host ? host.getBoundingClientRect() : viewportRect(),
      preferredWidth: DEFAULT_TOOLTIP_WIDTH,
      panelHeight: panel.offsetHeight,
      preferred,
    });

    setPlacement((previous) =>
      previous.left === next.left &&
      previous.top === next.top &&
      previous.width === next.width &&
      previous.placeBelow === next.placeBelow &&
      previous.align === next.align
        ? previous
        : next,
    );
  }, [preferred, triggerRef]);

  useLayoutEffect(() => {
    measure();
  });

  useEffect(() => {
    window.addEventListener("scroll", measure, true);
    window.addEventListener("resize", measure);

    return () => {
      window.removeEventListener("scroll", measure, true);
      window.removeEventListener("resize", measure);
    };
  }, [measure]);

  return { panelRef, placement };
}
```

### Step 1.4 — the panel

**Touch.** `src/components/shared/TooltipPanel.tsx` (create)

**Do.**

```tsx
import type { ReactNode, RefObject } from "react";

import { useTooltipPlacement } from "@/hooks/useTooltipPlacement";
import { cn } from "@/lib/utils";

export interface TooltipPanelProps {
  /**
   * The control the panel hangs off. It must sit inside a `[data-tooltip-host]` box for the
   * panel to be clamped to the page wrapper; without one it is clamped to the viewport.
   */
  triggerRef: RefObject<HTMLElement | null>;
  children: ReactNode;
  /** Which side to try first. Defaults to below. */
  preferred?: "below" | "above";
}

/**
 * The app's one tooltip panel (prep §4). It owns the measured position so neither of the two
 * tooltip components has to.
 *
 * It is `fixed` and its coordinates are viewport coordinates, which is why the hook re-measures
 * on scroll and resize instead of computing an offset.
 *
 * It is `invisible` rather than `hidden`: a `display: none` panel has no `offsetHeight`, and the
 * panel's height is an input to its own placement. `group-hover` / `group-focus-within` are
 * read off the `group/tooltip` wrapper the CALLER renders — this component supplies no trigger
 * and no group, because both call sites already have one.
 *
 * The wrapper the caller renders must be `position: static` (or `relative`, which is the same
 * for a fixed child): `position: fixed` is relative to the viewport, so the wrapper cannot move
 * the panel, it only has to be the hover/focus group.
 *
 * The panel is a plain flex item, so unlike the version this replaced it has ink — an
 * absolutely-positioned box with no ink does not grow an inline formatting context, which is
 * what let the old panel escape its row.
 */
export function TooltipPanel({
  triggerRef,
  children,
  preferred,
}: TooltipPanelProps) {
  const { panelRef, placement } = useTooltipPlacement(triggerRef, preferred);

  return (
    <div
      ref={panelRef}
      role="tooltip"
      style={{
        left: `${placement.left}px`,
        top: `${placement.top}px`,
        width: `${placement.width}px`,
      }}
      className="pointer-events-none invisible fixed z-30 opacity-0 transition-opacity duration-150 group-hover/tooltip:visible group-hover/tooltip:opacity-100 group-focus-within/tooltip:visible group-focus-within/tooltip:opacity-100"
    >
      <div className="relative rounded-xl border border-gray-800 bg-gray-900 p-3 text-xs font-normal text-gray-100 shadow-xl dark:border-gray-700 dark:bg-gray-950">
        <span
          aria-hidden="true"
          className={cn(
            "absolute h-2 w-2 rotate-45 border-gray-800 bg-gray-900 dark:border-gray-700 dark:bg-gray-950",
            placement.placeBelow
              ? "-top-1 border-l border-t"
              : "-bottom-1 border-b border-r",
            placement.align === "left" ? "left-2.5" : "right-2.5",
          )}
        />
        <div className="relative z-10 whitespace-normal break-words leading-relaxed text-gray-200">
          {children}
        </div>
      </div>
    </div>
  );
}
```
