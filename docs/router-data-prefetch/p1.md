# Phase 1 — one definition per query

**Prerequisite:** none.
**Gate:** `npm run gate`

This phase creates the substrate the rest of the feature stands on: one module per resource holding
**key and fetcher together**, so a route's prefetch and a page's hook cannot drift (D1, §4). It
closes **hole 4** (the `staleTime` mismatch) by exporting a single freshness constant both sides use.

Nothing is rewired yet — the pages still call their own inline `useQuery`. Phase 3 moves them.

---

### Step 1.1 — Add the shared prefetch helper

**Touch.** `src/lib/api/queries/shared.ts` (create)

**Do.**

```ts
/* eslint-disable @typescript-eslint/no-explicit-any */
import { queryClient } from "@/lib/api/queryClient";

/**
 * D5: ONE freshness window, shared by a route's prefetch and the page hook that
 * reads the same key. Prefetches used 30 s while `queryClient` defaulted to
 * 15 min, so a hover after 30 s refetched a query the page still considered
 * fresh — a wasted request (hole 4). The value stays 30 s rather than 15 min: a
 * prefetch is speculative, and it must not pin stale data on screen.
 */
export const PREFETCH_STALE_TIME_MS = 30_000;

export interface PrefetchSpec {
  queryKey: readonly unknown[];
  queryFn: (ctx: { signal: AbortSignal }) => Promise<unknown>;
}

/**
 * D14: a prefetch is best-effort. It must never reject into the router, and a
 * failure must never surface as a route error — the page's own `useQuery`
 * retries it and owns the error branch. `preloadRegistry` catches synchronously;
 * this catches the promise.
 */
export function prefetchQuietly(spec: PrefetchSpec): void {
  void queryClient
    .prefetchQuery({
      queryKey: spec.queryKey,
      queryFn: spec.queryFn,
      staleTime: PREFETCH_STALE_TIME_MS,
    })
    .catch(() => undefined);
}

/**
 * One resource = one key builder + one fetcher, defined together so a route's
 * prefetch and a page's hook cannot drift (§4: same key = same args, or the
 * cache sees two entries and fetches twice).
 *
 * `params` is deliberately loose at this boundary: each module keeps a precise
 * hook signature for its own callers, while the registry only forwards a bag.
 */
export interface ResourceDefinition {
  action: readonly string[];
  key: (params: any) => readonly unknown[];
  fetch: (params: any, ctx: { signal: AbortSignal }) => Promise<unknown>;
}
```

```ts
import { getApartmentsById, getApartmentsHostMe } from "@/api/generated/requests/sdk.gen";
import { withAuth } from "@/lib/api/api";
import { QUERY_ACTIONS } from "@/lib/api/queryKeys";
import { useQuery } from "@tanstack/react-query";
import { PREFETCH_STALE_TIME_MS } from "./shared";
import type { ResourceDefinition } from "./shared";

export interface ApartmentParams {
  id: string;
}

export function apartmentQueryKey(id: string) {
  return [...QUERY_ACTIONS.APARTMENTS_GET_ID, id] as const;
}

export async function fetchApartment(id: string, signal?: AbortSignal) {
  const config = await withAuth(signal ? { signal } : {});
  const response = await getApartmentsById({ ...config, path: { id } });
  return response.data;
}

export function useApartment(id: string | undefined) {
  return useQuery({
    queryKey: apartmentQueryKey(id ?? ""),
    queryFn: async ({ signal }) => {
      if (!id) throw new Error("apartmentIdRequired");
      return fetchApartment(id, signal);
    },
    enabled: Boolean(id),
    // D5: the same window the route's prefetch uses, from the one constant.
    staleTime: PREFETCH_STALE_TIME_MS,
  });
}

/** R15: the host's own apartment list — the `host` section prefetches this once. */
export function apartmentsHostQueryKey() {
  return [...QUERY_ACTIONS.APARTMENTS_GET_HOST] as const;
}

export async function fetchApartmentsHost(signal?: AbortSignal) {
  const config = await withAuth(signal ? { signal } : {});
  const response = await getApartmentsHostMe(config);
  return response.data;
}

export function useApartmentsHost() {
  return useQuery({
    queryKey: apartmentsHostQueryKey(),
    queryFn: ({ signal }) => fetchApartmentsHost(signal),
    staleTime: PREFETCH_STALE_TIME_MS,
  });
}

export const apartmentResource: ResourceDefinition = {
  action: QUERY_ACTIONS.APARTMENTS_GET_ID,
  key: ({ id }: ApartmentParams) => apartmentQueryKey(id),
  fetch: ({ id }: ApartmentParams, { signal }) => fetchApartment(id, signal),
};

export const apartmentsHostResource: ResourceDefinition = {
  action: QUERY_ACTIONS.APARTMENTS_GET_HOST,
  key: () => apartmentsHostQueryKey(),
  fetch: (_params: unknown, { signal }) => fetchApartmentsHost(signal),
};
```

`src/lib/api/queries/reservations.ts` — the key must reproduce the page's **exact** five-element
tail, because a key that differs by one element is a second cache entry and a second request (§4).
The page is `src/components/host/ReservationsSection.tsx`; reproduce its literal verbatim,
including the trailing `"asc"`:

```ts
import { getReservationsApartmentByApartmentId } from "@/api/generated/requests/sdk.gen";
import { withAuth } from "@/lib/api/api";
import { QUERY_ACTIONS } from "@/lib/api/queryKeys";
import { useQuery } from "@tanstack/react-query";
import { PREFETCH_STALE_TIME_MS } from "./shared";
import type { ResourceDefinition } from "./shared";

export const RESERVATIONS_DEFAULT_PAGE = 1;
export const RESERVATIONS_DEFAULT_LIMIT = 10;

export interface ReservationsParams {
  apartmentId: string;
  page?: number;
  limit?: number;
}

/**
 * §4: this key reproduces `ReservationsSection`'s literal EXACTLY — same element
 * count and order, including the trailing "asc". The request below sends
 * `order: "desc"`; the key's "asc" is pre-existing (see Noticed) and is preserved
 * because changing it would change cache identity and invalidate nothing.
 */
export function reservationsQueryKey({
  apartmentId,
  page = RESERVATIONS_DEFAULT_PAGE,
  limit = RESERVATIONS_DEFAULT_LIMIT,
}: ReservationsParams) {
  return [
    ...QUERY_ACTIONS.RESERVATIONS_GET_BY_APARTMENT,
    apartmentId,
    page,
    limit,
    "checkInDatetime",
    "asc",
  ] as const;
}

export async function fetchReservations(
  { apartmentId, page = RESERVATIONS_DEFAULT_PAGE, limit = RESERVATIONS_DEFAULT_LIMIT }: ReservationsParams,
  signal?: AbortSignal,
) {
  const config = await withAuth(signal ? { signal } : {});
  const response = await getReservationsApartmentByApartmentId({
    ...config,
    path: { apartmentId },
    query: {
      page,
      limit,
      sortBy: "checkInDatetime",
      history: true,
      order: "desc",
    },
  });
  return response.data;
}

export function useReservations(params: ReservationsParams) {
  return useQuery({
    queryKey: reservationsQueryKey(params),
    queryFn: ({ signal }) => fetchReservations(params, signal),
    enabled: !!params.apartmentId,
    staleTime: PREFETCH_STALE_TIME_MS,
  });
}

export const reservationsResource: ResourceDefinition = {
  action: QUERY_ACTIONS.RESERVATIONS_GET_BY_APARTMENT,
  key: (params: ReservationsParams) => reservationsQueryKey(params),
  fetch: (params: ReservationsParams, { signal }) => fetchReservations(params, signal),
};
```

### Step 1.5 — Build the resource registry

**Touch.** `src/lib/api/queries/registry.ts` (create)

**Do.**

```ts
import type { ParsedParams } from "@/lib/router/params";
import { apartmentResource, apartmentsHostResource } from "./apartment";
import { assetsResource } from "./assets";
import { shotsResource } from "./shots";
import { reservationsResource } from "./reservations";
import { inspectionResource } from "./inspection";
import {
  apartmentsAllResource,
  locationsResource,
  usersResource,
} from "./admin";
import type { PrefetchSpec, ResourceDefinition } from "./shared";

/**
 * Every prefetchable resource, by name. A route declares `data: ["assets", …]`
 * and the builder resolves each name here — which is what lets the coverage
 * check (phase 4) see a route's data dependencies as DATA rather than as an
 * opaque callback (D2).
 */
export const QUERY_RESOURCES = {
  apartment: apartmentResource,
  apartmentsHost: apartmentsHostResource,
  apartmentsAll: apartmentsAllResource,
  assets: assetsResource,
  shots: shotsResource,
  reservations: reservationsResource,
  inspection: inspectionResource,
  users: usersResource,
  locations: locationsResource,
} as const satisfies Record<string, ResourceDefinition>;

export type ResourceName = keyof typeof QUERY_RESOURCES;

/**
 * `params` returns the values the resource needs, or `null` when THIS navigation
 * does not need it — the tab-gated case (`?tab=apartments`), where prefetching
 * would waste a request for a tab the user may never open (D9, D13).
 */
export interface ResourceRequest {
  resource: ResourceName;
  params: (
    params: ParsedParams,
    searchParams: URLSearchParams,
  ) => Record<string, unknown> | null;
}

/**
 * Resolves a declaration to a concrete prefetch, or `null` when the declaration
 * declines this navigation. Returning `null` is normal, not an error.
 */
export function resolveResource(
  request: ResourceRequest,
  params: ParsedParams,
  searchParams: URLSearchParams,
): PrefetchSpec | null {
  const values = request.params(params, searchParams);
  if (values === null) return null;

  const definition = QUERY_RESOURCES[request.resource];

  return {
    queryKey: definition.key(values),
    queryFn: ({ signal }) => definition.fetch(values, { signal }),
  };
}
```

### Step 1.6 — Test the registry

**Touch.** `src/lib/api/queries/registry.test.ts` (create)

**Do.**

```ts
import { describe, expect, it } from "vitest";
import { QUERY_RESOURCES, resolveResource } from "./registry";
import type { ResourceName } from "./registry";
import { QUERY_ACTIONS } from "@/lib/api/queryKeys";
import { PREFETCH_STALE_TIME_MS } from "./shared";

const NAMES = Object.keys(QUERY_RESOURCES) as ResourceName[];

describe("QUERY_RESOURCES", () => {
  it("every entry carries a registry action, never an ad-hoc literal", () => {
    const known = Object.values(QUERY_ACTIONS) as readonly (readonly string[])[];

    for (const name of NAMES) {
      expect(
        known.some(
          (action) =>
            action.length === QUERY_RESOURCES[name].action.length &&
            action.every((part, i) => part === QUERY_RESOURCES[name].action[i]),
        ),
        `${name} action is not in QUERY_ACTIONS`,
      ).toBe(true);
    }
  });

  it("every resource can build a key without throwing", () => {
    for (const name of NAMES) {
      expect(() => QUERY_RESOURCES[name].key({}), name).not.toThrow();
    }
  });

  it("keys are non-empty arrays, so two resources cannot collide on []", () => {
    for (const name of NAMES) {
      const key = QUERY_RESOURCES[name].key({});
      expect(Array.isArray(key), name).toBe(true);
      expect(key.length, name).toBeGreaterThan(0);
    }
  });

  it("uses one shared freshness window", () => {
    expect(PREFETCH_STALE_TIME_MS).toBe(30_000);
  });
});

describe("resolveResource", () => {
  it("builds a spec whose key starts with the resource's own action", () => {
    const spec = resolveResource(
      { resource: "assets", params: (params) => ({ apartmentId: String(params.id) }) },
      { id: "3f2504e0-4f89-41d3-9a0c-0305e82c3301" },
      new URLSearchParams(),
    );

    expect(spec?.queryKey).toEqual([
      ...QUERY_ACTIONS.ASSETS_GET_BY_APARTMENT,
      "3f2504e0-4f89-41d3-9a0c-0305e82c3301",
    ]);
  });

  it("returns null when the declaration declines this navigation", () => {
    const spec = resolveResource(
      { resource: "assets", params: () => null },
      {},
      new URLSearchParams(),
    );

    expect(spec).toBeNull();
  });

  it("passes searchParams through, so a tab-gated declaration can read ?tab=", () => {
    const spec = resolveResource(
      {
        resource: "users",
        params: (_params, searchParams) =>
          searchParams.get("tab") === "users" ? {} : null,
      },
      {},
      new URLSearchParams("tab=users"),
    );

    expect(spec?.queryKey).toEqual([...QUERY_ACTIONS.USERS_GET_ALL]);
  });
});
```
