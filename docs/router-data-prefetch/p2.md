# Phase 2 — routes declare their data

**Prerequisite:** phase 1 (the registry the `data` field resolves through).
**Gate:** `npm run gate`

This phase makes a route's data dependencies **declarative** and closes **hole 1** — the one real
bug in the router itself. `buildPreload` currently bails when a node has no page/layout module:

```ts
const loadModule = (node as any).page ?? (node as any).layout;
if (!loadModule) return undefined;   // ← a group route with `prefetch` dies here
```

`hostRoutes` is a group (path + requires + prefetch + children, no page), so its section prefetch
never ran on the hover path. This phase also wires `searchParams` through (D3) and deletes the dead
`recordCheck` capability (D10).

### Step 2.2 — Fix `buildPreload` for group routes and resolve `data`

**Touch.** `src/lib/router/defineRoutes.tsx` (edit)

**Do.** Replace `buildPreload` and `guardPasses` with:

```tsx
/**
 * D4: this is hole 1's fix. The old version returned `undefined` when a node had
 * no page/layout module, so a GROUP route's `prefetch` — e.g. the `host`
 * section's apartments query — never ran on the hover path and the ~100 ms
 * intent window was thrown away. A node now preloads when it has a module to
 * import OR data/prefetch work to start; the import is simply skipped when there
 * is no module.
 */
function buildPreload(
  node: RouteNode,
  deps: DefineRoutesDeps,
): ((params: ParsedParams, searchParams: URLSearchParams) => void) | undefined {
  const loadModule = (node as any).page ?? (node as any).layout;
  const prefetch = (node as any).prefetch as
    | ((params: ParsedParams, searchParams: URLSearchParams) => void)
    | undefined;
  const data = ((node as any).data ?? []) as readonly ResourceRequest[];

  if (!loadModule && !prefetch && data.length === 0) return undefined;

  return (params: ParsedParams, searchParams: URLSearchParams) => {
    if (!guardPasses(node, deps, params)) return;

    for (const request of data) {
      const spec = resolveResource(request, params, searchParams);
      if (spec) prefetchQuietly(spec);
    }

    prefetch?.(params, searchParams);
    if (loadModule) void loadModule().catch(() => undefined);
  };
}
```

And in `makeLoader`, replace the `prefetch`/`recordCheck` extraction and body:

```tsx
function makeLoader(
  node: RouteNode,
  guard: GuardMeta,
  deps: DefineRoutesDeps,
): (args: { params: Record<string, string | undefined>; request: Request }) => Promise<null> {
  const loadModule = ((node as any).page ?? (node as any).layout) as
    | (() => Promise<PageModule>)
    | undefined;
  const prefetch = (node as any).prefetch as
    | ((params: ParsedParams, searchParams: URLSearchParams) => void)
    | undefined;
  const data = ((node as any).data ?? []) as readonly ResourceRequest[];

  return async ({ params, request }) => {
    for (const rule of guard.rules) {
      if (!deps.isAllowed(rule)) throw data(null, { status: 403 });
    }

    if (!areParamsValid(params, guard.params)) throw data(null, { status: 404 });

    const parsed = parseParams(params, guard.params);
    const searchParams = new URL(request.url).searchParams;

    // D2: the same work the intent path does, so a cold load (no hover) still
    // starts every request alongside the chunk instead of after it.
    for (const item of data) {
      const spec = resolveResource(item, parsed, searchParams);
      if (spec) prefetchQuietly(spec);
    }

    prefetch?.(parsed, searchParams);

    await (loadModule ? loadModule() : Promise.resolve());

    return null;
  };
}
```

Add the two imports beside the existing `params` import:

```tsx

import type { ResourceRequest } from "@/lib/api/queries/registry";
import { prefetchQuietly } from "@/lib/api/queries/shared";
```

### Step 2.3 — Pass `searchParams` on the intent path

**Touch.** `src/lib/router/createAppRouter.tsx` (edit)

**Do.** In the `registerRoutePreloader` callback, pass the URL's search params through:

```tsx
  const unregisterPreloader = registerRoutePreloader((href) => {
    const url = new URL(href, window.location.origin);
    const matches = matchRoutes(router.routes, url.pathname, basename);

    if (!matches) return;

    // Pass 1: every matched route's whole rule chain and every present param must
    // pass before ANY preload runs. This is what stops one route's chunk
    // downloading when another route in the same match chain is denied.
    for (const match of matches) {
      const guard = (match.route.handle as { guard?: GuardMeta } | undefined)?.guard;
      if (!guard) continue;

      for (const rule of guard.rules) {
        if (!access.isAllowed(rule)) return;
      }

      const present: Record<string, string | undefined> = {};
      for (const key of Object.keys(guard.params)) present[key] = match.params[key];

      if (!areParamsValid(present, guard.params)) return;
    }

    // Pass 2: the checks passed for the whole chain, so start the work.
    // D3: the href's search params go with it, so a declaration can gate on
    // `?tab=` — `Tabs` renders only the active tab, so a tab-scoped query is
    // otherwise unreachable by path params alone.
    for (const match of matches) {
      const guard = (match.route.handle as { guard?: GuardMeta } | undefined)?.guard;
      if (!guard) continue;

      const present: Record<string, string | undefined> = {};
      for (const key of Object.keys(guard.params)) present[key] = match.params[key];

      guard.preload?.(present as ParsedParams, url.searchParams);
    }
  });
```

describe("data declarations and group preloads", () => {
  it("a group route with data gets a preload (hole 1's regression test)", () => {
    const built = build([
      {
        path: "section",
        requires: "public",
        data: [{ resource: "apartmentsHost", params: () => ({}) }],
      } as RouteNode,
    ]);

    const guard = (built[0].handle as { guard?: GuardMeta }).guard;
    const started: string[] = [];
    vi.spyOn(QueryClient.prototype, "prefetchQuery").mockImplementation(
      async (options: { queryKey: readonly unknown[] }) => {
        started.push(JSON.stringify(options.queryKey));
        return undefined as never;
      },
    );

    expect(guard?.preload).toBeDefined();
    guard?.preload?.({}, new URLSearchParams());

    expect(started).toEqual([
      JSON.stringify([...QUERY_ACTIONS.APARTMENTS_GET_HOST]),
    ]);

    vi.restoreAllMocks();
  });

  it("a route with neither a module nor data has no preload", () => {
    const built = build([
      { path: "empty", requires: "public", children: [] } as RouteNode,
    ]);

    const guard = (built[0].handle as { guard?: GuardMeta }).guard;
    expect(guard?.preload).toBeUndefined();
  });

  it("prefetch receives the href's search params", () => {
    const seen: (string | null)[] = [];

    const built = build([
      {
        path: "tabs",
        requires: "public",
        prefetch: (_params, searchParams) => {
          seen.push(searchParams.get("tab"));
        },
      } as RouteNode,
    ]);

    const guard = (built[0].handle as { guard?: GuardMeta }).guard;
    guard?.preload?.({}, new URLSearchParams("tab=users"));

    expect(seen).toEqual(["users"]);
  });

  it("a declined declaration starts nothing", () => {
    const built = build([
      {
        path: "gated",
        requires: "public",
        data: [{ resource: "users", params: () => null }],
      } as RouteNode,
    ]);

    const spy = vi.spyOn(QueryClient.prototype, "prefetchQuery");
    const guard = (built[0].handle as { guard?: GuardMeta }).guard;
    guard?.preload?.({}, new URLSearchParams());

    expect(spy).not.toHaveBeenCalled();
    vi.restoreAllMocks();
  });

  it("the loader no longer carries recordCheck (D10)", async () => {
    const load = loaderOf(
      build([
        {
          path: "apartments/:id",
          requires: "public",
          params: { id: "uuid" },
          page: async () => ({ default: () => null }),
        } as RouteNode,
      ]),
      "/apartments/:id#page",
    );

    await expect(
      load({
        params: { id: VALID_UUID },
        request: new Request(`https://x.test/apartments/${VALID_UUID}`),
      }),
    ).resolves.toBeNull();
  });
});
