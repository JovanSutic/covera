# Phase 4 — the coverage check

**Prerequisite:** phase 3 (the declarations the check asserts against, and a green tree for its own Accept).
**Gate:** `npm run gate`

This is the structural guarantee the operator chose. A page can no longer read a query that no route
in its match chain prefetches — `npm run gate` fails at the step that introduces the gap.

**How it works.** Each route module names its page with a literal specifier
(`page: () => import("@/pages/host/IndividualApartment")`). The check reads those specifiers out of
the route source, follows each page's app-local (`@/…`) imports transitively, and collects every
**named import that is a registered query hook** (a `use…` export of `src/lib/api/queries/*`). That
set is the page's data surface, and it must be a subset of what the route declares.

Two properties of this design were **verified while the plan was written** and are load-bearing:

- It reads **named imports**, not call syntax. `useShots(id)` and `useAssets(id)` are indistinguishable
  to a syntax matcher, but only the import says which module — and therefore which resource — owns it.
- It follows **default imports** too. `Dashboard.tsx` writes `import Tabs from "@/components/Tabs"`,
  so a brace-only import regex silently returns `(none)` and the check passes while proving nothing.
  Step 4.4 guards against exactly that vacuity.

### Step 4.1 — Create the query-hook manifest

**Touch.** `src/lib/api/queries/manifest.ts` (create)

**Do.**

```ts
import * as admin from "./admin";
import * as apartment from "./apartment";
import * as assets from "./assets";
import * as inspection from "./inspection";
import * as reservations from "./reservations";
import * as shots from "./shots";

/**
 * D12: every data-reading hook the app has, named. The coverage check
 * (`src/lib/router/__checks__/prefetchCoverage.test.ts`) treats a page's imports
 * of these names as its data surface, and requires the page's route to declare
 * them.
 *
 * This is a module map, not a hand-written list: a new query module is added here
 * once, and from then on any page that forgets to declare it fails the gate.
 */
const MODULES = { admin, apartment, assets, inspection, reservations, shots } as const;

function collectHooks(): Record<string, string> {
  const hooks: Record<string, string> = {};

  for (const [moduleName, moduleExports] of Object.entries(MODULES)) {
    for (const [exportName, value] of Object.entries(moduleExports)) {
      // A query hook is a `use…` function. Fetchers (`fetchX`) and key builders
      // (`xQueryKey`) are deliberately NOT data surface: a component that builds a
      // key for an invalidation is not starting a request (§4).
      if (exportName.startsWith("use") && typeof value === "function") {
        hooks[exportName] = moduleName;
      }
    }
  }

  return hooks;
}

/** hook name → the query module that owns it. */
export const QUERY_HOOKS: Readonly<Record<string, string>> = collectHooks();
```

**Accept.** `npx tsc -b` → exit 0

**Verify.** `npx tsc -b`

### Step 4.2 — Write the coverage analysis as a pure module

**Touch.** `src/lib/router/__checks__/coverage.ts` (create)

**Do.**

```ts
import fs from "node:fs";
import path from "node:path";
import { QUERY_HOOKS } from "@/lib/api/queries/manifest";

const SRC = path.resolve(process.cwd(), "src");

/**
 * Route modules name their page with a literal specifier. This reads them out of
 * the route source rather than importing the module, because importing a page
 * pulls React, i18n and the generated client into the check.
 */
const PAGE_SPECIFIER = /(?:page|layout):\s*\(\)\s*=>\s*import\(\s*"([^"]+)"\s*\)/g;

/** `import … from "spec"` — every binding form, including default and type-only. */
const IMPORT_WITH_FROM = /import\s+(?:type\s+)?([\s\S]*?)\s+from\s+"([^"]+)"/g;

/** `import "spec"` — side-effect only. */
const IMPORT_SIDE_EFFECT = /import\s+"([^"]+)"/g;

const LOCAL_SPECIFIER = /^@\//;

export interface RouteDeclaration {
  /** The `page: () => import("…")` specifier, as written. */
  specifier: string;
  /** Resource names this route declares via `data:`. */
  resources: string[];
  /** The file the specifier resolves to. */
  file: string;
}

export interface ImportedNames {
  names: string[];
  specifier: string;
}

function resolveAlias(specifier: string): string | null {
  const relative = specifier.replace(/^@\//, "");
  const base = path.join(SRC, relative);
  const candidates = [
    base,
    `${base}.ts`,
    `${base}.tsx`,
    path.join(base, "index.ts"),
    path.join(base, "index.tsx"),
  ];

  for (const candidate of candidates) {
    if (fs.existsSync(candidate) && fs.statSync(candidate).isFile()) return candidate;
  }

  return null;
}

/**
 * Every import in a file with its bound names. Handles `import D from`, `import { A, B as C } from`,
 * `import * as N from`, `import D, { A } from` and `import "side-effect"`.
 * A brace-only regex would miss the default form, which `pages/admin/Dashboard.tsx` uses —
 * verified while writing this plan, and the reason step 4.4 exists.
 */
export function parseImports(source: string): ImportedNames[] {
  const found: ImportedNames[] = [];

  for (const match of source.matchAll(IMPORT_WITH_FROM)) {
    const clause = match[1];
    const names: string[] = [];

    const braces = clause.match(/\{([\s\S]*)\}/);
    if (braces) {
      for (const part of braces[1].split(",")) {
        const name = part.trim().split(/\s+as\s+/)[0].trim();
        if (name) names.push(name);
      }
    }

    const head = clause
      .split("{")[0]
      .replace(/^\s*\*\s*as\s*/, "")
      .split(",")[0]
      .trim();
    if (head) names.push(head);

    found.push({ names, specifier: match[2] });
  }

  for (const match of source.matchAll(IMPORT_SIDE_EFFECT)) {
    found.push({ names: [], specifier: match[1] });
  }

  return found;
}

/** Every `.ts`/`.tsx` under `src/lib/router/routes/`, in a stable order. */
export function routeModuleFiles(): string[] {
  const dir = path.join(SRC, "lib", "router", "routes");
  return fs
    .readdirSync(dir)
    .filter((name) => /\.tsx?$/.test(name) && !name.includes(".test."))
    .map((name) => path.join(dir, name))
    .sort();
}

/**
 * Reads every route module and pairs each page/layout specifier with the `data`
 * resource names declared in the same object literal.
 *
 * The pairing is textual: the source is split on `page:`/`layout:` occurrences, so
 * a `data:` block belongs to the nearest preceding specifier. A route file that
 * reorders these two keys beyond what this can read is a failed assertion in step
 * 4.3, not a silent pass.
 */
export function readRouteDeclarations(): RouteDeclaration[] {
  const declarations: RouteDeclaration[] = [];

  for (const file of routeModuleFiles()) {
    const source = fs.readFileSync(file, "utf8");
    const specifiers = [...source.matchAll(PAGE_SPECIFIER)].map((match) => ({
      specifier: match[1],
      at: match.index ?? 0,
    }));

    specifiers.forEach((entry, index) => {
      const end = specifiers[index + 1]?.at ?? source.length;
      const slice = source.slice(entry.at, end);
      const resources = [...slice.matchAll(/resource:\s*"([a-zA-Z]+)"/g)].map(
        (match) => match[1],
      );

      const resolved = resolveAlias(entry.specifier);
      if (!resolved) {
        throw new Error(
          `[coverage] ${path.relative(SRC, file)} declares page "${entry.specifier}" ` +
            `which does not resolve to a file under src/.`,
        );
      }

      declarations.push({ specifier: entry.specifier, resources, file: resolved });
    });
  }

  return declarations;
}

/**
 * The query hooks a module reads, following its app-local imports transitively.
 * A cycle is guarded by the visited set.
 */
export function hooksUsedBy(entryFile: string, seen = new Set<string>()): Set<string> {
  const hooks = new Set<string>();
  if (seen.has(entryFile) || !fs.existsSync(entryFile)) return hooks;
  seen.add(entryFile);

  for (const { names, specifier } of parseImports(fs.readFileSync(entryFile, "utf8"))) {
    for (const name of names) {
      if (name in QUERY_HOOKS) hooks.add(name);
    }

    // Follow app-local imports, but never into the generated client (it is not
    // data surface) and never into styles or JSON.
    if (!LOCAL_SPECIFIER.test(specifier)) continue;
    if (specifier.startsWith("@/api/")) continue;
    if (/\.(css|json)$/.test(specifier)) continue;

    const resolved = resolveAlias(specifier);
    if (resolved) {
      for (const hook of hooksUsedBy(resolved, seen)) hooks.add(hook);
    }
  }

  return hooks;
}

/**
 * D13: the one honest exemption. Content rendered only when a Drawer is opened is
 * genuinely conditional — the route cannot know about it at navigation time, so
 * requiring a declaration would force a wasted request on every visit.
 *
 * Keys are hook names; each entry carries its reason. Editing this list is a
 * deliberate act, and the reason is what makes it reviewable.
 */
export const INTERACTION_GATED_HOOKS: Readonly<Record<string, string>> = {
  useAssetTypes:
    "CreateAssetForm's asset-type read lives inside a Drawer (rendered only when opened); " +
    "declaring it would fetch a form's options on every navigation to the page.",
};

export interface CoverageGap {
  file: string;
  specifier: string;
  hook: string;
}

/** Every hook a page reads that its route neither declares nor exempts. */
export function findCoverageGaps(): CoverageGap[] {
  const gaps: CoverageGap[] = [];

  for (const declaration of readRouteDeclarations()) {
    const declared = new Set(declaration.resources);

    for (const hook of hooksUsedBy(declaration.file)) {
      if (hook in INTERACTION_GATED_HOOKS) continue;

      const owner = QUERY_HOOKS[hook];
      if (!owner || declared.has(owner)) continue;

      gaps.push({
        file: path.relative(SRC, declaration.file),
        specifier: declaration.specifier,
        hook,
      });
    }
  }

  return gaps;
}
```

### Step 4.3 — Assert the real tree, plus the parser's own guards

**Touch.** `src/lib/router/__checks__/prefetchCoverage.test.ts` (create)

**Do.**

```ts
import { describe, expect, it } from "vitest";
import {
  findCoverageGaps,
  hooksUsedBy,
  parseImports,
  readRouteDeclarations,
  INTERACTION_GATED_HOOKS,
} from "./coverage";
import { QUERY_HOOKS } from "@/lib/api/queries/manifest";

function declarationFor(specifier: string) {
  const found = readRouteDeclarations().find((entry) => entry.specifier === specifier);
  if (!found) throw new Error(`no route declares ${specifier}`);
  return found;
}

describe("the parser sees what it must (guards against a vacuous pass)", () => {
  it("reads a default import, not just a braced one", () => {
    const parsed = parseImports('import Tabs from "@/components/Tabs";\nimport { A, B as C } from "@/x";');

    expect(parsed.map((entry) => entry.specifier)).toContain("@/components/Tabs");
    expect(parsed.find((entry) => entry.specifier === "@/x")?.names).toEqual(["A", "B"]);
  });

  it("finds every page/layout specifier across the route modules", () => {
    const specifiers = readRouteDeclarations().map((entry) => entry.specifier);

    expect(specifiers).toContain("@/pages/host/IndividualApartment");
    expect(specifiers).toContain("@/pages/host/Dashboard");
    expect(specifiers).toContain("@/pages/admin/Dashboard");
    expect(specifiers).toContain("@/pages/InspectionPage");
    expect(specifiers.length).toBeGreaterThanOrEqual(9);
  });

  it("pairs each specifier with the resources declared beside it", () => {
    expect(declarationFor("@/pages/host/IndividualApartment").resources).toEqual([
      "apartment",
      "assets",
      "shots",
      "reservations",
    ]);

    expect(declarationFor("@/pages/InspectionPage").resources).toEqual(["inspection"]);
  });

  it("knows the query hooks", () => {
    expect(QUERY_HOOKS).toHaveProperty("useApartment");
    expect(QUERY_HOOKS).toHaveProperty("useInspection");
    expect(QUERY_HOOKS).toHaveProperty("useApartmentsHost");
  });

  it("detects hooks through a page's transitive local imports", () => {
    // The admin dashboard reads its data in the two section components it renders
    // by DEFAULT import, which a braced-only parser would miss entirely.
    const hooks = hooksUsedBy(declarationFor("@/pages/admin/Dashboard").file);

    expect(hooks).toContain("useUsers");
    expect(hooks).toContain("useApartmentsAll");
  });

  it("detects the host detail page's four hooks", () => {
    const hooks = hooksUsedBy(declarationFor("@/pages/host/IndividualApartment").file);

    expect([...hooks].sort()).toEqual([
      "useApartment",
      "useAssets",
      "useReservations",
      "useShots",
    ]);
  });
});

describe("prefetch coverage (D12)", () => {
  it("every page's queries are declared by its route, or explicitly exempt", () => {
    const gaps = findCoverageGaps();

    // The failure message names the page, the hook and the fix, so whoever adds a
    // query next knows exactly what to declare.
    expect(
      gaps.map(
        (gap) =>
          `${gap.file} reads ${gap.hook} but ${gap.specifier} does not declare it`,
      ),
    ).toEqual([]);
  });

  it("every exemption carries a reason", () => {
    for (const [hook, reason] of Object.entries(INTERACTION_GATED_HOOKS)) {
      expect(reason.length, hook).toBeGreaterThan(20);
    }
  });
});
```

**Accept.** `npx vitest run src/lib/router/__checks__/prefetchCoverage.test.ts` → **8 passed**;

Restore the line, confirm 8 passed again, and record both outputs.
