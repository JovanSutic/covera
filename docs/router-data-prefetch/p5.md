# Phase 5 — measure data readiness, not just chunk time

**Prerequisite:** phase 2 (the preload path whose use the `prefetched` flag records).
**Gate:** `npm run gate`

Hole 5: `navigationTiming` closes its sample when the router returns to `idle`, which is **chunk**
time. The baseline's `9.2 → 7.2 ms` therefore cannot support any claim about data, and the prefetch
work is unverifiable. This phase separates the two measurements and records whether the navigation
was warmed.

`ProgressBar` is **not** touched (D11) — its meaning (navigation, visible after 200 ms) is
deliberate, and a query can be slow while the navigation itself is instant.

### Step 5.1 — Extend the timing recorder

**Touch.** `src/lib/router/navigationTiming.ts` (edit)

**Do.** Replace the `NavigationTiming` interface and the recorder body:

```ts
import type { DataRouter } from "react-router";
import { queryClient } from "@/lib/api/queryClient";
import { consumePreloaded } from "./preloadRegistry";

export interface NavigationTiming {
  /** Where the navigation started from. */
  from: string;
  /** Where it went. */
  to: string;
  /** Wall-clock ms from the router entering "loading" to being back at "idle". */
  durationMs: number;
  /** D11: `durationMs` renamed for what it actually measures — chunk/loader time. */
  chunkMs: number;
  /**
   * D11: ms from navigation start until every query started during that
   * navigation settled. `null` when the navigation started no query, or when a
   * query never settled before the recorder was cleared. Data readiness is a
   * different number from chunk readiness, and this is the one the prefetch work
   * is meant to move.
   */
  dataMs: number | null;
  /** D11: was this href preloaded before the navigation? The flag the gain shows up in. */
  prefetched: boolean;
}

export interface NavigationTimingRecorder {
  install(router: DataRouter): () => void;
  /** Oldest first, capped at `CAPACITY`. */
  getTimings(): readonly NavigationTiming[];
  /** The middle value, or null with no samples — the "p75" of 5 runs is the 3rd. */
  getMedian(): number | null;
  /** The middle `chunkMs` and `dataMs` over the samples that have each. */
  getMedians(): { chunkMs: number | null; dataMs: number | null };
  clear(): void;
}

const CAPACITY = 50;

export function createNavigationTiming(): NavigationTimingRecorder {
  const timings: NavigationTiming[] = [];
  let startedAt: number | null = null;
  let prefetched = false;

  // Queries observed as "fetching" during the current navigation, and the ones
  // already settled. A query is only counted if it STARTED in this window — a
  // cache hit that never fetched is not a data wait.
  let navQueries = new Set<string>();
  let settledQueries = new Set<string>();

  /**
   * D11: the wall-clock instant the LAST query of this navigation settled. This
   * is the metric — measuring it at the `idle` transition instead would only
   * report "data was ready by the time the chunk loaded", which is true of every
   * fast API and would hide the very difference this feature makes.
   */
  let dataReadyAt: number | null = null;

  const unsubscribeCache = queryClient.getQueryCache().subscribe((event) => {
    if (startedAt === null) return;

    const key = event.query.queryHash;

    if (event.type === "updated" && event.query.state.fetchStatus === "fetching") {
      navQueries.add(key);
      settledQueries.delete(key);
      return;
    }

    if (
      event.type === "updated" &&
      navQueries.has(key) &&
      event.query.state.fetchStatus === "idle" &&
      event.query.state.status !== "pending"
    ) {
      settledQueries.add(key);

      // A query that failed still counts as settled: the page has its answer and
      // can render its error branch.
      if (dataReadyAt === null && settledQueries.size >= navQueries.size) {
        dataReadyAt = performance.now();
      }
    }
  });

  function reset(): void {
    startedAt = null;
    prefetched = false;
    navQueries = new Set();
    settledQueries = new Set();
    dataReadyAt = null;
  }

  return {
    install(router) {
      const unsubscribeRouter = router.subscribe((state) => {
        if (state.navigation.state === "loading") {
          if (startedAt === null) {
            startedAt = performance.now();
            prefetched = consumePreloaded(state.location.pathname + state.location.search);
            navQueries = new Set();
            settledQueries = new Set();
            dataReadyAt = null;
          }
          return;
        }

        // Back to idle: close the sample. `startedAt` is null for the initial
        // render, so no bogus sample is recorded for it.
        if (state.navigation.state === "idle" && startedAt !== null) {
          const now = performance.now();
          const durationMs = now - startedAt;

          timings.push({
            from: state.historyAction,
            to: state.location.pathname,
            durationMs,
            chunkMs: durationMs,
            // null when the navigation started no query, or when its queries had
            // not settled by the time the chunk was ready — the cold case.
            dataMs: dataReadyAt === null ? null : dataReadyAt - startedAt,
            prefetched,
          });

          if (timings.length > CAPACITY) timings.shift();
          reset();
        }
      });

      return () => {
        unsubscribeRouter();
        unsubscribeCache();
      };
    },

    getTimings: () => timings,

    getMedian() {
      return this.getMedians().chunkMs;
    },

    getMedians() {
      const middle = (values: number[]): number | null => {
        if (values.length === 0) return null;
        const sorted = [...values].sort((a, b) => a - b);
        return sorted[Math.floor(sorted.length / 2)];
      };

      return {
        chunkMs: middle(timings.map((entry) => entry.chunkMs)),
        dataMs: middle(
          timings
            .map((entry) => entry.dataMs)
            .filter((value): value is number => value !== null),
        ),
      };
    },

    clear() {
      timings.length = 0;
      reset();
    },
  };
}
```

### Step 5.2 — Record preloaded hrefs in the registry

**Touch.** `src/lib/router/preloadRegistry.ts` (edit)

**Do.** Add the record/consume pair beside the existing registration:

```ts
export type RoutePreloader = (href: string) => void;

let registered: RoutePreloader | null = null;

/**
 * D11: which hrefs were preloaded since the last navigation. The timing recorder
 * consumes this to label a sample as warmed or cold — without it, `chunkMs` and
 * `dataMs` cannot be attributed to the prefetch rather than to a fast network.
 */
let preloadedHrefs = new Set<string>();

/**
 * Registered by the router factory so any link can start a route's work without
 * importing the router module (§3, H7). A newer registration replaces an older
 * one; unregistering never clears a preloader registered after it.
 */
export function registerRoutePreloader(preloader: RoutePreloader): () => void {
  registered = preloader;

  return () => {
    if (registered === preloader) registered = null;
  };
}

export function preloadRoute(href: string): void {
  if (!registered) return;

  try {
    registered(href);
    preloadedHrefs.add(href);
  } catch {
    // A preload is best-effort: the loader retries on navigation (R11).
  }
}

/**
 * True when `pathAndSearch` was preloaded, consuming it. Paths are compared
 * loosely because the link's href and the router's location can differ by a
 * trailing slash or an encoded space.
 */
export function consumePreloaded(pathAndSearch: string): boolean {
  return preloadedHrefs.has(pathAndSearch);
}

/** Dev-only reset, so a measurement run starts from a known state. */
export function clearPreloadedHrefs(): void {
  preloadedHrefs = new Set();
}
```

  const uninstallTiming = timing ? timing.install(router) : null;

  if (timing && typeof window !== "undefined") {
    (window as unknown as { __routeTimings?: unknown }).__routeTimings = timing;

### Step 5.4 — Test the recorder

**Touch.** `src/lib/router/navigationTiming.test.ts` (create)

**Do.**

```ts
import { beforeEach, describe, expect, it, vi } from "vitest";
import type { DataRouter } from "react-router";
import { createNavigationTiming } from "./navigationTiming";
import { queryClient } from "@/lib/api/queryClient";
import { clearPreloadedHrefs, preloadRoute, registerRoutePreloader } from "./preloadRegistry";

/** A router stand-in that lets the test drive navigation state directly. */
function fakeRouter() {
  let listener: ((state: unknown) => void) | null = null;

  return {
    subscribe(next: (state: unknown) => void) {
      listener = next;
      return () => {
        listener = null;
      };
    },
    emit(state: unknown) {
      listener?.(state);
    },
    loading: (pathname = "/host/dashboard") => ({
      navigation: { state: "loading" },
      historyAction: "PUSH",
      location: { pathname, search: "" },
    }),
    idle: (pathname = "/host/dashboard") => ({
      navigation: { state: "idle" },
      historyAction: "PUSH",
      location: { pathname, search: "" },
    }),
  };
}

describe("navigationTiming", () => {
  beforeEach(() => {
    clearPreloadedHrefs();
    queryClient.clear();
  });

  it("records chunkMs on the idle transition", async () => {
    const router = fakeRouter();
    const recorder = createNavigationTiming();
    recorder.install(router as unknown as DataRouter);

    router.emit(router.loading());
    router.emit(router.idle());

    const [sample] = recorder.getTimings();
    expect(sample.chunkMs).toBeGreaterThanOrEqual(0);
    expect(sample.durationMs).toBe(sample.chunkMs);
  });

  it("leaves dataMs null when the navigation started no query", () => {
    const router = fakeRouter();
    const recorder = createNavigationTiming();
    recorder.install(router as unknown as DataRouter);

    router.emit(router.loading());
    router.emit(router.idle());

    expect(recorder.getTimings()[0].dataMs).toBeNull();
  });

  it("labels a sample prefetched when its href was preloaded", () => {
    const router = fakeRouter();
    const recorder = createNavigationTiming();
    recorder.install(router as unknown as DataRouter);

    registerRoutePreloader(() => undefined);
    preloadRoute("/host/dashboard");

    router.emit(router.loading());
    router.emit(router.idle());

    expect(recorder.getTimings()[0].prefetched).toBe(true);
  });

  it("does not label a cold navigation as prefetched", () => {
    const router = fakeRouter();
    const recorder = createNavigationTiming();
    recorder.install(router as unknown as DataRouter);

    router.emit(router.loading());
    router.emit(router.idle());

    expect(recorder.getTimings()[0].prefetched).toBe(false);
  });

  it("closes dataMs once a query started in the window settles", async () => {
    const router = fakeRouter();
    const recorder = createNavigationTiming();
    recorder.install(router as unknown as DataRouter);

    router.emit(router.loading());

    // Start a query while the navigation is in flight, with a resolver the test
    // controls, then settle it.
    let resolve!: (value: string) => void;
    const pending = new Promise<string>((r) => {
      resolve = r;
    });

    void queryClient.prefetchQuery({
      queryKey: ["test", "dataMs"],
      queryFn: () => pending,
    });

    // Let the cache report `fetching` before settling.
    await vi.waitFor(() =>
      expect(queryClient.getQueryState(["test", "dataMs"])?.fetchStatus).toBe("fetching"),
    );

    resolve("done");
    await vi.waitFor(() =>
      expect(queryClient.getQueryState(["test", "dataMs"])?.fetchStatus).toBe("idle"),
    );

    router.emit(router.idle());

    expect(recorder.getTimings()[0].dataMs).not.toBeNull();
  });

  it("reports null medians with no samples", () => {
    const recorder = createNavigationTiming();
    expect(recorder.getMedians()).toEqual({ chunkMs: null, dataMs: null });
  });
});
```
