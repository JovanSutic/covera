# Phase 2 — typed params and the access layer

**Prerequisite:** phase 1 (`SessionStore`, `SessionState`).
**Gate:** `npm run gate` — must be green before step 2.1.

**Why this phase is second:** playbook stage 1 (R2, R13). Every later phase calls
`deps.isAllowed(rule)` from a loader and `areParamsValid` before a download. Both are pure functions
here, which is why this phase carries two test files and no UI.

**Observable gain.** None on its own — this phase is a library. Its gain is *removal*: the async
per-navigation guard it replaces is deleted in phase 4, and this is what makes that possible. It ships
behind a green gate with tests, so it is independently runnable.

### Step 2.1 — Param types

**Touch.** `src/lib/router/params.ts` (create)

**Do.**

```ts
export type ParamType = "number" | "uuid" | "numbersList" | "name" | { oneOf: readonly string[] };

export type ParamsDeclaration = Record<string, ParamType>;

const UUID_PATTERN =
  /^[0-9a-f]{8}-[0-9a-f]{4}-[0-9a-f]{4}-[0-9a-f]{4}-[0-9a-f]{12}$/i;
const NAME_PATTERN = /^[\w.-]+$/;
const NUMBER_PATTERN = /^\d+$/;
const NUMBERS_LIST_PATTERN = /^\d+(,\d+)*$/;

export function isValidParam(value: string, type: ParamType): boolean {
  if (typeof type === "object") return type.oneOf.includes(value);

  switch (type) {
    case "number": {
      if (!NUMBER_PATTERN.test(value)) return false;
      const parsed = Number(value);
      return Number.isSafeInteger(parsed);
    }
    case "uuid":
      return UUID_PATTERN.test(value);
    case "numbersList":
      return NUMBERS_LIST_PATTERN.test(value);
    case "name":
      return NAME_PATTERN.test(value);
  }
}

export type ParsedParams = Record<string, string | number | number[]>;

export function parseParam(value: string, type: ParamType): string | number | number[] {
  if (typeof type === "object") return value;

  switch (type) {
    case "number":
      return Number(value);
    case "numbersList":
      return value.split(",").map((part) => Number(part));
    case "uuid":
    case "name":
      return value;
  }
}

export function stringifyParam(value: string | number | number[], type: ParamType): string {
  if (typeof type === "string" && type === "numbersList") {
    return Array.isArray(value) ? value.join(",") : String(value);
  }
  return String(value);
}

export function areParamsValid(
  params: Record<string, string | undefined>,
  types: ParamsDeclaration,
): boolean {
  return Object.entries(types).every(([key, type]) => {
    const value = params[key];
    if (value === undefined) return false;
    return isValidParam(value, type);
  });
}

export function parseParams(
  params: Record<string, string | undefined>,
  types: ParamsDeclaration,
): ParsedParams {
  const parsed: ParsedParams = {};
  for (const [key, type] of Object.entries(types)) {
    const value = params[key];
    if (value === undefined) continue;
    parsed[key] = parseParam(value, type);
  }
  return parsed;
}
```

A missing value is **invalid** (there is no optional-segment support — D14). There is no free
`string` type: every param is one of the five (H1).

**Accept.** `grep -c "export function" src/lib/router/params.ts` → `6`

const VALID_UUID = "3f2504e0-4f89-41d3-9a0c-0305e82c3301";

  it("accepts only listed members of a oneOf declaration", () => {

    expect(isValidParam("assets", type)).toBe(true);
    expect(isValidParam("users", type)).toBe(false);
  });

describe("parseParam and stringifyParam", () => {
  it("round-trips a number", () => {
    expect(stringifyParam(parseParam("42", "number"), "number")).toBe("42");
  });

  it("round-trips a numbersList", () => {
    expect(parseParam("1,2,3", "numbersList")).toEqual([1, 2, 3]);
    expect(stringifyParam([1, 2, 3], "numbersList")).toBe("1,2,3");
  });

  it("round-trips a uuid unchanged", () => {
    expect(stringifyParam(parseParam(VALID_UUID, "uuid"), "uuid")).toBe(VALID_UUID);
  });

  it("parses a oneOf member as the string itself", () => {
    expect(parseParam("assets", { oneOf: ["assets"] })).toBe("assets");
  });
});

describe("areParamsValid", () => {
  it("is true when every declared param is present and valid", () => {
    expect(areParamsValid({ id: VALID_UUID }, { id: "uuid" })).toBe(true);
  });

  it("is false when a declared param is absent", () => {
    expect(areParamsValid({}, { id: "uuid" })).toBe(false);
  });

  it("is false when any one param is invalid", () => {
    expect(areParamsValid({ id: VALID_UUID, page: "x" }, { id: "uuid", page: "number" })).toBe(
      false,
    );
  });

  it("is true for an empty declaration", () => {
    expect(areParamsValid({}, {})).toBe(true);
  });
});

describe("parseParams", () => {
  it("returns only the declared params, parsed", () => {
    expect(parseParams({ id: VALID_UUID, extra: "ignored" }, { id: "uuid" })).toEqual({
      id: VALID_UUID,
    });
  });

  it("omits an absent param rather than inventing one", () => {
    expect(parseParams({}, { id: "uuid" })).toEqual({});
  });
});

### Step 2.3 — The access layer

**Touch.** `src/lib/router/access.ts` (create)

**Do.**

Replace the entire file with exactly this:

```ts
import type { SessionState } from "./session";
import type { ParamsDeclaration } from "./params";
import type { UserRole } from "@/lib/auth";

export interface AccessSources {
  isAuthenticated: boolean;
  role: UserRole | null;
  isAdmin: boolean;
  isHost: boolean;
  userId: string | null;
}

export function selectSources(state: SessionState): AccessSources {
  const role = state.user?.role ?? null;

  return {
    isAuthenticated: state.user !== null,
    role,
    isAdmin: role === "admin",
    isHost: role === "host",
    userId: state.user?.id ?? null,
  };
}

export const ACCESS_RULES = {
  public: () => true,
  canUseApp: (sources: AccessSources) => sources.isAuthenticated,
  canOpenAdmin: (sources: AccessSources) =>
    sources.isAuthenticated && sources.isAdmin,
  canOpenHost: (sources: AccessSources) =>
    sources.isAuthenticated && sources.isHost,
} as const;

export type RuleName = keyof typeof ACCESS_RULES;

export interface PathRequirement {
  pattern: string;
  requires: readonly RuleName[];
  params: ParamsDeclaration;
}

export interface AccessLayer {
  isAllowed(rule: RuleName): boolean;
  canOpenPath(path: PathRequirement): boolean;
  getResults(): Record<RuleName, boolean>;
  getSources(): AccessSources;
  subscribe(listener: () => void): () => void;
  syncFromState(): boolean;
}

export function createAccess(getState: () => SessionState): AccessLayer {
  let state = getState();
  let sources = selectSources(state);
  let results = evaluate(sources);
  const listeners = new Set<() => void>();

  function evaluate(nextSources: AccessSources): Record<RuleName, boolean> {
    const evaluated = {} as Record<RuleName, boolean>;
    for (const name of Object.keys(ACCESS_RULES) as RuleName[]) {
      evaluated[name] = ACCESS_RULES[name](nextSources);
    }
    return evaluated;
  }

  // Re-reads the session and recomputes every result. Returns true only when at
  // least one boolean flipped — the router revalidates on that, never on
  // object identity (R13).
  function syncFromState(): boolean {
    const nextState = getState();
    if (nextState === state) return false;

    state = nextState;
    sources = selectSources(nextState);
    const nextResults = evaluate(sources);
    const flipped = (Object.keys(nextResults) as RuleName[]).some(
      (name) => nextResults[name] !== results[name],
    );
    results = nextResults;

    if (flipped) {
      for (const listener of [...listeners]) listener();
    }

    return flipped;
  }

  return {
    isAllowed: (rule) => results[rule],
    canOpenPath: (path) => path.requires.every((rule) => results[rule]),
    getResults: () => results,
    getSources: () => sources,
    subscribe(listener: () => void) {
      listeners.add(listener);
      return () => {
        listeners.delete(listener);
      };
    },
    syncFromState,
  };
}
```

`{ oneOf }`-style type-widening on `ACCESS_RULES` is avoided by the `as const`, so `RuleName` is the
literal union `"public" | "canUseApp" | "canOpenAdmin" | "canOpenHost"` and a typo in a `requires`
array is a compile error.

**Accept.** `grep -n "syncFromState" src/lib/router/access.ts` → `4` matches (interface member,
`function syncFromState`, the call inside it is not a match, and the `syncFromState,` shorthand in
the returned object), and `grep -n "canOpenHost: (sources" src/lib/router/access.ts` → `1` match.

**Verify.** `npx tsc -b` — exit 0.

### Step 2.4 — Access tests

**Touch.** `src/lib/router/access.test.ts` (create)

```ts
import { describe, expect, it } from "vitest";
import { ACCESS_RULES, createAccess, selectSources } from "./access";
import type { RuleName } from "./access";
import type { SessionState } from "./session";
import type { AuthenticatedUser, UserRole } from "@/lib/auth";

function stateFor(role: UserRole | null): SessionState {
  const user: AuthenticatedUser | null = role
    ? {
        id: `user-${role}`,
        email: `${role}@covera.test`,
        role,
        firstName: role,
        lastName: "User",
      }
    : null;

  return { status: "ready", user, role };
}

const ANONYMOUS = stateFor(null);

describe("rule registry", () => {
  it("gives every rule a unique name", () => {
    const names = Object.keys(ACCESS_RULES);
    expect(new Set(names).size).toBe(names.length);
  });

  it("names every rule can… or public", () => {
    for (const name of Object.keys(ACCESS_RULES)) {
      expect(name === "public" || name.startsWith("can")).toBe(true);
    }
  });
});

describe("rule tables", () => {
  it.each([
    ["public", null, true],
    ["public", "guest", true],
    ["public", "host", true],
    ["public", "admin", true],
    ["canUseApp", null, false],
    ["canUseApp", "guest", true],
    ["canUseApp", "host", true],
    ["canUseApp", "admin", true],
    ["canOpenHost", null, false],
    ["canOpenHost", "guest", false],
    ["canOpenHost", "host", true],
    ["canOpenHost", "admin", false],
    ["canOpenAdmin", null, false],
    ["canOpenAdmin", "guest", false],
    ["canOpenAdmin", "host", false],
    ["canOpenAdmin", "admin", true],
  ] as const)("%s with role %s → %s", (rule, role, expected) => {
    const sources = selectSources(stateFor(role));

    expect(ACCESS_RULES[rule as RuleName](sources)).toBe(expected);
  });
});

describe("selectSources", () => {
  it("maps an absent user to false and null, never to a default role", () => {
    expect(selectSources(ANONYMOUS)).toEqual({
      isAuthenticated: false,
      role: null,
      isAdmin: false,
      isHost: false,
      userId: null,
    });
  });
});

describe("createAccess", () => {
  it("answers synchronously from the snapshot it was built with", () => {
    const access = createAccess(() => stateFor("host"));

    expect(access.isAllowed("canOpenHost")).toBe(true);
    expect(access.isAllowed("canOpenAdmin")).toBe(false);
  });

  it("returns the same results object while the session is unchanged", () => {
    const access = createAccess(() => stateFor("host"));

    expect(access.getResults()).toBe(access.getResults());
  });

  it("requires every inherited rule for canOpenPath", () => {
    const access = createAccess(() => stateFor("host"));

    expect(
      access.canOpenPath({
        pattern: "/host/apartments/:id",
        requires: ["public", "canOpenHost"],
        params: { id: "uuid" },
      }),
    ).toBe(true);

    expect(
      access.canOpenPath({
        pattern: "/admin/dashboard",
        requires: ["public", "canOpenAdmin"],
        params: {},
      }),
    ).toBe(false);
  });

  it("reports a flip only when a result actually changes", () => {
    let state = stateFor("host");
    const access = createAccess(() => state);

    expect(access.syncFromState()).toBe(false);

    state = { ...state };
    expect(access.syncFromState()).toBe(false);

    state = stateFor("admin");
    expect(access.syncFromState()).toBe(true);
    expect(access.isAllowed("canOpenAdmin")).toBe(true);
    expect(access.isAllowed("canOpenHost")).toBe(false);

    expect(access.syncFromState()).toBe(false);
  });

  it("notifies subscribers only on a flip", () => {
    let state = stateFor("host");
    const access = createAccess(() => state);
    let notifications = 0;

    access.subscribe(() => {
      notifications += 1;
    });

    state = { ...state };
    access.syncFromState();
    expect(notifications).toBe(0);

    state = stateFor("admin");
    access.syncFromState();
    expect(notifications).toBe(1);
  });
});
```
