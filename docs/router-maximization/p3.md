# Phase 3 — router core and the public tree

**Prerequisite:** phase 2 (`params.ts`, `access.ts`).
**Gate:** `npm run gate` — must be green before step 3.1.

**Why this phase is third:** playbook stage 2's core (R1, R3–R6, §3). It is split from phase 4 as
D19: the builder plus the public tree is one coherent, reviewable change; the two guarded sections
are another.

**Observable gain.** `/inspection/abc` becomes a **404 before the chunk downloads**. It needs no
session, so this gain is provable in this phase by anyone, signed in or not, and it is the first of
the feature's two "wasted bytes" fixes (F4). The router is also built by a factory from data, so
every later phase can test the real tree.

**What this phase deliberately does NOT do:** it does not delete `requireRoleGuard` and does not
touch `/host` or `/admin`. Those are phase 4. Until then, `src/lib/router/routes/index.ts` describes
the public tree, and those two sections are described by phase 4.

```ts
/**
 * The `:param` names in a route pattern, with optional segments (`:x?`) stripped,
 * so `checkParamKeys` can compare them against a `params` declaration.
 */
export type PathParam<P extends string> = P extends `${string}:${infer Rest}`
  ? Rest extends `${infer Name}?/${infer Tail}`
    ? Name | PathParam<`/${Tail}`>
    : Rest extends `${infer Name}?`
      ? Name
      : Rest extends `${infer Name}/${infer Tail}`
        ? Name | PathParam<`/${Tail}`>
        : Rest
  : never;
```

```ts
import { generatePath } from "react-router";
import type { PathParam, ParamsDeclaration, ParamType } from "./params";
import { stringifyParam } from "./params";
import type { RuleName } from "./access";

export interface PathDefinition<
  P extends string = string,
  R extends readonly RuleName[] = readonly RuleName[],
> {
  pattern: P;
  requires: R;
  params: ParamsDeclaration;
}

/** Compile-time guard: the declared params must be exactly the pattern's params. */
type ParamKeysMatch<
  P extends string,
  Decl extends ParamsDeclaration,
> = PathParam<P> extends never
  ? keyof Decl extends never
    ? true
    : { error: "params declared for a pattern with no :params" }
  : Exclude<PathParam<P>, keyof Decl> extends never
    ? Exclude<keyof Decl, PathParam<P>> extends never
      ? true
      : { error: "params has a key the pattern does not contain" }
    : { error: "a :param in the pattern has no declared type" };

export function definePaths<
  const M extends Record<string, { pattern: string; requires: readonly RuleName[]; params: ParamsDeclaration }>,
>(
  map: M & { [K in keyof M]: ParamKeysMatch<M[K]["pattern"], M[K]["params"]> extends true ? unknown : ParamKeysMatch<M[K]["pattern"], M[K]["params"]> },
): M {
  return map;
}

export type ParamsFor<D extends PathDefinition> = {
  [K in keyof D["params"]]: D["params"][K] extends "number" | "numbersList"
    ? number | number[]
    : string;
};

/**
 * The only href builder in the app. `params` is required exactly when the
 * pattern has `:params`; a pattern without params takes no second argument.
 */
export function buildPath<D extends PathDefinition>(
  definition: D,
  ...args: keyof D["params"] extends never ? [] : [params: ParamsFor<D>]
): string {
  const params = (args[0] ?? {}) as Record<string, string | number | number[]>;
  const typed: Record<string, string> = {};

  for (const [key, type] of Object.entries(definition.params) as [string, ParamType][]) {
    const value = params[key];
    typed[key] = stringifyParam(value, type);
  }

  return generatePath(definition.pattern, typed);
}
```

`PathDefinition`'s default type arguments let `definePaths` infer the literal pattern, so
`buildPath(hostPaths.apartment, { id: "abc" })` is a compile error when `apartment.pattern` is
`/host/apartments/:id` and `id` is declared `uuid`… but only for *keys*, not for *validity*: validity
is runtime (`isValidParam`), by design (H2 says `definePaths` fails to compile on a key mismatch, not
on a bad value).

### Step 3.2 — Redirect specs

```ts
import { redirect } from "react-router";
import type { ParsedParams } from "./params";
import type { RuleName } from "./access";

export type RedirectTarget = string | ((params: ParsedParams) => string);

export interface RedirectSpec {
  to: RedirectTarget;
  search?: Record<string, string> | ((params: ParsedParams) => Record<string, string>);
}

export function redirectTo(
  to: RedirectTarget,
  options: { search?: RedirectSpec["search"] } = {},
): RedirectSpec {
  return { to, search: options.search };
}

export function resolveRedirect(
  spec: RedirectSpec,
  params: ParsedParams,
): { pathname: string; search: string } {
  const pathname = typeof spec.to === "function" ? spec.to(params) : spec.to;
  const searchRecord =
    typeof spec.search === "function" ? spec.search(params) : spec.search;
  const searchParams = new URLSearchParams(searchRecord ?? {});

  return { pathname, search: searchParams.toString() ? `?${searchParams.toString()}` : "" };
}

export interface FirstAllowedTarget {
  rule: RuleName;
  to: string;
}

/** The first target whose rule passes; `null` when none does (the caller throws 403). */
export function firstAllowed(
  targets: readonly FirstAllowedTarget[],
  isAllowed: (rule: RuleName) => boolean,
): string | null {
  for (const target of targets) {
    if (isAllowed(target.rule)) return target.to;
  }
  return null;
}

export function redirectResponse(spec: RedirectSpec, params: ParsedParams): Response {
  const { pathname, search } = resolveRedirect(spec, params);
  return redirect(`${pathname}${search}`);
}
```

`resolveRedirect` drops the incoming search and hash by construction: it builds a fresh
`URLSearchParams` from the spec only (H6).

### Step 3.3 — `defineRoutes`, the route builder

**Touch.** `src/lib/router/defineRoutes.tsx` (create)

**Do.** Replace the entire file with:

```tsx
/* eslint-disable @typescript-eslint/no-explicit-any */
import { Outlet, data } from "react-router";
import type { RouteObject } from "react-router";
import type { ComponentType } from "react";
import { createElement, lazy, Suspense } from "react";

import { areParamsValid, parseParams } from "./params";
import type { ParamsDeclaration, ParsedParams } from "./params";
import type { RuleName } from "./access";
import { redirectResponse } from "./redirects";
import type { RedirectSpec } from "./redirects";

export type Crumb = string;

export interface GuardMeta {
  pattern: string;
  rules: readonly RuleName[];
  params: ParamsDeclaration;
  redirect?: RedirectSpec;
  preload?: (params: ParsedParams) => void;
}

export interface Handle {
  guard: GuardMeta;
  crumb?: Crumb;
}

export interface RouteBase {
  requires?: RuleName;
  params?: ParamsDeclaration;
  variants?: readonly string[];
  handle?: { crumb?: Crumb };
}

export type PageModule = { default: ComponentType<any> };

export interface PageRoute extends RouteBase {
  path?: string;
  index?: boolean;
  page: () => Promise<PageModule>;
  prefetch?: (params: ParsedParams) => void;
  recordCheck?: (params: ParsedParams, options: { signal: AbortSignal }) => Promise<void>;
  children?: RouteNode[];
}

export interface LayoutRoute extends RouteBase {
  path?: string;
  layout: () => Promise<PageModule>;
  prefetch?: (params: ParsedParams) => void;
  children: RouteNode[];
}

export interface GroupRoute extends RouteBase {
  path: string;
  children: RouteNode[];
}

export interface RedirectRoute extends RouteBase {
  index: true;
  loader: RedirectSpec;
}

export type RouteNode = PageRoute | LayoutRoute | GroupRoute | RedirectRoute;

export type SectionRoute = (PageRoute | LayoutRoute | GroupRoute) & {
  path: string;
  requires: RuleName;
};

export interface DefineRoutesDeps {
  isAllowed(rule: RuleName): boolean;
  errorElement: React.ReactNode;
  /** Rendered by `<Suspense>` while a route's module resolves. */
  pendingElement?: React.ReactNode;
  variant?: string;
  parent?: ParentContext;
}

interface ParentContext {
  pattern: string;
  rules: readonly RuleName[];
  params: ParamsDeclaration;
  id: string;
}

const EMPTY_PARENT: ParentContext = {
  pattern: "",
  rules: [],
  params: {},
  id: "root",
};

function joinPattern(prefix: string, path: string | undefined): string {
  const segments = [prefix.replace(/\/$/, ""), (path ?? "").replace(/^\//, "")].filter(Boolean);
  const joined = segments.join("/");
  return joined.startsWith("/") ? joined : `/${joined}`;
}

function indexId(pattern: string): string {
  return pattern === "/" ? "/#index" : `${pattern}#index`;
}

function routeId(pattern: string, suffix: string): string {
  return `${pattern}#${suffix}`;
}

export function defineRoutes(routes: readonly RouteNode[], deps: DefineRoutesDeps): RouteObject[] {
  return buildLevel(routes, deps.parent ?? EMPTY_PARENT, deps);
}

function buildLevel(
  routes: readonly RouteNode[],
  parent: ParentContext,
  deps: DefineRoutesDeps,
): RouteObject[] {
  const built: RouteObject[] = [];
  const seenIds = new Set<string>();

  for (const node of routes) {
    if (deps.variant && node.variants && !node.variants.includes(deps.variant)) continue;

    const pattern = joinPattern(parent.pattern, (node as any).path);
    const rules: readonly RuleName[] = node.requires
      ? [...parent.rules, node.requires]
      : parent.rules;
    const params: ParamsDeclaration = { ...parent.params, ...(node.params ?? {}) };

    assertParamsDeclared(pattern, params);

    const children = (node as any).children as RouteNode[] | undefined;
    const isRedirect = (node as any).index === true && (node as any).loader !== undefined;
    const isLayout = (node as any).layout !== undefined;
    const isGroup = !isLayout && !isRedirect && (node as any).page === undefined;

    const id = isRedirect
      ? indexId(parent.pattern)
      : routeId(pattern, isLayout ? "layout" : isGroup ? "group" : "page");

    if (seenIds.has(id)) {
      throw new Error(`[router] duplicate route id "${id}" among siblings of "${parent.pattern}".`);
    }
    seenIds.add(id);

    const guard: GuardMeta = {
      pattern,
      rules,
      params,
      redirect: isRedirect ? ((node as any).loader as RedirectSpec) : undefined,
      preload: buildPreload(node, deps),
    };

    const handle: Handle = { guard, crumb: node.handle?.crumb };

    if (isRedirect) {
      const spec = (node as any).loader as RedirectSpec;
      built.push({
        id,
        index: true,
        handle,
        loader: () => {
          for (const rule of rules) {
            if (!deps.isAllowed(rule)) throw data(null, { status: 403 });
          }
          throw redirectResponse(spec, {});
        },
      });
      continue;
    }

    const childParent: ParentContext = { pattern, rules, params, id };
    const childRoutes = children ? buildLevel(children, childParent, deps) : [];

    if (isLayout) {
      built.push({
        id,
        path: (node as any).path,
        handle,
        loader: makeLoader(node, guard, deps),
        Component: makeComponent(node.layout, deps),
        children: withBoundary(childRoutes, routeId(pattern, "boundary"), deps),
      } as RouteObject);
      continue;
    }

    if (isGroup) {
      const hasGuardWork = Boolean(node.requires) || Boolean((node as any).params);
      built.push({
        id,
        path: (node as any).path,
        handle,
        ...(hasGuardWork ? { loader: makeLoader(node, guard, deps) } : {}),
        children: childRoutes,
      } as RouteObject);
      continue;
    }

    built.push({
      id,
      ...((node as any).index === true ? { index: true } : { path: (node as any).path }),
      handle,
      loader: makeLoader(node, guard, deps),
      Component: makeComponent((node as any).page, deps),
      children: childRoutes.length ? withBoundary(childRoutes, routeId(pattern, "boundary"), deps) : undefined,
    } as RouteObject);
  }

  return built;
}

function withBoundary(
  children: RouteObject[],
  id: string,
  deps: DefineRoutesDeps,
): RouteObject[] {
  return [{ id, errorElement: deps.errorElement, children }];
}

function assertParamsDeclared(pattern: string, params: ParamsDeclaration): void {
  const names = pattern
    .split("/")
    .filter((segment) => segment.startsWith(":"))
    .map((segment) => segment.slice(1));

  for (const name of names) {
    if (!(name in params)) {
      throw new Error(
        `[router] ${pattern}: ":${name}" has no declared param type. Add ` +
          `params: { ${name}: "uuid" } to the route (R4).`,
      );
    }
  }
}

function buildPreload(
  node: RouteNode,
  deps: DefineRoutesDeps,
): ((params: ParsedParams) => void) | undefined {
  const loadModule = (node as any).page ?? (node as any).layout;
  if (!loadModule) return undefined;

  const prefetch = (node as any).prefetch as ((params: ParsedParams) => void) | undefined;

  return (params: ParsedParams) => {
    if (!guardPasses(node, deps, params)) return;
    prefetch?.(params);
    void loadModule().catch(() => undefined);
  };
}

function guardPasses(node: RouteNode, deps: DefineRoutesDeps, params: ParsedParams): boolean {
  // The preloader (phase 6) already ran the rule and param checks for the whole
  // matched chain; this is the per-route re-check so preload can never start
  // work for a URL the user cannot open.
  if (node.requires && !deps.isAllowed(node.requires)) return false;
  const declared = node.params ?? {};
  return areParamsValid(
    Object.fromEntries(
      Object.keys(declared).map((key) => [key, params[key] === undefined ? undefined : String(params[key])]),
    ),
    declared,
  );
}

function makeLoader(
  node: RouteNode,
  guard: GuardMeta,
  deps: DefineRoutesDeps,
): (args: { params: Record<string, string | undefined>; request: Request }) => Promise<null> {
  const loadModule = ((node as any).page ?? (node as any).layout) as
    | (() => Promise<PageModule>)
    | undefined;
  const prefetch = (node as any).prefetch as ((params: ParsedParams) => void) | undefined;
  const recordCheck = (node as any).recordCheck as
    | ((params: ParsedParams, options: { signal: AbortSignal }) => Promise<void>)
    | undefined;

  return async ({ params, request }) => {
    for (const rule of guard.rules) {
      if (!deps.isAllowed(rule)) throw data(null, { status: 403 });
    }

    if (!areParamsValid(params, guard.params)) throw data(null, { status: 404 });

    const parsed = parseParams(params, guard.params);

    prefetch?.(parsed);

    await Promise.all([
      loadModule ? loadModule() : Promise.resolve(),
      recordCheck?.(parsed, { signal: request.signal }),
    ]);

    return null;
  };
}

function makeComponent(
  loadModule: (() => Promise<PageModule>) | undefined,
  deps: DefineRoutesDeps,
): ComponentType | undefined {
  if (!loadModule) return undefined;

  const LazyComponent = lazy(async () => {
    const module = await loadModule();
    return { default: module.default };
  });

  return function RoutedComponent(props: Record<string, unknown>) {
    return createElement(
      Suspense,
      { fallback: deps.pendingElement ?? null },
      createElement(LazyComponent, props),
    );
  };
}
```

The builder never copies `loader`, `Component`, `element`, `lazy` or `ErrorBoundary` from the input
node — it only reads `path`, `index`, `requires`, `params`, `variants`, `handle`, `page`/`layout`,
`prefetch`, `recordCheck` and `children`. That is what makes a hand-written `loader` impossible rather
than merely forbidden (R1), and it is why `noUnusedLocals` stays satisfied without a guard helper.

**Accept.** `npm run gate` → exit 0. Plus `grep -c "export function defineRoutes" src/lib/router/defineRoutes.tsx` → `1` and `grep -c "writeGuard" src/lib/router/defineRoutes.tsx` → `0`.

**Verify.** `npm run gate`

```tsx
import { createBrowserRouter, createMemoryRouter } from "react-router";
import type { DataRouter, InitialEntry, RouteObject } from "react-router";

import { defineRoutes } from "./defineRoutes";
import type { DefineRoutesDeps, RouteNode } from "./defineRoutes";
import type { AccessLayer } from "./access";
import type { SessionStore } from "./session";

export interface CreateAppRouterOptions {
  session: SessionStore;
  access: AccessLayer;
  errorElement: React.ReactNode;
  pendingElement?: React.ReactNode;
  routes: readonly RouteNode[];
  basename?: string;
  variant?: string;
  /** Present → a memory router, for tests. Absent → a browser router. */
  initialEntries?: InitialEntry[];
  /** Extra route objects prepended to the built tree (public routes, boundaries). */
  prepend?: RouteObject[];
  /** Wraps the built tree (the shell). Receives the built children. */
  wrap?: (children: RouteObject[]) => RouteObject[];
}

export interface AppRouter {
  router: DataRouter;
  access: AccessLayer;
  dispose(): void;
}

export function createAppRouter({
  session,
  access,
  errorElement,
  pendingElement,
  routes,
  basename,
  variant,
  initialEntries,
  prepend,
  wrap,
}: CreateAppRouterOptions): AppRouter {
  let previousResults = access.getResults();

  const deps: DefineRoutesDeps = {
    isAllowed: (rule) => access.isAllowed(rule),
    errorElement,
    pendingElement,
    variant,
  };

  const built = defineRoutes(routes, deps);
  const children = wrap ? wrap(built) : built;
  const tree = [...(prepend ?? []), ...children];

  const router = initialEntries
    ? createMemoryRouter(tree, { initialEntries, basename })
    : createBrowserRouter(tree, { basename });

  // R13: revalidate only when an access answer flips — never on session
  // identity. A denied page becomes reachable (or the reverse) without a reload.
  const unsubscribeSession = session.subscribe(() => {
    const flipped = access.syncFromState();
    if (flipped) void router.revalidate();
    previousResults = access.getResults();
  });

  return {
    router,
    access,
    dispose() {
      unsubscribeSession();
      void previousResults;
    },
  };
}
```

### Step 3.10 — The navigation-timing instrument

**Do.** Part (a): create `src/lib/router/navigationTiming.ts`:

```ts
import type { DataRouter } from "react-router";

export interface NavigationTiming {
  /** Where the navigation started from. */
  from: string;
  /** Where it went. */
  to: string;
  /** Wall-clock ms from the router entering "loading" to being back at "idle". */
  durationMs: number;
}

/**
 * Records route-transition durations so the NT p75 in
 * docs/routing-performance-baseline.md can be measured without a profiler
 * (playbook §0, R33). Kept deliberately tiny: a subscribe callback and a ring
 * buffer, no timers, no DOM.
 */
export interface NavigationTimingRecorder {
  install(router: DataRouter): () => void;
  /** Oldest first, capped at `CAPACITY`. */
  getTimings(): readonly NavigationTiming[];
  /** The middle value, or null with no samples — the "p75" of 5 runs is the 3rd. */
  getMedian(): number | null;
  clear(): void;
}

const CAPACITY = 50;

export function createNavigationTiming(): NavigationTimingRecorder {
  const timings: NavigationTiming[] = [];
  let startedAt: number | null = null;

  return {
    install(router) {
      return router.subscribe((state) => {
        if (state.navigation.state === "loading") {
          startedAt ??= performance.now();
          return;
        }

        // Back to idle: close the sample. `startedAt` is null for the initial
        // render, so no bogus sample is recorded for it.
        if (state.navigation.state === "idle" && startedAt !== null) {
          timings.push({
            from: state.historyAction,
            to: state.location.pathname,
            durationMs: performance.now() - startedAt,
          });

          if (timings.length > CAPACITY) timings.shift();
          startedAt = null;
        }
      });
    },

    getTimings: () => timings,

    getMedian() {
      if (timings.length === 0) return null;
      const sorted = [...timings].map((entry) => entry.durationMs).sort((a, b) => a - b);
      return sorted[Math.floor(sorted.length / 2)];
    },

    clear() {
      timings.length = 0;
      startedAt = null;
    },
  };
}
```
