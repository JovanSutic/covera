# Phase 5 — shell: errors, pending, scroll, a11y

**Prerequisite:** phase 4 (host and admin sections in the tree).
**Gate:** `npm run gate` — must be green before step 5.1.

**Why this phase is fifth:** playbook stage 3 (R7–R10, R33). It is what makes the layout survive an
error and stops the screen being blank during a wait — both of which only have meaning once every
in-shell route exists.

**Observable gain.** Four:

1. An error in one page keeps the header and the frame (today it replaces the entire app).
2. An unknown URL is a **404 inside the shell**, not a dead-end full-page error.
3. A slow navigation **keeps the old page** and shows a progress bar after 200 ms, instead of a blank
   wait.
4. The header renders **once**, from the shell, instead of once per page — and focus plus the document
   title update on every page change.

**This phase is test-exempt (D44).** It is markup, two error components and three hooks whose
behaviour is DOM timing; a jsdom render of a progress bar proves nothing about the 200 ms threshold.
The numbered browser checklist in step 5.7 **is** the verification, and it is the phase's acceptance.

**Four new translation keys.** They are given verbatim in step 5.3 for all four locales.

```tsx
import { useEffect, useRef, useState } from "react";
import { useLocation } from "react-router";

/**
 * R33/H17: on a pathname change — not the first render, not a search-only
 * change — focus the main region without scrolling and announce the new title.
 */
export function RouteAnnouncer() {
  const location = useLocation();
  const [message, setMessage] = useState("");
  const previousPathname = useRef<string | null>(null);
  const hasRendered = useRef(false);

  useEffect(() => {
    if (!hasRendered.current) {
      hasRendered.current = true;
      previousPathname.current = location.pathname;
      return;
    }

    if (previousPathname.current === location.pathname) return;
    previousPathname.current = location.pathname;

    const title = document.title;

    const main = document.querySelector("main");
    if (main) {
      if (!main.hasAttribute("tabindex")) main.setAttribute("tabindex", "-1");
      main.focus({ preventScroll: true });
    }

    setMessage(title);
  }, [location.pathname]);

  return (
    <div aria-live="polite" aria-atomic="true" className="sr-only">
      {message}
    </div>
  );
}
```

```tsx
import { useEffect, useState } from "react";
import { cn } from "@/lib/utils";

const VISIBLE_AFTER_MS = 200;

/**
 * A navigation that finishes under 200 ms must not flash a bar (R9). The bar
 * appears only once a wait is long enough to be worth acknowledging.
 */
export function ProgressBar({ isNavigating }: { isNavigating: boolean }) {
  const [isVisible, setIsVisible] = useState(false);

  useEffect(() => {
    if (!isNavigating) {
      setIsVisible(false);
      return;
    }

    const timer = window.setTimeout(() => setIsVisible(true), VISIBLE_AFTER_MS);
    return () => window.clearTimeout(timer);
  }, [isNavigating]);

  return (
    <div
      aria-hidden="true"
      className={cn(
        "fixed inset-x-0 top-0 z-50 h-0.5 bg-blue-600 transition-opacity duration-150",
        isVisible ? "opacity-100" : "opacity-0",
      )}
    />
  );
}
```

### Step 5.2 — The route error element and stale chunks

**Touch.** `src/components/router/RouteError.tsx` (create), `src/pages/ErrorPage.tsx` (edit)

**Do.** Part (a): create `src/components/router/RouteError.tsx`:

```tsx
import { isRouteErrorResponse, useRouteError } from "react-router";
import { useTranslation } from "react-i18next";
import PageLayout from "@/components/layout/PageLayout";
import PageTitle from "@/components/PageTitle";

export const CHUNK_RELOAD_KEY = "covera:chunk-reload";

function isChunkLoadError(error: unknown): boolean {
  if (!(error instanceof Error)) return false;

  return (
    error.name === "ChunkLoadError" ||
    /dynamically imported module|Importing a module script failed|Loading chunk .* failed/i.test(
      error.message,
    )
  );
}

/**
 * Reloads once per session for a stale chunk (R8). A storage failure means
 * "do not reload" — a browser that blocks sessionStorage must never loop.
 */
export function reloadOnceForStaleChunk(): boolean {
  try {
    if (sessionStorage.getItem(CHUNK_RELOAD_KEY) === "1") return false;
    sessionStorage.setItem(CHUNK_RELOAD_KEY, "1");
    window.location.reload();
    return true;
  } catch {
    return false;
  }
}

/** Cleared by the router factory after the next successful idle navigation. */
export function clearStaleChunkFlag(): void {
  try {
    sessionStorage.removeItem(CHUNK_RELOAD_KEY);
  } catch {
    // Storage unavailable: nothing to clear, and nothing to leak.
  }
}

export function RouteError() {
  const { t } = useTranslation("general");
  const error = useRouteError();

  if (isRouteErrorResponse(error) && error.status === 403) {
    return (
      <PageLayout size="lg">
        <PageTitle
          title={t("errorPage.forbiddenTitle")}
          subtitle={t("errorPage.forbiddenSubtitle")}
        />
      </PageLayout>
    );
  }

  if (isRouteErrorResponse(error) && error.status === 404) {
    return (
      <PageLayout size="lg">
        <PageTitle
          title={t("errorPage.notFoundTitle")}
          subtitle={t("errorPage.notFoundSubtitle")}
        />
      </PageLayout>
    );
  }

  if (isChunkLoadError(error) && reloadOnceForStaleChunk()) {
    // The reload is in flight; render nothing so the user does not see a flash.
    return null;
  }

  return (
    <PageLayout size="lg">
      <PageTitle
        title={t("errorPage.unexpectedTitle")}
        subtitle={error instanceof Error ? error.message : t("errorPage.unexpectedSubtitle")}
      />
    </PageLayout>
  );
}
```
