# Phase 6 — links, preload, prefetch

**Prerequisite:** phase 5 (the shell is a route; the factory builds inside and outside trees).
**Gate:** `npm run gate` — must be green before step 6.1.

**Why this phase is sixth:** playbook stages 4 and 5 (R11, R12, R16, R20, §4). It is the first phase
users clearly *feel*, and it is where the apartment query stops waiting for the chunk.

**Observable gain.** Three:

1. **Hovering a list/menu link for 100 ms downloads its chunk**, so the click shows no progress bar.
2. **The apartment detail's main API request starts with its chunk**, not after it — the waterfall
   (wait B → wait D) collapses to zero on that page.
3. **The unresolved `<Navigate to="/apartments">` dead end is gone**: a missing apartment id is now a
   404 from the route, not a navigation to a route that does not exist (D21).

### Step 6.1 — The preload registry

**Touch.** `src/lib/router/preloadRegistry.ts` (create)

**Do.**

```ts
export type RoutePreloader = (href: string) => void;

let registered: RoutePreloader | null = null;

/**
 * Registered by the router factory so any link can start a route's work without
 * importing the router module (§3, H7). A newer registration replaces an older
 * one; unregistering never clears a preloader registered after it.
 */
export function registerRoutePreloader(preloader: RoutePreloader): () => void {
  registered = preloader;

  return () => {
    if (registered === preloader) registered = null;
  };
}

export function preloadRoute(href: string): void {
  if (!registered) return;

  try {
    registered(href);
  } catch {
    // A preload is best-effort: the loader retries on navigation (R11).
  }
}
```

### Step 6.2 — The preloader, wired by the factory

**Touch.** `src/lib/router/createAppRouter.tsx` (edit)

**Do.** Add the imports:

```ts
import { matchRoutes } from "react-router";
import { registerRoutePreloader } from "./preloadRegistry";
import { areParamsValid } from "./params";
import type { ParsedParams } from "./params";
import type { GuardMeta } from "./defineRoutes";
```

Then, inside `createAppRouter`, after the router is created and before the `session.subscribe` block,
add:

```ts
  // R11: match the href against the built tree, run the SAME rule and param checks
  // the loader would, and only then start each matched route's imports. A denied or
  // invalid href preloads nothing (principle 2).
  const unregisterPreloader = registerRoutePreloader((href) => {
    const url = new URL(href, window.location.origin);
    const matches = matchRoutes(router.routes, url.pathname, basename);

    if (!matches) return;

    // Pass 1: every matched route's whole rule chain and every present param must
    // pass before ANY preload runs. This is what stops one route's chunk
    // downloading when another route in the same match chain is denied.
    for (const match of matches) {
      const guard = (match.route.handle as { guard?: GuardMeta } | undefined)?.guard;
      if (!guard) continue;

      for (const rule of guard.rules) {
        if (!access.isAllowed(rule)) return;
      }

      const present: Record<string, string | undefined> = {};
      for (const key of Object.keys(guard.params)) present[key] = match.params[key];

      if (!areParamsValid(present, guard.params)) return;
    }

    // Pass 2: the checks passed for the whole chain, so start the work.
    for (const match of matches) {
      const guard = (match.route.handle as { guard?: GuardMeta } | undefined)?.guard;
      if (!guard) continue;

      const present: Record<string, string | undefined> = {};
      for (const key of Object.keys(guard.params)) present[key] = match.params[key];

      guard.preload?.(present as ParsedParams);
    }
  });
```

Finally, add `unregisterPreloader()` to the returned `dispose()`.

### Step 6.4 — `usePreloadRoute`

**Touch.** `src/hooks/usePreloadRoute.ts` (create)

**Do.**

```ts
import { useCallback, useEffect, useMemo, useRef } from "react";
import { preloadRoute } from "@/lib/router/preloadRegistry";

const INTENT_DELAY_MS = 100;

export interface PreloadHandlers {
  onMouseEnter: () => void;
  onMouseLeave: () => void;
  onFocus: () => void;
  onBlur: () => void;
  onPointerDown: () => void;
}

/**
 * H8: hover or focus held for 100 ms starts the route's work; a pointer press
 * starts it immediately. Leaving cancels. Timers are cleared on unmount.
 */
export function usePreloadRoute(href: string | null, enabled = true): PreloadHandlers {
  const timer = useRef<number | null>(null);

  const cancel = useCallback(() => {
    if (timer.current !== null) {
      window.clearTimeout(timer.current);
      timer.current = null;
    }
  }, []);

  useEffect(() => cancel, [cancel]);

  const start = useCallback(() => {
    if (!enabled || !href) return;
    cancel();
    timer.current = window.setTimeout(() => {
      timer.current = null;
      preloadRoute(href);
    }, INTENT_DELAY_MS);
  }, [cancel, enabled, href]);

  const startNow = useCallback(() => {
    if (!enabled || !href) return;
    cancel();
    preloadRoute(href);
  }, [cancel, enabled, href]);

  return useMemo(
    () => ({
      onMouseEnter: start,
      onFocus: start,
      onMouseLeave: cancel,
      onBlur: cancel,
      onPointerDown: startNow,
    }),
    [cancel, start, startNow],
  );
}
```

`href` is `string | null` so a denied target (whose href is not built) can pass `null` and get no-ops
from a single hook instance, rather than a conditional hook call.

### Step 6.5 — Timer test for `usePreloadRoute`

**Touch.** `src/hooks/usePreloadRoute.test.ts` (create)

**Do.**

```ts
import { renderHook } from "@testing-library/react";
import { afterEach, beforeEach, describe, expect, it, vi } from "vitest";
import { usePreloadRoute } from "./usePreloadRoute";
import { registerRoutePreloader } from "@/lib/router/preloadRegistry";

describe("usePreloadRoute", () => {
  let preloaded: string[];

  beforeEach(() => {
    vi.useFakeTimers();
    preloaded = [];
    registerRoutePreloader((href) => {
      preloaded.push(href);
    });
  });

  afterEach(() => {
    vi.useRealTimers();
  });

  it("preloads after a 100 ms hover", () => {
    const { result } = renderHook(() => usePreloadRoute("/host/apartments"));

    result.current.onMouseEnter();
    expect(preloaded).toEqual([]);

    vi.advanceTimersByTime(100);
    expect(preloaded).toEqual(["/host/apartments"]);
  });

  it("does not preload when the pointer leaves before 100 ms", () => {
    const { result } = renderHook(() => usePreloadRoute("/host/apartments"));

    result.current.onMouseEnter();
    vi.advanceTimersByTime(50);
    result.current.onMouseLeave();
    vi.advanceTimersByTime(1000);

    expect(preloaded).toEqual([]);
  });

  it("preloads immediately on pointer down, without waiting", () => {
    const { result } = renderHook(() => usePreloadRoute("/host/apartments"));

    result.current.onPointerDown();

    expect(preloaded).toEqual(["/host/apartments"]);
  });

  it("never preloads when disabled", () => {
    const { result } = renderHook(() =>
      usePreloadRoute("/host/apartments", false),
    );

    result.current.onMouseEnter();
    vi.advanceTimersByTime(500);
    result.current.onPointerDown();

    expect(preloaded).toEqual([]);
  });

  it("never preloads a null href", () => {
    const { result } = renderHook(() => usePreloadRoute(null));

    result.current.onMouseEnter();
    result.current.onPointerDown();
    vi.advanceTimersByTime(500);

    expect(preloaded).toEqual([]);
  });

  it("clears a pending timer on unmount", () => {
    const { result, unmount } = renderHook(() =>
      usePreloadRoute("/host/apartments"),
    );

    result.current.onMouseEnter();
    unmount();
    vi.advanceTimersByTime(1000);

    expect(preloaded).toEqual([]);
  });
});
```

`registerRoutePreloader` sets a module-level preloader, so this file's `beforeEach` replaces whatever
an earlier test in the same file registered — which is exactly the "newer registration wins" contract
of step 6.1. Tests in other files are unaffected because each test file gets its own module registry.

### Step 6.6 — The apartment query, in one place

**Touch.** `src/lib/api/queries/apartment.ts` (create)

**Do.**

```ts
import { getApartmentsById } from "@/api/generated/requests/sdk.gen";
import { withAuth } from "@/lib/api/api";
import { QUERY_ACTIONS } from "@/lib/api/queryKeys";
import { useQuery } from "@tanstack/react-query";

/**
 * The apartment detail query, defined ONCE so the route's `prefetch` and the
 * page's `useQuery` cannot drift apart (§4: same key = same args, or the cache
 * sees two entries and fetches twice).
 */
export function apartmentQueryKey(id: string) {
  return [...QUERY_ACTIONS.APARTMENTS_GET_ID, id] as const;
}

export async function fetchApartment(id: string, signal?: AbortSignal) {
  const config = await withAuth(signal ? { signal } : {});
  const response = await getApartmentsById({ ...config, path: { id } });
  return response.data;
}

export function useApartment(id: string | undefined) {
  return useQuery({
    queryKey: apartmentQueryKey(id ?? ""),
    queryFn: async ({ signal }) => {
      if (!id) throw new Error("apartmentIdRequired");
      return fetchApartment(id, signal);
    },
    enabled: Boolean(id),
    // Prefetched queries must be fresh for a while, or the page's hook refetches
    // on mount and the prefetch was for nothing (D37, §4).
    staleTime: 30_000,
  });
}
```

`staleTime: 30_000` here **overrides** `src/main.tsx`'s 15-minute default for this query only — that
is D37, and it is the setting that decides whether a prefetch is reused.

**Accept.** `npm run gate` → exit 0. Plus
`grep -c "QUERY_ACTIONS.APARTMENTS_GET_ID" src/lib/api/queries/apartment.ts` → `1` — the key is

```tsx
import { Link, NavLink } from "react-router";
import type { ReactNode } from "react";

import { useAccessStore } from "@/lib/router/AppContext";
import { buildPath } from "@/lib/router/paths";
import type { PathDefinition } from "@/lib/router/paths";
import { usePreloadRoute } from "@/hooks/usePreloadRoute";

type ParamsArg<D extends PathDefinition> = keyof D["params"] extends never
  ? { params?: undefined }
  : { params: Record<string, string | number | number[]> };

export type AccessLinkProps<D extends PathDefinition> = {
  to: D;
  children: ReactNode;
  className?: string;
  /** Renders a `NavLink`; the callback receives its active state. */
  navClassName?: (state: { isActive: boolean; isPending: boolean }) => string;
} & ParamsArg<D>;

/**
 * R20/H10: the only link that can point at an app path. A viewer who cannot open
 * the target gets plain text, never a link that ends in a 403 (principle 4), and
 * a viewer who can gets intent preloading for free.
 */
export function AccessLink<D extends PathDefinition>({
  to,
  params,
  children,
  className,
  navClassName,
}: AccessLinkProps<D>) {
  const access = useAccessStore();

  const canOpen = access.canOpenPath(to);
  const href = canOpen
    ? buildPath(to, ...((params === undefined ? [] : [params]) as never))
    : null;

  const preloadHandlers = usePreloadRoute(href, canOpen);

  if (!href) {
    return <span className={className}>{children}</span>;
  }

  if (navClassName) {
    return (
      <NavLink
        to={href}
        className={({ isActive, isPending }) => navClassName({ isActive, isPending })}
        {...preloadHandlers}
      >
        {children}
      </NavLink>
    );
  }

  return (
    <Link to={href} className={className} {...preloadHandlers}>
      {children}
    </Link>
  );
}
```
