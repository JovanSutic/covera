# Phase 1 — the shared design system

**Prerequisite:** none
**Gate:** `npm run gate`

This phase is **purely additive**. It creates six files under `src/components/shared/` and
refactors two headers onto `BackLink`. Nothing in it changes the reservation table or the
reservation page, so the phase cannot break them. That is why it lands first: phases 3–6 consume
these primitives, and a phase that both defines a primitive and rebuilds a surface on it cannot
tell you which half broke.

Class strings marked **verbatim** are copied character for character from the file named in the
comment above them (D30). Do not paraphrase them.

### Step 1.1 — `InfoTooltip`

**Touch.** `src/components/shared/InfoTooltip.tsx` (create)

**Do.** The panel's classes are **verbatim** from `DescriptionTooltip` in
`src/components/host/AssetItemCard.tsx`, with two changes: the anchor flips from `right-0` to
`left-0` (the trigger sits LEFT of a label, so a right-anchored panel would cover the label), and
the container is `inline-flex` because it is a flex item.

```tsx
import { Info } from "lucide-react";

interface InfoTooltipProps {
  /** The explanatory text, already translated. */
  content: string;
  /** The trigger's `aria-label`; it is an icon-only button, so it needs one. */
  label: string;
  className?: string;
}

/**
 * An explanatory tooltip whose trigger is a real <button>, so it is reachable by
 * keyboard — the `DescriptionTooltip` version it is modelled on (AssetItemCard) is a
 * <button> that does nothing on click and shows only on `group-hover`, which makes
 * it a dead tab stop.
 *
 * `group-focus-within` alongside `group-hover` is the point: tabbing to the trigger
 * reveals the panel.
 *
 * It is contained (`z-30` inside a `relative` span, no portal) rather than built on
 * `Popover`. `Popover` portals to `document.body` at `z-9999`, which would clash
 * with the `Modal` layer the proof dialog sits in.
 *
 * It renders a SIBLING of the button it explains, never a child: a <button> cannot
 * contain a <button>, and a nested one would also bubble its click into the action.
 *
 * Desktop only by design (D6a) — a surface inside a table cell or a horizontally
 * scrolling strip passes no `tooltip` to its buttons, because an absolutely
 * positioned panel is clipped there.
 */
export function InfoTooltip({ content, label, className }: InfoTooltipProps) {
  return (
    <span
      className={`group/tooltip relative inline-flex shrink-0 ${className ?? ""}`}
    >
      <button
        type="button"
        aria-label={label}
        className="inline-flex cursor-help items-center justify-center rounded-full text-gray-400 transition-colors hover:text-gray-600 focus-visible:outline-2 focus-visible:outline-offset-1 focus-visible:outline-gray-900 dark:text-gray-500 dark:hover:text-gray-300 dark:focus-visible:outline-gray-100"
      >
        <Info className="h-3.5 w-3.5" />
      </button>

      <span className="pointer-events-none absolute left-0 top-full z-30 mt-1.5 hidden w-64 max-w-xs animate-in fade-in zoom-in-95 duration-150 group-hover/tooltip:block group-focus-within/tooltip:block">
        <span className="relative block rounded-xl border border-gray-800 bg-gray-900 p-3 text-xs font-normal text-gray-100 shadow-xl dark:border-gray-700 dark:bg-gray-950">
          <span className="absolute -top-1 left-2.5 h-2 w-2 rotate-45 border-l border-t border-gray-800 bg-gray-900 dark:border-gray-700 dark:bg-gray-950" />
          <span className="relative z-10 block whitespace-normal break-words leading-relaxed text-gray-200">
            {content}
          </span>
        </span>
      </span>
    </span>
  );
}
```

### Step 1.2 — `ActionButton`, the app's button primitive

**Touch.** `src/components/shared/ActionButton.tsx` (create)

**Do.** The rendered shape is a **fragment**: the `<button>`, then the tooltip as a sibling flex
item. A wrapper element would break `gap-*` on the row that hosts it, and nesting the tooltip
inside the button is invalid HTML (D6).

```tsx
import type { ComponentPropsWithoutRef, ReactNode } from "react";
import { cva, type VariantProps } from "class-variance-authority";
import { Loader2 } from "lucide-react";

import { cn } from "@/lib/utils";
import { InfoTooltip } from "./InfoTooltip";

/**
 * The app's one action button: an icon and a label, in a configurable variant and
 * size, with an optional spinner and an optional info tooltip (prep §5).
 *
 * Built on CVA because `Card`, `GradientContainer` and `ConfirmActionButton`
 * already are — one convention for variants, not two.
 *
 * It is deliberately NOT `formItems/Button`: that one is a `children`-only submit
 * button for forms with no icon slot, and it styles with four tokens that do not
 * exist in `index.css`. It stays as it is (D20).
 *
 * The rendered shape is a fragment so this stays ONE flex item's worth of layout.
 */
const actionButtonVariants = cva(
  "inline-flex items-center justify-center gap-2 rounded-lg border text-sm font-medium transition-colors cursor-pointer disabled:opacity-40 disabled:cursor-not-allowed",
  {
    variants: {
      variant: {
        // newly authored, lifted from `InspectionIntroduction`'s CTA — the app's
        // existing dark primary button.
        primary:
          "border-transparent bg-gray-900 text-white hover:bg-gray-800 dark:bg-gray-100 dark:text-gray-900 dark:hover:bg-gray-200",
        // verbatim — `ReservationHeaderActions.tsx`, the create-inspection container
        neutral:
          "border-gray-300 text-gray-700 dark:border-gray-700 dark:text-gray-300",
        // verbatim — `ReservationHeaderActions.tsx`, the copy-message button
        info: "border-blue-200 dark:border-blue-800 text-blue-600 bg-blue-50 hover:bg-blue-100 dark:bg-blue-950/50 dark:text-blue-400 dark:hover:bg-blue-900/50",
        // verbatim — `ReservationHeaderActions.tsx`, the delete container
        danger: "border-rose-200 dark:border-rose-900 text-rose-600 dark:text-rose-400",
      },
      size: {
        sm: "px-2.5 py-1.5 text-xs",
        md: "px-3 py-2 text-sm",
      },
    },
    defaultVariants: { variant: "neutral", size: "md" },
  },
);

export interface ActionButtonProps
  extends Omit<ComponentPropsWithoutRef<"button">, "className">,
    VariantProps<typeof actionButtonVariants> {
  /** Placed before the label. Sized by the caller; `h-4 w-4` is the house size. */
  icon?: ReactNode;
  /** An already-translated string — never a key name. */
  label: string;
  /** Appends a `Loader2` spinner and disables the button. */
  loading?: boolean;
  /** Explanatory copy for the `InfoTooltip` beside the label. Omit for none. */
  tooltip?: string;
  /** The tooltip trigger's accessible name. Required when `tooltip` is given. */
  tooltipLabel?: string;
  className?: string;
}

export function ActionButton({
  icon,
  label,
  loading = false,
  tooltip,
  tooltipLabel,
  variant,
  size,
  className,
  disabled,
  type = "button",
  ...props
}: ActionButtonProps) {
  return (
    <>
      <button
        type={type}
        disabled={disabled || loading}
        className={cn(actionButtonVariants({ variant, size }), className)}
        {...props}
      >
        {icon}
        <span>{label}</span>
        {loading && <Loader2 className="h-4 w-4 shrink-0 animate-spin" />}
      </button>

      {tooltip && (
        // D6a: desktop-only. `hidden sm:contents` makes the wrapper vanish from the
        // layout at `sm` and up, so the tooltip becomes a real flex item beside the
        // button and the row's `gap` applies to it — and below `sm` there is simply
        // no tooltip. The BUTTON itself is unaffected at every width; only the
        // explanation is desktop-only. Putting this here rather than at each call
        // site is what makes "is this a desktop surface" one decision, not eight.
        <span className="hidden sm:contents">
          <InfoTooltip content={tooltip} label={tooltipLabel ?? label} />
        </span>
      )}
    </>
  );
}
```

### Step 1.3 — `ConfirmButton`, the delete pattern

**Touch.** `src/components/shared/ConfirmButton.tsx` (create)

**Do.** The confirm/cancel pair's classes are **verbatim** from `AssetItemCard`'s delete markup.
`confirmLabel` and `cancelLabel` arrive as resolved strings, so this primitive hardcodes no
user-facing text.

The confirmation step keeps the two colours the inline cell had (D32) — `danger` for a delete,
`success` for the create-inspection step — now that the two *triggers* are identical (D31).

The trigger is `children`, and the toggle reaches it through `cloneElement`. That is what keeps
this a *content* component: the same primitive renders the menu's full-width row and the page
header's `ActionButton` without knowing either exists (D3).

```tsx
import { cloneElement, useState, type ReactElement, type ReactNode } from "react";
import { Check, Loader2 } from "lucide-react";

import { cn } from "@/lib/utils";

export type ConfirmButtonVariant = "danger" | "success";

/**
 * The confirmation step's colour. `danger` is the delete pattern lifted from
 * `AssetItemCard`; `success` is the create-inspection step the previous plan's
 * inline cell rendered emerald. The two triggers are identical (D31), so this is
 * the only thing left that says which action is being confirmed (D32).
 */
const CONFIRM_STEP_CLASSES: Record<ConfirmButtonVariant, string> = {
  // verbatim — `AssetItemCard.tsx`, the confirm button
  danger:
    "flex cursor-pointer items-center gap-1 rounded-lg bg-rose-600 px-2 py-1 text-[11px] font-semibold text-white shadow-xs transition-colors hover:bg-rose-700 disabled:opacity-50",
  success:
    "flex cursor-pointer items-center gap-1 rounded-lg bg-emerald-600 px-2 py-1 text-[11px] font-semibold text-white shadow-xs transition-colors hover:bg-emerald-700 disabled:opacity-50",
};

export interface ConfirmButtonProps {
  /**
   * The trigger. Must accept `onClick` and `disabled` — `ActionButton`, a raw
   * `<button>`, anything that forwards its props.
   */
  children: ReactElement<{
    onClick?: (event: React.MouseEvent) => void;
    disabled?: boolean;
  }>;
  /** Fires on the confirm click. Awaiting it keeps the confirm step busy. */
  onConfirm: () => void | Promise<unknown>;
  /** An already-translated string. */
  confirmLabel: string;
  /** An already-translated string. */
  cancelLabel: string;
  /** The confirmation step's leading icon. Defaults to a check. */
  icon?: ReactNode;
  /** The confirmation step's colour. Defaults to the delete pattern's. */
  variant?: ConfirmButtonVariant;
  /** A pending mutation, from outside — the confirm button spins. */
  isLoading?: boolean;
  /** Also disables the trigger, independent of the mutation. */
  disabled?: boolean;
  /** Applied to the confirmation step's container, so a surface can position it. */
  className?: string;
}

/**
 * The in-place "are you sure" pattern, extracted from `AssetItemCard` so delete
 * looks and behaves identically wherever it exists (prep §1, §5).
 *
 * The confirmation step is unconditional: nothing here can delete on one click.
 * That is what lets D4 style delete like every other button and stay safe.
 */
export function ConfirmButton({
  children,
  onConfirm,
  confirmLabel,
  cancelLabel,
  icon,
  variant = "danger",
  isLoading = false,
  disabled = false,
  className,
}: ConfirmButtonProps) {
  const [isConfirming, setIsConfirming] = useState(false);
  const [isPending, setIsPending] = useState(false);

  const busy = isLoading || isPending;

  const handleConfirm = async () => {
    try {
      setIsPending(true);
      await onConfirm();
    } finally {
      setIsPending(false);
      setIsConfirming(false);
    }
  };

  if (isConfirming) {
    return (
      <div
        className={cn(
          "flex items-center gap-1 animate-in fade-in duration-150",
          className,
        )}
      >
        <button
          type="button"
          onClick={handleConfirm}
          disabled={busy}
          className={CONFIRM_STEP_CLASSES[variant]}
        >
          {busy ? (
            <Loader2 className="h-3 w-3 animate-spin" />
          ) : (
            icon ?? <Check className="h-3 w-3" />
          )}
          <span>{confirmLabel}</span>
        </button>

        {/* verbatim — `AssetItemCard.tsx`, the cancel button */}
        <button
          type="button"
          onClick={() => setIsConfirming(false)}
          disabled={busy}
          className="cursor-pointer rounded-lg bg-gray-100 px-2 py-1 text-[11px] font-semibold text-gray-600 transition-colors hover:bg-gray-200 disabled:opacity-50 dark:bg-gray-800 dark:text-gray-300 dark:hover:bg-gray-700"
        >
          {cancelLabel}
        </button>
      </div>
    );
  }

  return cloneElement(children, {
    onClick: () => setIsConfirming(true),
    disabled: disabled || children.props.disabled,
  });
}
```

### Step 1.4 — `BackLink`

**Touch.** `src/components/shared/BackLink.tsx` (create)

**Do.** Class strings are **verbatim** from `ApartmentOverviewHeader`'s back link.

```tsx
import { ArrowLeft } from "lucide-react";

import { AccessLink } from "@/components/router/AccessLink";
import Typography from "@/components/Typography";
import type { PathDefinition } from "@/lib/router";

type ParamsFor<D extends PathDefinition> = keyof D["params"] extends never
  ? { params?: undefined }
  : { params: Record<string, string | number | number[]> };

export type BackLinkProps<D extends PathDefinition> = {
  to: D;
  /** An already-translated string. The label differs per screen by design. */
  label: string;
} & ParamsFor<D>;

/**
 * The one back affordance (prep §3). `AccessLink`, not `Link`: it keeps the role
 * guard — a viewer who cannot open the target gets plain text rather than a link
 * that ends in a 403 — and the 100 ms hover preload.
 *
 * `label` is a resolved string rather than a key, so the primitive owns no copy.
 */
export function BackLink<D extends PathDefinition>({
  to,
  label,
  ...rest
}: BackLinkProps<D>) {
  const params = (rest as { params?: Record<string, string | number | number[]> })
    .params;

  return (
    <AccessLink
      to={to}
      {...((params === undefined ? {} : { params }) as never)}
      className="group inline-flex items-center gap-1.5 text-gray-500 transition-colors hover:text-gray-900 dark:text-gray-400 dark:hover:text-white"
    >
      <ArrowLeft className="h-3.5 w-3.5 transition-transform group-hover:-translate-x-0.5" />
      <Typography
        type="caption"
        className="font-medium text-gray-500 group-hover:text-gray-900 dark:text-gray-400 dark:group-hover:text-white"
      >
        {label}
      </Typography>
    </AccessLink>
  );
}
```

### Step 1.5 — `SectionCard`

```tsx
import type { ReactNode } from "react";

import Typography from "@/components/Typography";
import { cn } from "@/lib/utils";
import { Card } from "./Card";

interface SectionCardProps {
  /** An already-translated string, rendered as the section's `h3`. */
  title?: string;
  /** Right-aligned in the heading row — an edit affordance, or a status chip. */
  action?: ReactNode;
  children: ReactNode;
  className?: string;
}

/**
 * One content section (prep §3: "section style in general should be the same as in
 * other pages"). `ReservationPage.tsx` repeats the same four-class section wrapper
 * three times; this is that container, once, wrapping the shared `Card`.
 *
 * `Card` itself is NOT modified (D21) — it is used app-wide, and reshaping its
 * padding to suit this page would change every page that uses it.
 *
 * The background is passed explicitly rather than inherited from `Card`'s `white`
 * variant, whose `dark:bg-gray-800/50` references an unregistered token (D12).
 */
export function SectionCard({
  title,
  action,
  children,
  className,
}: SectionCardProps) {
  return (
    <Card
      as="section"
      variant="white"
      padding="md"
      radius="xl"
      className={cn(
        "border-gray-200 bg-white dark:border-gray-800 dark:bg-gray-900",
        className,
      )}
    >
      {(title || action) && (
        <div className="mb-4 flex items-center justify-between gap-4">
          {title && <Typography type="h3">{title}</Typography>}
          {action}
        </div>
      )}
      {children}
    </Card>
  );
}
```
