**Gate:** `npm run gate`

Two new components in `src/components/host/`, plus the apartment lookup that feeds the header.

### Step 3.1 — `ReservationHeader`

**Touch.** `src/components/host/ReservationHeader.tsx` (create)

**Do.** The layout mirrors `ApartmentOverviewHeader` (prep §3: "something similar in the similar
style should be the reservation header as well"). It is **presentational**: `apartment` arrives as
a prop rather than being fetched here, which is what lets phase 6 mount it in a test before any
data resolves (D14).

```tsx
import { useTranslation } from "react-i18next";
import { Building2, Mail, Calendar, MapPin } from "lucide-react";

import Typography from "@/components/Typography";
import { BackLink } from "@/components/shared/BackLink";
import { appPaths } from "@/lib/router";
import type { ApartmentWithLocation } from "@/api/generated/requests/types.gen";
import type { ReservationWithInspection } from "@/api/generated/requests/types.gen";

interface ReservationHeaderProps {
  reservation: ReservationWithInspection;
  /**
   * D15: looked up from `useApartmentsHost`, NOT fetched here. `ReservationWithInspection`
   * carries only `apartmentId`, and the `host` section route already prefetches the
   * apartment list on every navigation into the section — so this is a cache read.
   * `undefined` until it resolves, and also when the list has not loaded at all.
   */
  apartment?: ApartmentWithLocation;
}

function formatInstant(value: string | null | undefined, fallback: string) {
  if (!value) return fallback;

  const parsed = new Date(value);
  if (!Number.isFinite(parsed.getTime())) return fallback;

  return parsed.toLocaleString(undefined, {
    dateStyle: "medium",
    timeStyle: "short",
  });
}

/**
 * The reservation page's header (prep §3). It carries what the guest's name alone
 * cannot: who the guest is, when they arrive, and — the reason it exists — WHICH
 * apartment this is, because the host shares this link and the guest opening it has
 * no other way to know.
 *
 * This is why the page has no separate Guest section (prep §3, D16): the name, the
 * email and the check-in are all here.
 */
export function ReservationHeader({
  reservation,
  apartment,
}: ReservationHeaderProps) {
  const { t } = useTranslation("host");

  return (
    <div className="rounded-xl bg-white p-4 dark:bg-gray-900/50">
      <div className="mb-3">
        {/* `AccessLink`, not `Link`, because `BackLink` uses it: it keeps the role
            guard and the 100 ms hover preload the apartment page's back link has.
            The destination comes from the reservation, not from the URL — the two
            are the same id, and this one is the authoritative copy. */}
        <BackLink
          to={appPaths.hostApartment}
          params={{ id: reservation.apartmentId }}
          label={t("reservationPage.backToApartment")}
        />
      </div>

      <div className="flex flex-col gap-4 sm:flex-row sm:items-center sm:justify-between">
        <div className="min-w-0">
          <Typography
            type="h2"
            className="font-semibold text-gray-900 dark:text-white"
          >
            {reservation.guestName}
          </Typography>

          <div className="mt-1.5 flex flex-col gap-1">
            <div className="flex items-center gap-1.5">
              <Mail className="h-3.5 w-3.5 shrink-0 text-gray-400" />
              <Typography
                type="body-sm"
                className="truncate text-gray-500 dark:text-gray-400"
              >
                {reservation.guestEmail || t("reservationPage.notAvailable")}
              </Typography>
            </div>

            <div className="flex items-center gap-1.5">
              <Calendar className="h-3.5 w-3.5 shrink-0 text-gray-400" />
              <Typography
                type="body-sm"
                className="text-gray-500 dark:text-gray-400"
              >
                {formatInstant(
                  reservation.alternativeCheckInDatetime ||
                    reservation.checkInDatetime,
                  t("reservationPage.notAvailable"),
                )}
              </Typography>
            </div>
          </div>
        </div>

        {/* The apartment block. Rendered only when the lookup resolved: an
            "undefined" address would be worse than no address. */}
        {apartment && (
          <div className="flex items-center gap-3.5 sm:justify-end">
            <div className="flex h-12 w-12 shrink-0 items-center justify-center rounded-lg border border-gray-200 bg-gray-50 text-gray-700 dark:border-gray-700 dark:bg-gray-800 dark:text-gray-300">
              <Building2 className="h-6 w-6" />
            </div>
            <div className="min-w-0">
              <Typography
                type="body-sm"
                className="font-medium text-gray-900 dark:text-white"
              >
                {apartment.name}
              </Typography>
              <div className="mt-0.5 flex items-center gap-1.5">
                <MapPin className="h-3.5 w-3.5 shrink-0 text-gray-400" />
                <Typography
                  type="body-sm"
                  className="truncate text-gray-500 dark:text-gray-400"
                >
                  {apartment.location
                    ? `${apartment.location.name}, ${apartment.address}`
                    : apartment.address}
                </Typography>
              </div>
            </div>
          </div>
        )}
      </div>
    </div>
  );
}
```

**Accept.** `grep -c "ReservationHeader" src/components/host/ReservationHeader.tsx` → `1`
**Verify.** `npm run gate`

### Step 3.2 — `ReservationActionsBar`

**Touch.** `src/components/host/ReservationActionsBar.tsx` (create)

**Do.** The whole file. Prep §3: a block of general actions above the sections, every button
identical, info tooltips on everything, and delete rendered through the extracted `ConfirmButton`
(D3, D4, D17, D18).

```tsx
import { useState } from "react";
import { useTranslation } from "react-i18next";
import { ClipboardPlus, Link as LinkIcon, Trash2 } from "lucide-react";

import { ActionButton } from "@/components/shared/ActionButton";
import { ConfirmButton } from "@/components/shared/ConfirmButton";
import { Popover, PopoverContent, PopoverTrigger } from "@/components/Popover";
import { GuestInspectionMessage } from "./GuestInspectionMessage";
import { useReservationActions } from "@/hooks/useReservationActions";
import type { UseReservationActionsParams } from "@/hooks/useReservationActions";

interface ReservationActionsBarProps {
  reservation: UseReservationActionsParams["reservation"];
  apartmentId: string;
  inspectionId: string | null;
  onDeleted: () => void;
}

/**
 * The reservation page's general actions (prep §3, D17): three actions, one shape
 * each, all rendered through `ActionButton`, with delete going through
 * `ConfirmButton`.
 *
 * There is no "take photo proofs" here, deliberately — that action belongs to the
 * Proof section, and rendering it twice would give one behaviour two triggers with
 * two different disabled states. The previous plan's D16 made the same call.
 *
 * On mobile this is a horizontally scrolling strip (D19), so the tooltips are
 * desktop-only: the `InfoTooltip` panel is absolutely positioned and would be
 * clipped by the strip's overflow, and an info dot on every button is noise on a
 * phone anyway (D6a). The BUTTONS are always rendered — only the explanation is
 * desktop-only.
 *
 * Every button here is `variant="neutral"` / `size="md"` (D31). Prep §3 asks for
 * buttons that "look exactly the same, icon and then text, same color", and colour
 * is the only thing that can make two identical shapes read as different weights.
 * The one colour on the page is the confirm step: rose for delete, emerald for
 * create — which is also what keeps delete safe without being pre-red (D4).
 */
export function ReservationActionsBar({
  reservation,
  apartmentId,
  inspectionId,
  onDeleted,
}: ReservationActionsBarProps) {
  const { t } = useTranslation("host");
  const [isCopyOpen, setIsCopyOpen] = useState(false);

  const {
    hasInspection,
    isDeletable,
    isPast,
    guestInspectionUrl,
    deleteMutation,
    createInspectionMutation,
  } = useReservationActions({
    reservation,
    apartmentId,
    inspectionId,
    onDeleted,
  });

  // Each tooltip is the mutually exclusive NEXT reason the action is unavailable,
  // which is what makes the precedence below load-bearing.
  const createInspectionTooltip = isPast
    ? t("reservationPage.actions.tooltip.hasInspectionPast")
    : hasInspection
      ? t("reservationPage.actions.tooltip.hasInspection")
      : t("reservationPage.actions.tooltip.createInspection");

  const copyTooltip = isPast
    ? t("reservationPage.actions.tooltip.copyMessagePast")
    : t("reservationPage.actions.tooltip.copyGuestMessage");

  const deleteTooltip = isPast
    ? t("reservationPage.actions.tooltip.cannotDeletePast")
    : isDeletable
      ? t("reservationPage.actions.tooltip.delete")
      : t("reservationPage.actions.tooltip.cannotDelete");

  return (
    <div className="flex items-center gap-2 overflow-x-auto pb-1">
      {/* D18: the confirmation step is retained from the previous plan; only the
          trigger's appearance changes — and `variant="success"` is why this confirm
          step is emerald rather than the delete pattern's rose (D32). */}
      <ConfirmButton
        className="shrink-0"
        variant="success"
        confirmLabel={t("reservationActionsCell.confirm.createLabel")}
        cancelLabel={t("reservationPage.actions.confirmCancel")}
        icon={<ClipboardPlus className="h-3 w-3" />}
        isLoading={createInspectionMutation.isPending}
        disabled={isPast || hasInspection}
        onConfirm={() => createInspectionMutation.mutateAsync()}
      >
        <ActionButton
          variant="neutral"
          size="md"
          className="shrink-0"
          icon={<ClipboardPlus className="w-4 h-4 shrink-0" />}
          label={t("reservationPage.actions.createInspection")}
          disabled={isPast || hasInspection}
          tooltip={createInspectionTooltip}
          tooltipLabel={t("reservationPage.actions.createInspection")}
        />
      </ConfirmButton>

      {/* A disabled button inside `PopoverTrigger` would still open the popover —
          the trigger div is the click target — so a past stay renders the same
          button outside a popover instead. */}
      {isPast ? (
        <ActionButton
          variant="neutral"
          size="md"
          className="shrink-0"
          icon={<LinkIcon className="w-4 h-4 shrink-0" />}
          label={t("reservationPage.actions.copyGuestMessage")}
          disabled
        />
      ) : (
        hasInspection && (
          <Popover onOpenChange={setIsCopyOpen}>
            <PopoverTrigger>
              <ActionButton
                variant="neutral"
                size="md"
                icon={<LinkIcon className="w-4 h-4 shrink-0" />}
                label={t("reservationPage.actions.copyGuestMessage")}
                tooltip={copyTooltip}
                tooltipLabel={t("reservationPage.actions.copyGuestMessage")}
              />
            </PopoverTrigger>
            <PopoverContent align="left">
              <GuestInspectionMessage
                guestName={reservation.guestName}
                guestInspectionUrl={guestInspectionUrl}
                onClose={() => setIsCopyOpen(false)}
              />
            </PopoverContent>
          </Popover>
        )
      )}

      {/* D4: styled exactly like the others — safety comes from the confirmation
          step, not from colour. D6b keeps the info tooltip out of the way while
          that step is showing, which is this button's only difference. `variant`
          defaults to `danger` on `ConfirmButton`, so the step itself is rose. */}
      <ConfirmButton
        className="shrink-0"
        confirmLabel={t("reservationActionsCell.confirm.deleteLabel")}
        cancelLabel={t("reservationPage.actions.confirmCancel")}
        icon={<Trash2 className="h-3 w-3" />}
        isLoading={deleteMutation.isPending}
        disabled={isPast || !isDeletable}
        onConfirm={() => deleteMutation.mutateAsync()}
      >
        <ActionButton
          variant="neutral"
          size="md"
          className="shrink-0"
          icon={<Trash2 className="w-4 h-4 shrink-0" />}
          label={t("reservationPage.actions.delete")}
          disabled={isPast || !isDeletable}
          tooltip={deleteTooltip}
          tooltipLabel={t("reservationPage.actions.delete")}
        />
      </ConfirmButton>
    </div>
  );
}
```
