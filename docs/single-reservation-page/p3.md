### Step 3.1 — The proof window helper

**Touch.** `src/lib/helpers/proofWindow.ts` (create)

**Do.** Create the file with exactly this content:

```ts
import type { ReservationWithInspection } from "@/api/generated/requests/types.gen";

/** D9: the window arithmetic, defined once. See `getProofWindowState` below. */
export const PROOF_WINDOW_FALLBACK_HOURS = 4;

/** After check-in the window stays open exactly this long, then closes for good. */
export const PROOF_WINDOW_GRACE_MS = 60 * 60 * 1000;

export type ProofWindowStateName =
  | "NOT_STARTED"
  | "TOO_EARLY"
  | "OPEN"
  | "EXPIRED"
  | "ALREADY_SUBMITTED";

/**
 * Everything the three surfaces need in order to agree about "can this host take
 * shots": the answer, why not, and the numbers the copy needs.
 */
export interface ProofWindowState {
  state: ProofWindowStateName;
  /** `checkInDatetime`, or `alternativeCheckInDatetime` when one is set (D9). */
  effectiveCheckInDatetime: string | null;
  /** Epoch ms of the effective check-in, or `null` when there is no usable date. */
  checkInTime: number | null;
  /** The window's opening instant, in epoch ms, or `null` when there is no check-in. */
  windowStartTime: number | null;
  /** The closing instant — check-in plus one hour — or `null`. */
  windowEndTime: number | null;
  windowHours: number;
  /** Whole hours until the window opens; `0` unless the state is `TOO_EARLY`. */
  hoursUntilWindow: number;
  /** Whether the host has already satisfied the reservation. */
  hasSubmittedProofs: boolean;
  /** True only for `OPEN`. The single answer to "may I take shots". */
  canTakeShots: boolean;
}

/**
 * A reservation as far as this helper is concerned. Structural, not nominal: the
 * page passes a `ReservationWithInspection`, the table passes a `ReservationRow`,
 * and both satisfy this — so neither has to be widened to call it.
 */
export interface ProofWindowSubject {
  checkInDatetime: string | null;
  alternativeCheckInDatetime?: string | null;
  proofWindowHours?: number | null;
  hasPhotoProof?: boolean | null;
  status?: string | null;
  alternativeCheckOutDatetime?: string | null;
}

/**
 * D9: the ONLY computation of "can this host take shots". The table's menu, the
 * reservation page's proof section and the guide page's own precondition all call
 * this. It is the arithmetic that used to be inline in `ReservationActionsCell`
 * (effective check-in; `proofWindowHours ?? 4`; check-in − windowHours through
 * check-in + 1h; the too-early/too-late/already-submitted flags), lifted verbatim
 * so the button that offers the action and the page that refuses it cannot drift.
 *
 * `now` is a parameter, not `Date.now()`, so the caller owns the clock and the
 * behaviour is testable without faking timers. Callers render once with `Date.now()`
 * captured at render — the same thing the old inline version did.
 *
 * ORDER OF THE CHECKS IS THE BEHAVIOUR. `ALREADY_SUBMITTED` is tested before the
 * window bounds: a host who has submitted must see "already submitted" even if the
 * clock has since run past the close, which is what the inline version did.
 */
export function getProofWindowState(
  reservation: ProofWindowSubject,
  now: number,
): ProofWindowState {
  const effectiveCheckInDatetime =
    reservation.alternativeCheckInDatetime || reservation.checkInDatetime || null;

  const parsed = effectiveCheckInDatetime
    ? new Date(effectiveCheckInDatetime).getTime()
    : Number.NaN;

  // Number.isFinite, not a truthiness check: `new Date("garbage").getTime()` is
  // NaN, and an unparseable date must read as "no window", not as epoch 0 —
  // which would make every bound comparison silently false and report OPEN.
  const checkInTime = Number.isFinite(parsed) ? parsed : null;

  const windowHours = reservation.proofWindowHours ?? PROOF_WINDOW_FALLBACK_HOURS;

  const windowStartTime =
    checkInTime === null
      ? null
      : checkInTime - windowHours * 60 * 60 * 1000;

  const windowEndTime =
    checkInTime === null ? null : checkInTime + PROOF_WINDOW_GRACE_MS;

  const hasSubmittedProofs =
    Boolean(reservation.hasPhotoProof) || reservation.status === "COVERED";

  const state = resolveState({
    checkInTime,
    windowStartTime,
    windowEndTime,
    hasSubmittedProofs,
    now,
  });

  const isTooEarly = state === "TOO_EARLY";

  return {
    state,
    effectiveCheckInDatetime,
    checkInTime,
    windowStartTime,
    windowEndTime,
    windowHours,
    hoursUntilWindow:
      isTooEarly && windowStartTime !== null
        ? Math.ceil((windowStartTime - now) / (1000 * 60 * 60))
        : 0,
    hasSubmittedProofs,
    canTakeShots: state === "OPEN",
  };
}

function resolveState({
  checkInTime,
  windowStartTime,
  windowEndTime,
  hasSubmittedProofs,
  now,
}: {
  checkInTime: number | null;
  windowStartTime: number | null;
  windowEndTime: number | null;
  hasSubmittedProofs: boolean;
  now: number;
}): ProofWindowStateName {
  if (hasSubmittedProofs) return "ALREADY_SUBMITTED";

  // No usable check-in: nothing to open a window around.
  if (
    checkInTime === null ||
    windowStartTime === null ||
    windowEndTime === null
  ) {
    return "NOT_STARTED";
  }

  if (now < windowStartTime) return "TOO_EARLY";
  if (now > windowEndTime) return "EXPIRED";

  return "OPEN";
}
```

**Two details that carry the behaviour, and are the reason this is not a one-liner.**

1. **`Number.isFinite(parsed)`, not `Boolean(parsed)`.** The old inline code wrote
   `checkInTime = effectiveCheckIn ? new Date(effectiveCheckIn).getTime() : 0` and then guarded
   every comparison with `Boolean(checkInTime)`. That relies on `0` being falsy to mean "no date".
   It works for a missing date but not for `new Date("not-a-date").getTime()`, which is `NaN` —
   truthy under `Boolean`, and every `<`/`>` against `NaN` is `false`, so the old code would fall
   through to `canTakeShots = true` and offer the camera for a reservation with a corrupt date.
   `Number.isFinite` collapses both cases to "no window".
2. **The check order.** `ALREADY_SUBMITTED` is resolved before the bounds, and `state` is a single
   value rather than three independent booleans — so "too early AND already submitted" has one
   answer, not two, and the copy can never claim the wrong one.

### Step 3.2 — Test the proof window

**Touch.** `src/lib/helpers/proofWindow.test.ts` (create)

**Do.** Write the test with exactly this content:

```ts
import { describe, expect, it } from "vitest";
import {
  PROOF_WINDOW_FALLBACK_HOURS,
  getProofWindowState,
} from "./proofWindow";
import type { ProofWindowSubject } from "./proofWindow";

const HOUR = 60 * 60 * 1000;

/** Anchored, not relative: a fixed instant keeps every assertion readable. */
const CHECK_IN = "2026-09-01T14:00:00.000Z";
const CHECK_IN_MS = new Date(CHECK_IN).getTime();

function at(offsetMs: number): number {
  return CHECK_IN_MS + offsetMs;
}

function subject(overrides: Partial<ProofWindowSubject> = {}): ProofWindowSubject {
  return {
    checkInDatetime: CHECK_IN,
    alternativeCheckInDatetime: null,
    proofWindowHours: 4,
    hasPhotoProof: false,
    status: "PENDING_PROOF",
    ...overrides,
  };
}

describe("getProofWindowState — the four window states", () => {
  it("TOO_EARLY before check-in minus the window, and reports the hours remaining", () => {
    // 6h before check-in, with a 4h window: the window opens in 2h.
    const state = getProofWindowState(subject(), at(-6 * HOUR));

    expect(state.state).toBe("TOO_EARLY");
    expect(state.canTakeShots).toBe(false);
    expect(state.hoursUntilWindow).toBe(2);
    expect(state.windowStartTime).toBe(at(-4 * HOUR));
    expect(state.windowEndTime).toBe(at(HOUR));
  });

  it("OPEN on the opening instant itself, and for the whole window", () => {
    expect(getProofWindowState(subject(), at(-4 * HOUR)).state).toBe("OPEN");
    expect(getProofWindowState(subject(), at(0)).state).toBe("OPEN");
    expect(getProofWindowState(subject(), at(HOUR)).state).toBe("OPEN");
  });

  it("EXPIRED one millisecond past check-in plus the grace hour", () => {
    const state = getProofWindowState(subject(), at(HOUR + 1));

    expect(state.state).toBe("EXPIRED");
    expect(state.canTakeShots).toBe(false);
    expect(state.hoursUntilWindow).toBe(0);
  });

  it("ALREADY_SUBMITTED wins over the window bounds, in both directions", () => {
    // Covered AND inside the window.
    expect(
      getProofWindowState(subject({ status: "COVERED" }), at(0)).state,
    ).toBe("ALREADY_SUBMITTED");

    // Covered AND past the close — still reported as already submitted, not expired.
    expect(
      getProofWindowState(subject({ status: "COVERED" }), at(48 * HOUR)).state,
    ).toBe("ALREADY_SUBMITTED");

    // The flag alone, with a pending status, is enough.
    expect(
      getProofWindowState(subject({ hasPhotoProof: true }), at(0)).state,
    ).toBe("ALREADY_SUBMITTED");

    // Even before the window would have opened.
    expect(
      getProofWindowState(subject({ status: "COVERED" }), at(-99 * HOUR)).state,
    ).toBe("ALREADY_SUBMITTED");
  });
});

describe("getProofWindowState — the effective check-in", () => {
  it("prefers the alternative check-in when one is set", () => {
    const alternative = "2026-09-10T14:00:00.000Z";
    const altMs = new Date(alternative).getTime();

    // The ORIGINAL check-in's window is wide open at this instant…
    const state = getProofWindowState(
      subject({ alternativeCheckInDatetime: alternative }),
      CHECK_IN_MS,
    );

    // …but the alternative is 9 days out, so the window is still ahead.
    expect(state.effectiveCheckInDatetime).toBe(alternative);
    expect(state.checkInTime).toBe(altMs);
    expect(state.state).toBe("TOO_EARLY");
    expect(state.windowStartTime).toBe(altMs - 4 * HOUR);
  });

  it("falls back to the plain check-in when the alternative is null or empty", () => {
    for (const value of [null, "", undefined]) {
      const state = getProofWindowState(
        subject({ alternativeCheckInDatetime: value }),
        CHECK_IN_MS,
      );

      expect(state.effectiveCheckInDatetime, String(value)).toBe(CHECK_IN);
      expect(state.state, String(value)).toBe("OPEN");
    }
  });
});

describe("getProofWindowState — the hours default and degenerate input", () => {
  it("defaults proofWindowHours when the field is missing or null", () => {
    expect(getProofWindowState(subject({ proofWindowHours: null }), at(0)).windowHours).toBe(
      PROOF_WINDOW_FALLBACK_HOURS,
    );

    const noField = subject();
    delete noField.proofWindowHours;

    const state = getProofWindowState(noField, at(0));

    expect(state.windowHours).toBe(PROOF_WINDOW_FALLBACK_HOURS);
    expect(state.windowStartTime).toBe(at(-PROOF_WINDOW_FALLBACK_HOURS * HOUR));
  });

  it("honours a non-default proofWindowHours", () => {
    const state = getProofWindowState(subject({ proofWindowHours: 12 }), at(0));

    expect(state.windowHours).toBe(12);
    expect(state.windowStartTime).toBe(at(-12 * HOUR));
    expect(state.state).toBe("OPEN");
  });

  it("reports NOT_STARTED — never OPEN — for a reservation with no usable check-in", () => {
    // A null check-in and an unparseable one must both read as "no window". The
    // second case is the one the old inline arithmetic got wrong: NaN is truthy,
    // and every comparison against NaN is false, so it fell through to "can take".
    expect(
      getProofWindowState(subject({ checkInDatetime: null }), at(0)).state,
    ).toBe("NOT_STARTED");

    const corrupt = getProofWindowState(
      subject({ checkInDatetime: "not-a-date" }),
      at(0),
    );

    expect(corrupt.state).toBe("NOT_STARTED");
    expect(corrupt.canTakeShots).toBe(false);
    expect(corrupt.checkInTime).toBeNull();
    expect(corrupt.windowStartTime).toBeNull();
    expect(corrupt.windowEndTime).toBeNull();
    expect(corrupt.hoursUntilWindow).toBe(0);
  });
});
```

### Step 3.3 — The alternative-date validator

**Touch.** `src/lib/validations/reservationDates.ts` (create)

**Do.** Create the file with exactly this content:

```ts
import { z } from "zod";

/**
 * D12: the rules for the two alternative date fields, extracted from
 * `CreateReservationForm`'s schema so the reservation page's inline editor and the
 * create form cannot disagree. The strings are i18n KEYS, resolved by the caller's
 * `t()` — the same convention `CreateReservationForm` already uses (its messages
 * are `"createReservationForm.validation.*"` literals).
 *
 * Keys live under `reservationPage.validation.*` because the reservation page is
 * the surface that renders them.
 */
export const ALTERNATIVE_DATE_MESSAGE_KEYS = {
  altCheckInFuture: "reservationPage.validation.altCheckInFuture",
  altCheckOutFuture: "reservationPage.validation.altCheckOutFuture",
  altCheckOutAfterAltCheckIn:
    "reservationPage.validation.altCheckOutAfterAltCheckIn",
  checkInFuture: "reservationPage.validation.checkInFuture",
} as const;

/**
 * Both fields are optional — clearing them is how a host removes an alternative —
 * but a non-empty one must be in the future.
 *
 * The date is compared as an instant, not as a string: `datetime-local` produces
 * `YYYY-MM-DDTHH:mm` with no zone, and `new Date(...)` reads that as local time,
 * which is what the input meant. `CreateReservationForm` compares the same way.
 */
const optionalFutureDatetime = (messageKey: string) =>
  z
    .string()
    .optional()
    .refine((value) => !value || isFutureDatetime(value), messageKey);

export function isFutureDatetime(value: string, now = Date.now()): boolean {
  const parsed = new Date(value).getTime();
  // An unparseable date is NOT "future": it must fail, not pass silently.
  return Number.isFinite(parsed) && parsed > now;
}

export function isAfterDatetime(later: string, earlier: string): boolean {
  const laterMs = new Date(later).getTime();
  const earlierMs = new Date(earlier).getTime();

  if (!Number.isFinite(laterMs) || !Number.isFinite(earlierMs)) return false;

  return laterMs > earlierMs;
}

export const alternativeDatesSchema = z
  .object({
    alternativeCheckInDatetime: optionalFutureDatetime(
      ALTERNATIVE_DATE_MESSAGE_KEYS.altCheckInFuture,
    ),
    alternativeCheckOutDatetime: optionalFutureDatetime(
      ALTERNATIVE_DATE_MESSAGE_KEYS.altCheckOutFuture,
    ),
  })
  .refine(
    (data) => {
      // Only a COMPLETE pair can be ordered. One field alone is legal: a host may
      // record a new check-in now and the check-out later.
      if (
        !data.alternativeCheckInDatetime ||
        !data.alternativeCheckOutDatetime
      ) {
        return true;
      }

      return isAfterDatetime(
        data.alternativeCheckOutDatetime,
        data.alternativeCheckInDatetime,
      );
    },
    {
      message: ALTERNATIVE_DATE_MESSAGE_KEYS.altCheckOutAfterAltCheckIn,
      path: ["alternativeCheckOutDatetime"],
    },
  );

export type AlternativeDatesValues = z.infer<typeof alternativeDatesSchema>;

/** `datetime-local` needs `YYYY-MM-DDTHH:mm` in LOCAL time, not an ISO instant. */
export function toDatetimeLocalValue(iso: string | null | undefined): string {
  if (!iso) return "";

  const date = new Date(iso);
  if (!Number.isFinite(date.getTime())) return "";

  const offsetMs = date.getTimezoneOffset() * 60 * 1000;
  return new Date(date.getTime() - offsetMs).toISOString().slice(0, 16);
}

/** The inverse: a local `datetime-local` value back to the ISO instant the API takes. */
export function fromDatetimeLocalValue(value: string | undefined): string | null {
  if (!value) return null;

  const date = new Date(value);
  if (!Number.isFinite(date.getTime())) return null;

  return date.toISOString();
}
```

**Why the two conversion helpers are here and not in the component.** The API stores an ISO instant
with a zone (`"2026-09-01T14:00:00.000Z"`); a `datetime-local` input reads and writes a
zone-less local string (`"2026-09-01T16:00"` in a UTC+2 locale — the same instant). Round-tripping
those two without a conversion is the classic off-by-timezone bug, and it is pure string maths, so
it belongs beside the rules that validate the result. `toDatetimeLocalValue` is the
`getMinDatetimeLocal` trick already in `CreateReservationForm` (subtract the offset, then slice),
made total — it returns `""` for a null or corrupt instant instead of throwing.

### Step 3.4 — Test the date rules

**Touch.** `src/lib/validations/reservationDates.test.ts` (create)

**Do.** Write the test with exactly this content:

```ts
import { describe, expect, it } from "vitest";
import {
  ALTERNATIVE_DATE_MESSAGE_KEYS,
  alternativeDatesSchema,
  fromDatetimeLocalValue,
  isAfterDatetime,
  isFutureDatetime,
  toDatetimeLocalValue,
} from "./reservationDates";

const FUTURE = "2099-01-01T10:00";
const FUTURE_LATER = "2099-01-05T10:00";
const PAST = "2020-01-01T10:00";

/** The first issue's message, which is the i18n key the form renders. */
function firstMessage(result: ReturnType<typeof alternativeDatesSchema.safeParse>) {
  return result.success ? null : result.error.issues[0].message;
}

describe("the alternative-date rules", () => {
  it("accepts a complete, future-dated, correctly ordered pair", () => {
    const result = alternativeDatesSchema.safeParse({
      alternativeCheckInDatetime: FUTURE,
      alternativeCheckOutDatetime: FUTURE_LATER,
    });

    expect(result.success).toBe(true);
  });

  it("accepts the empty case, because clearing the fields is how an alternative is removed", () => {
    const result = alternativeDatesSchema.safeParse({
      alternativeCheckInDatetime: "",
      alternativeCheckOutDatetime: "",
    });

    expect(result.success).toBe(true);
  });

  it("rejects a past check-in with the future rule's key", () => {
    const result = alternativeDatesSchema.safeParse({
      alternativeCheckInDatetime: PAST,
      alternativeCheckOutDatetime: FUTURE_LATER,
    });

    expect(result.success).toBe(false);
    expect(firstMessage(result)).toBe(
      ALTERNATIVE_DATE_MESSAGE_KEYS.altCheckInFuture,
    );
  });

  it("rejects a past check-out with the future rule's key", () => {
    const result = alternativeDatesSchema.safeParse({
      alternativeCheckInDatetime: FUTURE,
      alternativeCheckOutDatetime: PAST,
    });

    expect(result.success).toBe(false);
    expect(firstMessage(result)).toBe(
      ALTERNATIVE_DATE_MESSAGE_KEYS.altCheckOutFuture,
    );
  });

  it("rejects a check-out at or before the check-in, pinned to the check-out field", () => {
    for (const checkOut of [FUTURE, "2098-12-31T10:00"]) {
      const result = alternativeDatesSchema.safeParse({
        alternativeCheckInDatetime: FUTURE,
        alternativeCheckOutDatetime: checkOut,
      });

      expect(result.success, checkOut).toBe(false);
      expect(result.success ? [] : result.error.issues.map((i) => i.path.join("."))).toEqual([
        "alternativeCheckOutDatetime",
      ]);
    }
  });

  it("accepts a lone field, because an incomplete pair cannot be ordered", () => {
    expect(
      alternativeDatesSchema.safeParse({
        alternativeCheckInDatetime: FUTURE,
        alternativeCheckOutDatetime: "",
      }).success,
    ).toBe(true);

    expect(
      alternativeDatesSchema.safeParse({
        alternativeCheckInDatetime: "",
        alternativeCheckOutDatetime: FUTURE_LATER,
      }).success,
    ).toBe(true);
  });
});

describe("the two datetime predicates", () => {
  it("isFutureDatetime rejects an unparseable value rather than passing it", () => {
    const now = Date.now();

    expect(isFutureDatetime("2099-01-01T10:00", now)).toBe(true);
    expect(isFutureDatetime("2020-01-01T10:00", now)).toBe(false);
    expect(isFutureDatetime("not-a-date", now)).toBe(false);
    expect(isFutureDatetime("", now)).toBe(false);
  });

  it("isAfterDatetime is strict, and false for anything unparseable", () => {
    expect(isAfterDatetime(FUTURE_LATER, FUTURE)).toBe(true);
    expect(isAfterDatetime(FUTURE, FUTURE)).toBe(false);
    expect(isAfterDatetime(FUTURE, FUTURE_LATER)).toBe(false);
    expect(isAfterDatetime("nonsense", FUTURE)).toBe(false);
    expect(isAfterDatetime(FUTURE, "nonsense")).toBe(false);
  });
});

describe("the datetime-local conversion round-trips the instant", () => {
  it("converts an ISO instant to a local datetime-local value and back", () => {
    const iso = "2026-09-01T14:00:00.000Z";

    const local = toDatetimeLocalValue(iso);

    // 16 characters, zone-less: exactly what the input element accepts.
    expect(local).toHaveLength(16);
    expect(local).not.toContain("Z");

    // The round trip returns the SAME instant, which is the property that matters;
    // the intermediate string is deliberately local and will differ by timezone.
    expect(fromDatetimeLocalValue(local)).toBe(iso);
  });

  it("maps a missing or corrupt instant to the empty value, never a crash", () => {
    expect(toDatetimeLocalValue(null)).toBe("");
    expect(toDatetimeLocalValue(undefined)).toBe("");
    expect(toDatetimeLocalValue("")).toBe("");
    expect(toDatetimeLocalValue("not-a-date")).toBe("");

    expect(fromDatetimeLocalValue("")).toBeNull();
    expect(fromDatetimeLocalValue(undefined)).toBeNull();
    expect(fromDatetimeLocalValue("not-a-date")).toBeNull();
  });
});
```

## Phase 3 invariants

- [ ] Both modules are **pure** — no React, no `useQuery`, no generated-client import, no `Date.now()`
      inside a tested code path (the clock is a parameter with a default)
- [ ] `getProofWindowState` is the only place the window arithmetic exists (invariant 9); grep for
      `proofWindowHours` under `src/` returns this module, the page sections, and the generated types —
      **nothing that recomputes a bound**
- [ ] No hardcoded user-facing string: the validator's messages are i18n keys, and no `t()` is called here
- [ ] The validator's message keys are the exact strings step 5.x writes into
      `public/locales/*/host.json` under `reservationPage.validation.*`
- [ ] `CreateReservationForm.tsx` is **not** modified in this phase. Consolidating its inline schema
      onto this module is a real improvement and is deliberately **not** done here — it would put an
      unrequested change to the create form inside a feature about the reservation page. It is
      recorded as `Noticed, untouched:` by the executor instead.
- [ ] Tests live beside the code, matching the repo's existing `src/lib/helpers/assetTypes.test.ts`
      and `src/lib/api/queries/apartment.test.ts` convention

## Note for phase 5

Phase 5's time section renders the two validation messages by calling `t(messageKey)` where
`messageKey` is what zod put in `error.issues[0].message` — the same passthrough
`CreateReservationForm` already does (`error={errors.guestName?.message ? t(errors.guestName.message) : undefined}`).
Do not translate the key inside the validator.
