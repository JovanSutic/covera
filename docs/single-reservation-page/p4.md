(`getProofWindowState`).

repoint the caller, *then* delete. The tree compiles at every step boundary — with one exception,
which is resolved below and is **not** left to the executor.

**Grouping.** Run the steps in order: 4.3 imports 4.2, 4.4 imports 4.3. Steps 4.6's two deletions are

              return (
                <tr
                  key={rowIdx}
                  aria-disabled={disabled}
                  onClick={
                    onRowClick && !disabled

                          // D14: an action inside the row (the kebab, its menu, a

### Step 4.2 — Extract the shared action hook

**Touch.** `src/hooks/useReservationActions.ts` (create)

**Do.** Create the file with exactly this content:

```ts
import { useMutation, useQueryClient } from "@tanstack/react-query";
import { useTranslation } from "react-i18next";
import { toast } from "sonner";

import {
  deleteReservationsById,
  postInspections,
} from "@/api/generated/requests/services.gen";
import { withAuth } from "@/lib/api/api";
import { QUERY_ACTIONS } from "@/lib/api/queryKeys";
import { getProofWindowState } from "@/lib/helpers/proofWindow";
import type { ProofWindowSubject } from "@/lib/helpers/proofWindow";
import { appPaths, buildPath } from "@/lib/router";

/**
 * D16: ONE owner of the four reservation actions, shared by the table's kebab and
 * the reservation page's header. It owns the mutations, the invalidation, the
 * disable rules and the tooltips; each surface owns only its own markup. Without
 * this the two renderings would be two copies of the same behaviour, drifting
 * apart the first time one of them is touched.
 *
 * It deliberately does NOT own the guide-page URL. That path is defined in phase 5
 * (step 5.5) and there is exactly one place that builds it — the page that links to
 * it. A `guidePath` here would be a forward reference into a phase that has not
 * landed, which is the one thing this phase cannot afford.
 */
export interface UseReservationActionsParams {
  reservation: ProofWindowSubject & { id: string; guestName: string };
  apartmentId: string;
  /** The guest-inspection id, when one exists — it is the shared link's only source. */
  inspectionId?: string | null;
  /** Called after a successful delete. The page navigates away; the table does nothing. */
  onDeleted?: () => void;
}

export function useReservationActions({
  reservation,
  apartmentId,
  inspectionId,
  onDeleted,
}: UseReservationActionsParams) {
  const { t } = useTranslation("host");
  const queryClient = useQueryClient();

  // D9: the window answer comes from the helper, never from arithmetic here.
  // `Date.now()` is captured at render, matching the inline version this replaced.
  const now = Date.now();
  const window = getProofWindowState(reservation, now);

  const hasInspection = Boolean(inspectionId);

  // Deletion is allowed strictly before the effective check-in — the rule the
  // inline cell had, now read off the helper's own effective value.
  const isDeletable = window.checkInTime !== null && window.checkInTime > now;

  const guestInspectionUrl = inspectionId
    ? buildPath(appPaths.inspection, { id: inspectionId })
    : "";

  const invalidateReservations = () => {
    queryClient.invalidateQueries({
      queryKey: [...QUERY_ACTIONS.RESERVATIONS_GET_BY_APARTMENT, apartmentId],
    });
    // The single reservation too: after a delete the detail entry is a stale 404
    // waiting to be read, and after a create its `inspection` field has appeared.
    queryClient.invalidateQueries({
      queryKey: [...QUERY_ACTIONS.RESERVATIONS_GET_ID, reservation.id],
    });
  };

  const deleteMutation = useMutation({
    mutationFn: async () => {
      const config = await withAuth();
      await deleteReservationsById({ ...config, path: { id: reservation.id } });
    },
    onSuccess: () => {
      invalidateReservations();
      toast.success(t("reservationActionsCell.toast.deleteSuccess"));
      onDeleted?.();
    },
    // eslint-disable-next-line @typescript-eslint/no-explicit-any
    onError: (error: any) => {
      toast.error(
        error?.message || t("reservationActionsCell.toast.deleteError"),
      );
    },
  });

  const createInspectionMutation = useMutation({
    mutationFn: async () => {
      const config = await withAuth();
      const response = await postInspections({
        ...config,
        body: { reservationId: reservation.id },
      });
      return response.data;
    },
    onSuccess: () => {
      invalidateReservations();
      toast.success(t("reservationActionsCell.toast.createSuccess"));
    },
    // eslint-disable-next-line @typescript-eslint/no-explicit-any
    onError: (error: any) => {
      toast.error(
        error?.message || t("reservationActionsCell.toast.createError"),
      );
    },
  });

  /** The camera button's tooltip, derived from the helper's state — never recomputed. */
  const takeShotsTooltip = (): string => {
    switch (window.state) {
      case "ALREADY_SUBMITTED":
        return t("reservationActionsCell.tooltip.hasSubmittedProofs");
      case "EXPIRED":
        return t("reservationActionsCell.tooltip.isTooLate");
      case "TOO_EARLY":
        return t("reservationActionsCell.tooltip.isTooEarly", {
          windowHours: window.windowHours,
          hoursUntilWindow: window.hoursUntilWindow,
        });
      case "NOT_STARTED":
        return t("reservationActionsCell.tooltip.notStarted");
      case "OPEN":
        return t("reservationActionsCell.tooltip.active");
    }
  };

  return {
    window,
    hasInspection,
    isDeletable,
    guestInspectionUrl,
    takeShotsTooltip,
    deleteMutation,
    createInspectionMutation,
  };
}
```

**`notStarted` is one new key**, added in step 4.7. The helper has a `NOT_STARTED` state the inline
code could not express — it collapsed "no check-in" into "not deletable, not shootable" with no
message of its own.

### Step 4.3 — The kebab menu

**Touch.** `src/components/host/ReservationActionsMenu.tsx` (create)

**Do.** Create the file with exactly this content:

```tsx
import { Camera, ClipboardPlus, Link as LinkIcon, MoreVertical, Trash2 } from "lucide-react";
import { useTranslation } from "react-i18next";

import { Popover, PopoverContent, PopoverTrigger } from "@/components/Popover";
import { ConfirmActionButton } from "@/components/ConfirmationButton";
import { GuestInspectionMessage } from "./GuestInspectionMessage";
import { useReservationActions } from "@/hooks/useReservationActions";
import type { ReservationRow } from "@/types/component.types";

interface ReservationActionsMenuProps {
  reservation: ReservationRow;
  apartmentId: string;
  onTakeProofs: (reservation: ReservationRow) => void;
}

const ITEM_CLASSES =
  "w-full flex items-center gap-2.5 px-3 py-2 text-sm text-left text-gray-700 dark:text-gray-300 hover:bg-gray-50 dark:hover:bg-gray-800 disabled:opacity-40 disabled:cursor-not-allowed disabled:hover:bg-transparent cursor-pointer transition-colors";

/**
 * D13/D16: the table's action cell. The kebab replaces the four inline icons the
 * prep asks to retire; the actions themselves are unchanged, because they come
 * from the shared hook. Each keeps its current enabled/disabled rule and tooltip
 * (`reservationActionsCell.*`), which is what the prep asks for.
 */
export function ReservationActionsMenu({
  reservation,
  apartmentId,
  onTakeProofs,
}: ReservationActionsMenuProps) {
  const { t } = useTranslation("host");

  const {
    window,
    hasInspection,
    isDeletable,
    guestInspectionUrl,
    takeShotsTooltip,
    deleteMutation,
    createInspectionMutation,
  } = useReservationActions({
    reservation,
    apartmentId,
    inspectionId: reservation.inspection?.id ?? null,
  });

  return (
    <div
      className="flex items-center justify-end"
      // D14: the whole cell opts out of the row click, so opening the menu never
      // navigates to the reservation the menu is about.
      data-row-click-stop
    >
      <Popover>
        <PopoverTrigger>
          <button
            type="button"
            title={t("reservationActionsMenu.triggerTitle")}
            className="p-1.5 rounded-lg text-gray-500 hover:text-gray-700 hover:bg-gray-100 dark:text-gray-400 dark:hover:text-gray-200 dark:hover:bg-gray-800 transition-colors cursor-pointer"
          >
            <MoreVertical className="w-4 h-4" />
          </button>
        </PopoverTrigger>

        <PopoverContent align="right">
          <div className="w-64 py-1">
            <button
              type="button"
              disabled={!window.canTakeShots}
              title={takeShotsTooltip()}
              onClick={() => window.canTakeShots && onTakeProofs(reservation)}
              className={ITEM_CLASSES}
            >
              <Camera className="w-4 h-4 shrink-0" />
              {t("reservationActionsMenu.takeProofs")}
            </button>

            {/* D13: confirmation is RETAINED, not dropped. ConfirmActionButton swaps
                its icon for a Yes/No pair in place; at w-64 that fits when it carries
                no confirmMessage. Keeping it is what makes this a faithful extraction
                of the inline cell rather than a behaviour change smuggled into a
                layout change. */}
            <div className="px-3 py-2 flex items-center justify-between gap-2">
              <span className="flex items-center gap-2.5 text-sm text-gray-700 dark:text-gray-300">
                <ClipboardPlus className="w-4 h-4 shrink-0" />
                {t("reservationActionsMenu.createInspection")}
              </span>
              <ConfirmActionButton
                icon={<ClipboardPlus className="w-4 h-4" />}
                variant="success"
                confirmLabel={t("reservationActionsCell.confirm.createLabel")}
                disabled={hasInspection}
                title={
                  hasInspection
                    ? t("reservationActionsCell.tooltip.hasInspection")
                    : t("reservationActionsCell.tooltip.createInspection")
                }
                isLoading={createInspectionMutation.isPending}
                onConfirm={() => createInspectionMutation.mutateAsync()}
              />
            </div>

            {hasInspection && (
              <Popover>
                <PopoverTrigger>
                  <button type="button" className={ITEM_CLASSES}>
                    <LinkIcon className="w-4 h-4 shrink-0" />
                    {t("reservationActionsMenu.copyGuestMessage")}
                  </button>
                </PopoverTrigger>
                <PopoverContent align="left">
                  <GuestInspectionMessage
                    guestName={reservation.guestName}
                    guestInspectionUrl={guestInspectionUrl}
                  />
                </PopoverContent>
              </Popover>
            )}

            <div className="px-3 py-2 flex items-center justify-between gap-2">
              <span className="flex items-center gap-2.5 text-sm text-rose-600 dark:text-rose-400">
                <Trash2 className="w-4 h-4 shrink-0" />
                {t("reservationActionsMenu.delete")}
              </span>
              <ConfirmActionButton
                icon={<Trash2 className="w-4 h-4" />}
                variant="danger"
                confirmLabel={t("reservationActionsCell.confirm.deleteLabel")}
                disabled={!isDeletable}
                title={
                  isDeletable
                    ? t("reservationActionsCell.tooltip.delete")
                    : t("reservationActionsCell.tooltip.cannotDelete")
                }
                isLoading={deleteMutation.isPending}
                onConfirm={() => deleteMutation.mutateAsync()}
              />
            </div>
          </div>
        </PopoverContent>
      </Popover>
    </div>
  );
}
```

**Why the two confirmed actions keep a label-span *and* a `ConfirmActionButton`.** The inline cell was
icon-only, so an icon doubling as the trigger read fine. In a menu the row needs words. The
`ConfirmActionButton` stays as the trigger-and-confirm pair so the two-click safety on the destructive
action is unchanged; the label span beside it is what the menu adds. This is the one place the new
component is not a literal copy, and it is a layout accommodation, not a behaviour change.

**`onTakeProofs` is a callback, not a link**, because phase 4 has no guide page yet (the guide route
lands in phase 5 step 5.5, and its real page in phase 6). Phase 6 replaces `IndividualApartment`'s
handler with a navigation; this prop shape survives.

```tsx
/* eslint-disable @typescript-eslint/no-explicit-any */
import { useCallback, useMemo, useState } from "react";
import { useNavigate } from "react-router";
import { useTranslation } from "react-i18next";

import Typography from "@/components/Typography";
import { DataTable } from "@/components/DataTable";
import { AccessLink } from "@/components/router/AccessLink";
import { useReservations } from "@/lib/api/queries/reservations";
import { appPaths, buildPath } from "@/lib/router";
import type { ColumnDef, ReservationRow } from "@/types/component.types";
import { ReservationActionsMenu } from "./ReservationActionsMenu";

interface ReservationsTableProps {
  apartmentId: string;
  onOpenCreateReservation?: () => void;
  onTakeProofs?: (reservation: ReservationRow) => void;
}

const STATUS_BADGE_CLASSES: Record<string, string> = {
  PENDING_PROOF:
    "bg-amber-100 text-amber-800 dark:bg-amber-900/30 dark:text-amber-300 border-amber-200 dark:border-amber-800",
  COVERED:
    "bg-emerald-100 text-emerald-800 dark:bg-emerald-900/30 dark:text-emerald-300 border-emerald-200 dark:border-emerald-800",
  DISPUTED:
    "bg-rose-100 text-rose-800 dark:bg-rose-900/30 dark:text-rose-300 border-rose-200 dark:border-rose-800",
  RESOLVED:
    "bg-blue-100 text-blue-800 dark:bg-blue-900/30 dark:text-blue-300 border-blue-200 dark:border-blue-800",
  CLOSED:
    "bg-gray-100 text-gray-800 dark:bg-gray-800 dark:text-gray-300 border-gray-200 dark:border-gray-700",
};

/**
 * D15: `ReservationsSection`'s view and query, extracted and renamed. It owns the
 * page/limit state and the read — the same `useReservations` call the route
 * prefetches, so this is a cache read, not a second request.
 */
export function ReservationsTable({
  apartmentId,
  onOpenCreateReservation,
  onTakeProofs,
}: ReservationsTableProps) {
  const { t } = useTranslation("host");
  const navigate = useNavigate();
  const [page, setPage] = useState(1);
  const [limit, setLimit] = useState(10);

  const {
    data: reservationsResponse,
    isLoading,
    isError,
    error,
  } = useReservations({ apartmentId, page, limit });

  const isPastDate = (dateString: string): boolean => {
    if (!dateString) return false;
    return new Date(dateString).getTime() < Date.now();
  };

  const reservations = useMemo<ReservationRow[]>(() => {
    if (!reservationsResponse) return [];
    return Array.isArray(reservationsResponse)
      ? reservationsResponse
      : (reservationsResponse as any)?.data || [];
  }, [reservationsResponse]);

  const formatDate = useCallback((isoString: string) => {
    if (!isoString) return "-";
    return new Date(isoString).toLocaleDateString(undefined, {
      month: "short",
      day: "numeric",
      year: "numeric",
      hour: "2-digit",
      minute: "2-digit",
    });
  }, []);

  // D14: the row click. `AccessLink` in the guest cell is what gives the row a real
  // <a> — keyboard, middle-click and the 100 ms hover preload — while this handler
  // covers the rest of the row's width.
  const openReservation = useCallback(
    (reservation: ReservationRow) => {
      navigate(
        buildPath(appPaths.hostReservation, {
          id: apartmentId,
          reservationId: reservation.id,
        }),
      );
    },
    [apartmentId, navigate],
  );

  const columns = useMemo<ColumnDef<ReservationRow>[]>(
    () => [
      {
        header: t("apartmentReservationsManager.columns.guest"),
        accessorKey: (row) => (
          <div>
            <AccessLink
              to={appPaths.hostReservation}
              params={{ id: apartmentId, reservationId: row.id }}
              className="font-medium text-gray-900 dark:text-gray-100 hover:underline"
            >
              {row.guestName}
            </AccessLink>
            {row.guestEmail && (
              <div className="text-xs text-gray-500 dark:text-gray-400">
                {row.guestEmail}
              </div>
            )}
          </div>
        ),
      },
      {
        header: t("apartmentReservationsManager.columns.checkIn"),
        accessorKey: (row) => formatDate(row.checkInDatetime),
      },
      {
        header: t("apartmentReservationsManager.columns.checkOut"),
        accessorKey: (row) => formatDate(row.checkOutDatetime),
      },
      {
        header: t("apartmentReservationsManager.columns.platformId"),
        className: "font-mono text-xs text-gray-500 dark:text-gray-400",
        accessorKey: (row) => row.platformReservationId || "-",
      },
      {
        header: t("apartmentReservationsManager.columns.status"),
        accessorKey: (row) => {
          const statusLabel = t(
            `apartmentReservationsManager.status.${row.status}`,
            { defaultValue: row.status?.replace(/_/g, " ") },
          );

          return (
            <span
              className={`inline-flex items-center px-2.5 py-0.5 rounded-full text-xs font-medium border ${
                STATUS_BADGE_CLASSES[row.status] ||
                "bg-gray-100 text-gray-800 border-gray-200"
              }`}
            >
              {statusLabel}
            </span>
          );
        },
      },
      {
        header: "",
        className: "text-right w-20",
        accessorKey: (row) => (
          <ReservationActionsMenu
            reservation={row}
            apartmentId={apartmentId}
            onTakeProofs={(reservation) => onTakeProofs?.(reservation)}
          />
        ),
      },
    ],
    [t, formatDate, apartmentId, onTakeProofs],
  );

  return (
    <div className="space-y-6">
      <div className="flex flex-col sm:flex-row sm:items-center justify-between gap-4">
        <div>
          <Typography type="h3">
            {t("apartmentReservationsManager.title")}
          </Typography>
          <p className="text-sm text-gray-500 dark:text-gray-400 mt-1">
            {t("apartmentReservationsManager.description")}
          </p>
        </div>
        <div className="flex items-center justify-end gap-3 w-full sm:w-auto">
          <button
            type="button"
            onClick={onOpenCreateReservation}
            className="inline-flex items-center justify-center px-4 py-2 text-sm font-medium text-white bg-gray-900 border border-transparent rounded-lg shadow-sm hover:bg-gray-800 transition-colors cursor-pointer dark:bg-gray-100 dark:text-gray-900 dark:hover:bg-gray-200"
          >
            {t("apartmentReservationsManager.newReservationButton")}
          </button>
        </div>
      </div>

      {isError && !isLoading && (
        <div className="border border-rose-200 dark:border-rose-900/50 rounded-xl p-6 text-center bg-rose-50/50 dark:bg-rose-950/20">
          <Typography
            type="h4"
            className="text-rose-700 dark:text-rose-400 mb-1"
          >
            {t("apartmentReservationsManager.error.title")}
          </Typography>
          <p className="text-sm text-rose-500 dark:text-rose-400">
            {(error as any)?.message ||
              t("apartmentReservationsManager.error.defaultMessage")}
          </p>
        </div>
      )}

      {!isError && (
        <DataTable
          data={reservations}
          columns={columns}
          isLoading={isLoading}
          isRowDisabled={(reservation) => isPastDate(reservation.checkOutDatetime)}
          onRowClick={openReservation}
          pagination={{
            page,
            limit,
            total: reservationsResponse?.pagination.totalItems || 0,
            onPageChange: setPage,
            onLimitChange: (newLimit: number) => {
              setLimit(newLimit);
              setPage(1);
            },
          }}
          emptyMessage={
            <div className="flex flex-col items-center justify-center py-4">
              <Typography
                type="h4"
                className="text-gray-700 dark:text-gray-300 mb-2"
              >
                {t("apartmentReservationsManager.empty.title")}
              </Typography>
              <p className="text-sm text-gray-500 dark:text-gray-400 max-w-md mx-auto mb-6">
                {t("apartmentReservationsManager.empty.description")}
              </p>
              <button
                type="button"
                onClick={onOpenCreateReservation}
                className="inline-flex items-center justify-center px-4 py-2 text-sm font-medium text-gray-900 dark:text-white bg-white dark:bg-gray-800 border border-gray-300 dark:border-gray-700 rounded-lg shadow-sm hover:bg-gray-50 dark:hover:bg-gray-700 transition-colors cursor-pointer"
              >
                {t("apartmentReservationsManager.newReservationButton")}
              </button>
            </div>
          }
        />
      )}
    </div>
  );
}
```

**The `eslint-disable @typescript-eslint/no-explicit-any` at the top is carried over, not new** — the
file keeps `ReservationsSection`'s `(reservationsResponse as any)?.data` and `(error as any)?.message`
verbatim. Both are pre-existing shapes of the paginated response and the thrown error; tightening them
is out of scope for an extraction.
