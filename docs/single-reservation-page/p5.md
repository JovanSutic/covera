# Phase 5 — the reservation page

**Prerequisite:** phase 3 (`getProofWindowState`, the date validator) and phase 4 (the table navigates
here, and `useReservationActions` exists).
**Gate:** `npm run gate`.

This phase replaces the placeholder page phase 2 created at `src/pages/host/ReservationPage.tsx` and
adds the four sections it composes. **It touches no router file for the reservation route** — the path
definition, the route entry and the placeholder all landed in phase 2, because `tree.test.ts` fails a
path definition that has no route, so the two halves could not be split across phases.

**Step 5.5 does define a new path** (`appPaths.inspectionGuide`) and its route. That is not a
contradiction of the paragraph above; it is the same rule applied to the guide page. Path and route
must land together, and the page module both of them name must exist when `coverage.ts` resolves the
specifier — so 5.5 also writes the guide page's placeholder, which phase 6 replaces. This is what
removes the phase-5 → phase-6 forward reference that would otherwise exist, since phase 5's page
links to the guide.

**Grouping.** 5.1–5.4 are four new components; the only ordering constraint is **5.2 before 5.3**,
because the proof section imports the gallery. 5.5 must precede 5.6 (the page links to the path 5.5
defines). 5.7 is the copy plus the translation run.

### Step 5.1 — The time section

**Touch.** `src/components/host/ReservationTimeSection.tsx` (create)

**Do.** Create the file with exactly this content:

```tsx
/* eslint-disable @typescript-eslint/no-explicit-any */
import { useEffect, useState } from "react";
import { useTranslation } from "react-i18next";
import { useForm } from "react-hook-form";
import { zodResolver } from "@hookform/resolvers/zod";
import { useMutation, useQueryClient } from "@tanstack/react-query";
import { toast } from "sonner";

import Input from "@/components/formItems/Input";
import Button from "@/components/formItems/Button";
import Typography from "@/components/Typography";
import { patchReservationsById } from "@/api/generated/requests/services.gen";
import { withAuth } from "@/lib/api/api";
import { QUERY_ACTIONS } from "@/lib/api/queryKeys";
import {
  alternativeDatesSchema,
  fromDatetimeLocalValue,
  toDatetimeLocalValue,
} from "@/lib/validations/reservationDates";
import type { AlternativeDatesValues } from "@/lib/validations/reservationDates";

interface ReservationTimeSectionProps {
  reservationId: string;
  apartmentId: string;
  checkInDatetime: string;
  checkOutDatetime: string;
  alternativeCheckInDatetime: string | null;
  alternativeCheckOutDatetime: string | null;
  /** The page's answer: alternative dates are editable only before the effective check-in. */
  canEdit: boolean;
}

function formatInstant(
  iso: string | null | undefined,
  fallback: string,
): string {
  if (!iso) return fallback;

  const date = new Date(iso);
  if (!Number.isFinite(date.getTime())) return fallback;

  return date.toLocaleString(undefined, {
    dateStyle: "medium",
    timeStyle: "short",
  });
}

/**
 * The Time section, with the alternative dates editable in place (D12). The page
 * passes `canEdit`; this component does not re-derive it — the effective check-in
 * is `getProofWindowState`'s business, and the page already has that answer
 * (invariant 9).
 */
export function ReservationTimeSection({
  reservationId,
  apartmentId,
  checkInDatetime,
  checkOutDatetime,
  alternativeCheckInDatetime,
  alternativeCheckOutDatetime,
  canEdit,
}: ReservationTimeSectionProps) {
  const { t } = useTranslation("host");
  const queryClient = useQueryClient();
  const [isEditing, setIsEditing] = useState(false);

  const emptyValues: AlternativeDatesValues = {
    alternativeCheckInDatetime: toDatetimeLocalValue(alternativeCheckInDatetime),
    alternativeCheckOutDatetime: toDatetimeLocalValue(
      alternativeCheckOutDatetime,
    ),
  };

  const {
    register,
    handleSubmit,
    reset,
    formState: { errors },
  } = useForm<AlternativeDatesValues>({
    resolver: zodResolver(alternativeDatesSchema),
    defaultValues: emptyValues,
    mode: "onChange",
  });

  // The reservation is the source of truth for the fields, so a refetch must not
  // leave stale text behind. `isEditing` is deliberately NOT a dependency:
  // including it would reset the form to the server values the moment the host
  // opened the editor, discarding what they had typed.
  useEffect(() => {
    reset({
      alternativeCheckInDatetime: toDatetimeLocalValue(
        alternativeCheckInDatetime,
      ),
      alternativeCheckOutDatetime: toDatetimeLocalValue(
        alternativeCheckOutDatetime,
      ),
    });
    // eslint-disable-next-line react-hooks/exhaustive-deps
  }, [alternativeCheckInDatetime, alternativeCheckOutDatetime, reset]);

  const saveMutation = useMutation({
    mutationFn: async (values: AlternativeDatesValues) => {
      const config = await withAuth();

      // No cast: `UpdateReservation` makes every field optional, so a two-field
      // partial is already a legal body. (`CreateReservationForm` casts because
      // the CREATE type marks its dates required-nullable — different type,
      // different problem.)
      const response = await patchReservationsById({
        ...config,
        path: { id: reservationId },
        body: {
          alternativeCheckInDatetime: fromDatetimeLocalValue(
            values.alternativeCheckInDatetime,
          ),
          alternativeCheckOutDatetime: fromDatetimeLocalValue(
            values.alternativeCheckOutDatetime,
          ),
        },
      });
      return response.data;
    },
    onSuccess: () => {
      // Both keys: the list's row shows the effect of the effective dates, and the
      // detail entry is what this page renders.
      queryClient.invalidateQueries({
        queryKey: [...QUERY_ACTIONS.RESERVATIONS_GET_BY_APARTMENT, apartmentId],
      });
      queryClient.invalidateQueries({
        queryKey: [...QUERY_ACTIONS.RESERVATIONS_GET_ID, reservationId],
      });
      toast.success(t("reservationPage.time.toast.saveSuccess"));
      setIsEditing(false);
    },
    onError: (error: any) => {
      toast.error(
        error?.error?.message || t("reservationPage.time.toast.saveError"),
      );
    },
  });

  const onSubmit = (values: AlternativeDatesValues) => {
    saveMutation.mutate(values);
  };

  const fieldRow = (
    labelKey: string,
    value: string | null | undefined,
    emptyKey: string,
  ) => (
    <div>
      <dt className="text-xs font-medium text-gray-500 dark:text-gray-400 uppercase tracking-wide">
        {t(labelKey)}
      </dt>
      <dd className="text-sm text-gray-900 dark:text-gray-100 mt-1">
        {formatInstant(value, t(emptyKey))}
      </dd>
    </div>
  );

  return (
    <section className="border border-gray-200 dark:border-gray-800 rounded-xl p-5 bg-white dark:bg-gray-900">
      <div className="flex items-center justify-between gap-4 mb-4">
        <Typography type="h3">{t("reservationPage.time.title")}</Typography>

        {canEdit && !isEditing && (
          <button
            type="button"
            onClick={() => setIsEditing(true)}
            className="text-xs font-medium text-blue-600 hover:text-blue-700 dark:text-blue-400 dark:hover:text-blue-300 cursor-pointer"
          >
            {t("reservationPage.time.editAlternatives")}
          </button>
        )}
      </div>

      <dl className="grid grid-cols-1 sm:grid-cols-2 gap-4">
        {fieldRow(
          "reservationPage.time.checkIn",
          checkInDatetime,
          "reservationPage.notAvailable",
        )}
        {fieldRow(
          "reservationPage.time.checkOut",
          checkOutDatetime,
          "reservationPage.notAvailable",
        )}
      </dl>

      {isEditing ? (
        <form
          onSubmit={handleSubmit(onSubmit)}
          className="mt-5 pt-5 border-t border-gray-100 dark:border-gray-800 space-y-4"
        >
          <div className="grid grid-cols-1 sm:grid-cols-2 gap-4">
            <Input
              label={t("reservationPage.time.alternativeCheckIn")}
              type="datetime-local"
              error={
                errors.alternativeCheckInDatetime?.message
                  ? t(errors.alternativeCheckInDatetime.message)
                  : undefined
              }
              {...register("alternativeCheckInDatetime")}
            />

            <Input
              label={t("reservationPage.time.alternativeCheckOut")}
              type="datetime-local"
              error={
                errors.alternativeCheckOutDatetime?.message
                  ? t(errors.alternativeCheckOutDatetime.message)
                  : undefined
              }
              {...register("alternativeCheckOutDatetime")}
            />
          </div>

          <div className="flex items-center gap-3">
            <Button type="submit" isLoading={saveMutation.isPending}>
              {saveMutation.isPending
                ? t("reservationPage.time.saving")
                : t("reservationPage.time.save")}
            </Button>

            <button
              type="button"
              onClick={() => {
                reset(emptyValues);
                setIsEditing(false);
              }}
              disabled={saveMutation.isPending}
              className="text-sm font-medium text-gray-500 hover:text-gray-700 dark:text-gray-400 dark:hover:text-gray-200 disabled:opacity-40 cursor-pointer transition-colors"
            >
              {t("reservationPage.time.cancel")}
            </button>
          </div>
        </form>
      ) : (
        <dl className="grid grid-cols-1 sm:grid-cols-2 gap-4 mt-5 pt-5 border-t border-gray-100 dark:border-gray-800">
          {fieldRow(
            "reservationPage.time.alternativeCheckIn",
            alternativeCheckInDatetime,
            "reservationPage.time.noAlternative",
          )}
          {fieldRow(
            "reservationPage.time.alternativeCheckOut",
            alternativeCheckOutDatetime,
            "reservationPage.time.noAlternative",
          )}
        </dl>
      )}
    </section>
  );
}
```

**Three things this component deliberately does not do.**

1. **It does not compute whether editing is allowed.** The page owns that answer, derived from
   `getProofWindowState`, and passes `canEdit`. Two places deciding "has this reservation started" is
   the drift invariant 9 exists to prevent.
2. **It does not build query keys.** The invalidations spread `QUERY_ACTIONS.*` (invariant 5).
3. **It does not translate inside the resolver.** zod carries the i18n key in `message` and the JSX
   calls `t(...)` on it — the passthrough `CreateReservationForm` already uses.

**It takes fields, not the reservation object.** The generated `ReservationWithInspection` and the
hand-written `ReservationRow` disagree on nullability (`alternativeCheckInDatetime: string | null`
versus `?: string`), so a component typed against either one is awkward to feed from the other.
Named primitives sidestep that entirely, and this component needs six values, not a record.

**`Button` disables itself from `isLoading`** (`src/components/formItems/Button.tsx`), so passing
`disabled={saveMutation.isPending}` as well would be redundant.

**Accept.** `npx tsc -b` → exit 0. Plus:
- `grep -c "getProofWindowState" src/components/host/ReservationTimeSection.tsx` → `0`
- `grep -c "QUERY_ACTIONS" src/components/host/ReservationTimeSection.tsx` → `3` (the import and the
  two spreads)
- `grep -c "as any" src/components/host/ReservationTimeSection.tsx` → `0` (the `any` is on the caught
  error only, via the file-level disable)

**Verify.** `npx tsc -b`

### Step 5.2 — The photo gallery

**Touch.** `src/components/host/ReservationPhotoGallery.tsx` (create)

**Do.** Create the file with exactly this content:

```tsx
import { useTranslation } from "react-i18next";
import type { ShotWithAssets } from "@/api/generated/requests/types.gen";
import { RoomPhotoItem } from "@/components/guest/RoomPhotoItem";
import Typography from "@/components/Typography";

/**
 * D11: the photos are rendered by the SAME component the guest sees
 * (`RoomPhotoItem`), grouped by room, unmodified. Only the flag affordance
 * differs: `isFlagDisabled` is passed `true` because flagging an issue is the
 * guest's action, and the host's copy of this view is a record of what the guest
 * submitted.
 */
export function ReservationPhotoGallery({
  shots,
}: {
  shots: ShotWithAssets[];
}) {
  const { t } = useTranslation(["host", "assets"]);

  const roomGroups = new Map<string, ShotWithAssets[]>();
  for (const shot of shots) {
    const existing = roomGroups.get(shot.roomLocation) ?? [];
    existing.push(shot);
    roomGroups.set(shot.roomLocation, existing);
  }

  return (
    <div className="mt-4 space-y-6">
      {[...roomGroups.entries()].map(([roomLocation, roomShots]) => (
        <div key={roomLocation}>
          <Typography type="h4" className="mb-3">
            {t(`assets:roomLocations.${roomLocation}`, {
              defaultValue: roomLocation,
            })}
          </Typography>

          <div className="grid grid-cols-1 md:grid-cols-2 gap-4">
            {roomShots.map((shot) => (
              <RoomPhotoItem key={shot.id} shot={shot} isFlagDisabled />
            ))}
          </div>
        </div>
      ))}
    </div>
  );
}
```

**Why grouping is re-derived here instead of reusing `mapShotsToRoomFlowSteps`.** That helper
(`src/lib/helpers/shots.ts`) returns `RoomFlowStep[]` for the guest's step-by-step flow — it carries
per-room completion counts and takes a `TFunction<"assets">` for labels this component does not show.
Its output is room *steps*, not shots-per-room, so reusing it would mean projecting its result back
down to a photo grid. The five-line group-by in this file is what is actually rendered.

**Accept.** `npx tsc -b` → exit 0. Plus:
- `grep -c "RoomPhotoItem" src/components/host/ReservationPhotoGallery.tsx` → `2` (the import and the
  element)
- `grep -c "isFlagDisabled" src/components/host/ReservationPhotoGallery.tsx` → `1`

### Step 5.3 — The proof section

**Touch.** `src/components/host/ReservationProofSection.tsx` (create)

**Do.** Create the file with exactly this content:

```tsx
import { useState } from "react";
import { useTranslation } from "react-i18next";
import {
  Camera,
  CheckCircle2,
  Clock,
  Eye,
  EyeOff,
  TriangleAlert,
} from "lucide-react";

import Typography from "@/components/Typography";
import { useInspection } from "@/lib/api/queries/inspection";
import { getProofWindowState } from "@/lib/helpers/proofWindow";
import type { DetailedInspection } from "@/api/generated/requests/types.gen";
import type { ReservationWithInspection } from "@/api/generated/requests/types.gen";
import { ReservationPhotoGallery } from "./ReservationPhotoGallery";

interface ReservationProofSectionProps {
  reservation: ReservationWithInspection;
  onTakeProofs: () => void;
}

/**
 * D10: the coverage photos are read ONLY when the host opens the control.
 * `ReservationProofPhotos` is a separate component so the `useInspection` call
 * lives BELOW the open/closed boundary — it cannot mount, and therefore cannot
 * request, until the control is opened. The route declares the `inspection`
 * resource with `params: () => null`, so the prefetch starts nothing either: the
 * two halves of "never fetched on page load".
 *
 * The window answer comes from the helper, never from arithmetic here
 * (invariant 9). This is the one action whose availability is that answer.
 */
export function ReservationProofSection({
  reservation,
  onTakeProofs,
}: ReservationProofSectionProps) {
  const { t } = useTranslation("host");
  const [isPhotosOpen, setIsPhotosOpen] = useState(false);

  const window = getProofWindowState(reservation, Date.now());
  const status = reservation.status ?? "PENDING_PROOF";

  const statusIcon =
    status === "COVERED" ? (
      <CheckCircle2 className="w-3.5 h-3.5" />
    ) : window.state === "EXPIRED" || window.state === "NOT_STARTED" ? (
      <TriangleAlert className="w-3.5 h-3.5" />
    ) : (
      <Clock className="w-3.5 h-3.5" />
    );

  return (
    <section className="border border-gray-200 dark:border-gray-800 rounded-xl p-5 bg-white dark:bg-gray-900">
      <Typography type="h3" className="mb-4">
        {t("reservationPage.proof.title")}
      </Typography>

      <span
        className={`inline-flex items-center gap-1.5 px-2.5 py-1 rounded-full text-xs font-medium border ${
          status === "COVERED"
            ? "bg-emerald-100 text-emerald-800 dark:bg-emerald-900/30 dark:text-emerald-300 border-emerald-200 dark:border-emerald-800"
            : "bg-amber-100 text-amber-800 dark:bg-amber-900/30 dark:text-amber-300 border-amber-200 dark:border-amber-800"
        }`}
      >
        {statusIcon}
        {t(`apartmentReservationsManager.status.${status}`, {
          defaultValue: status.replace(/_/g, " "),
        })}
      </span>

      <div className="mt-4">
        {window.canTakeShots ? (
          <button
            type="button"
            onClick={onTakeProofs}
            className="inline-flex items-center gap-2 px-4 py-2 text-sm font-medium rounded-lg bg-amber-100 text-amber-800 border border-amber-300 hover:bg-amber-200 dark:bg-amber-900/40 dark:text-amber-300 dark:border-amber-700 dark:hover:bg-amber-900/70 transition-colors cursor-pointer"
          >
            <Camera className="w-4 h-4" />
            {t("reservationPage.proof.takeProofs")}
          </button>
        ) : (
          <p className="text-sm text-gray-500 dark:text-gray-400">
            {window.state === "ALREADY_SUBMITTED" &&
              t("reservationPage.proof.alreadySubmitted")}
            {window.state === "TOO_EARLY" &&
              t("reservationPage.proof.tooEarly", {
                windowHours: window.windowHours,
                hoursUntilWindow: window.hoursUntilWindow,
              })}
            {window.state === "EXPIRED" && t("reservationPage.proof.expired")}
            {window.state === "NOT_STARTED" &&
              t("reservationPage.proof.notStarted")}
          </p>
        )}
      </div>

      {/* The collapsed photo control. Rendered only when an inspection exists —
          without one there is nothing to show, and a disabled control would be a
          dead affordance. */}
      {reservation.inspection?.id && (
        <div className="mt-5 pt-5 border-t border-gray-100 dark:border-gray-800">
          <button
            type="button"
            onClick={() => setIsPhotosOpen((open) => !open)}
            className="inline-flex items-center gap-2 text-sm font-medium text-gray-700 dark:text-gray-300 hover:text-gray-900 dark:hover:text-gray-100 cursor-pointer transition-colors"
          >
            {isPhotosOpen ? (
              <EyeOff className="w-4 h-4" />
            ) : (
              <Eye className="w-4 h-4" />
            )}
            {isPhotosOpen
              ? t("reservationPage.proof.hidePhotos")
              : t("reservationPage.proof.showPhotos")}
          </button>

          {isPhotosOpen && (
            <ReservationProofPhotos inspectionId={reservation.inspection.id} />
          )}
        </div>
      )}
    </section>
  );
}

/**
 * Mounted only when the control is open (D10) — the whole reason this is a child
 * component rather than an `enabled` flag on a hook called above it.
 */
function ReservationProofPhotos({ inspectionId }: { inspectionId: string }) {
  const { t } = useTranslation("host");
  const { data, isLoading, isError } = useInspection(inspectionId);

  if (isLoading) {
    return (
      <div className="mt-4 flex items-center gap-2 text-sm text-gray-500 dark:text-gray-400">
        <div className="h-4 w-4 animate-spin rounded-full border-2 border-gray-900 dark:border-gray-100 border-t-transparent" />
        {t("reservationPage.proof.photosLoading")}
      </div>
    );
  }

  if (isError) {
    return (
      <p className="mt-4 text-sm text-rose-500 dark:text-rose-400">
        {t("reservationPage.proof.photosError")}
      </p>
    );
  }

  // `GetInspectionsByIdResponse` is `Inspection | DetailedInspection`, and plain
  // `Inspection` has no `shots` field at all — so the narrow is required, not
  // decorative. `InspectionPage.tsx` performs the same one.
  const shots = (data as DetailedInspection | undefined)?.shots ?? [];

  if (shots.length === 0) {
    return (
      <p className="mt-4 text-sm text-gray-500 dark:text-gray-400">
        {t("reservationPage.proof.photosEmpty")}
      </p>
    );
  }

  return <ReservationPhotoGallery shots={shots} />;
}
```

**The section reads no reservation query.** It is handed one. `useReservation` belongs to the page,
which is also what the route prefetches — a second call here would be a cache read with no benefit and
one more thing to keep in step.

**The state machine is flattened into four sentences rather than reusing
`reservationActionsCell.tooltip.*`.** The tooltips are written for a hover target ("Inspection window
active! Click to capture photo proof"); on a page they read wrong. The *states* are shared and come
from the helper; only the wording differs. `ALREADY_SUBMITTED` gets a sentence the tooltip set never
needed, because a tooltip could only ever describe a disabled button.

  the single call)
- `grep -c "useInspection" src/components/host/ReservationProofSection.tsx` → `2` (the import and the
  call inside the child)
- `grep -c "useReservation\|reservationQueryKey" src/components/host/ReservationProofSection.tsx` → `0`

import { ClipboardPlus, Link as LinkIcon, Trash2 } from "lucide-react";
import { useTranslation } from "react-i18next";

import { Popover, PopoverContent, PopoverTrigger } from "@/components/Popover";
import { ConfirmActionButton } from "@/components/ConfirmationButton";
import { GuestInspectionMessage } from "./GuestInspectionMessage";
import { useReservationActions } from "@/hooks/useReservationActions";
import type { UseReservationActionsParams } from "@/hooks/useReservationActions";

interface ReservationHeaderActionsProps {
  reservation: UseReservationActionsParams["reservation"];
  apartmentId: string;
  onDeleted: () => void;
}

/**
 * D16: the page-header rendering of the same actions the table's kebab holds. Both
 * read `useReservationActions`, so the mutations, the disable rules and the
 * invalidation exist once. The difference is markup only — the prep asks for
 * buttons here rather than icons.

export function ReservationHeaderActions({

  apartmentId,
  onDeleted,

  const { t } = useTranslation("host");

  const {
    hasInspection,
    isDeletable,
    guestInspectionUrl,

  return (
    <div className="flex flex-wrap items-center gap-2">

          onConfirm={() => createInspectionMutation.mutateAsync()}
        />
      </div>

      {hasInspection && (
        <Popover>
          {/* `PopoverTrigger` is itself a <div onClick={toggleOpen}> — the button
              inside it carries NO handler, exactly as phase 4's kebab menu does.
              A second onClick here would bubble and toggle twice. */}
          <PopoverTrigger>

              type="button"
              className="inline-flex items-center gap-2 px-3 py-2 text-sm font-medium rounded-lg border border-blue-200 dark:border-blue-800 text-blue-600 bg-blue-50 hover:bg-blue-100 dark:bg-blue-950/50 dark:text-blue-400 dark:hover:bg-blue-900/50 transition-colors cursor-pointer"

          </PopoverContent>
        </Popover>
      )}

      <div
        className={`inline-flex items-center gap-2 px-3 py-2 text-sm font-medium rounded-lg border ${
          isDeletable
            ? "text-rose-600 border-rose-200 dark:text-rose-400 dark:border-rose-900"
            : "text-gray-400 border-gray-200 dark:text-gray-500 dark:border-gray-800"
        }`}
      >
        <Trash2 className="w-4 h-4 shrink-0" />
